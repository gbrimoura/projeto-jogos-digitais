@tool
extends Node2D

const ATLAS_TILE_SIZE := 32
const DRAW_TILE_SIZE := 64
const SEGMENT_COUNT := 10
const FINAL_SOURCE_HEIGHT := 4
const FINAL_DRAW_HEIGHT := 8

@export var atlas_texture: Texture2D:
	set(value):
		atlas_texture = value
		queue_redraw()

@export var atlas_source_id: int = 0:
	set(value):
		atlas_source_id = value
		queue_redraw()

@export var atlas_coordinates: Vector2i = Vector2i(0, 1):
	set(value):
		atlas_coordinates = value
		queue_redraw()

@export var flip_h: bool = false:
	set(value):
		flip_h = value
		queue_redraw()

# A zero opening height leaves this wall continuous (the default for other callers).
@export var opening_top: float = -1.0:
	set(value):
		opening_top = value
		queue_redraw()

@export var opening_height: float = 0.0:
	set(value):
		opening_height = value
		queue_redraw()

func _enter_tree() -> void:
	texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
	queue_redraw()

func _draw() -> void:
	if atlas_texture == null or atlas_source_id != 0 or atlas_coordinates.x < 0 or atlas_coordinates.y < 0:
		return

	var source_x := atlas_coordinates.x * ATLAS_TILE_SIZE
	var source_y := atlas_coordinates.y * ATLAS_TILE_SIZE
	var full_source := Rect2(source_x, source_y, ATLAS_TILE_SIZE, ATLAS_TILE_SIZE)
	var has_opening := opening_height > 0.0 and opening_top >= 0.0
	if flip_h:
		# Refletir no centro do segmento aponta o acabamento do atlas para dentro.
		draw_set_transform(Vector2(DRAW_TILE_SIZE, 0.0), 0.0, Vector2(-1.0, 1.0))

	for segment_index in SEGMENT_COUNT:
		var destination_y := segment_index * DRAW_TILE_SIZE
		_draw_clipped_segment(
			Rect2(0.0, destination_y, DRAW_TILE_SIZE, DRAW_TILE_SIZE),
			full_source,
			has_opening
		)

	# A última peça reutiliza os quatro pixels superiores da célula, escalados por dois.
	_draw_clipped_segment(
		Rect2(0.0, SEGMENT_COUNT * DRAW_TILE_SIZE, DRAW_TILE_SIZE, FINAL_DRAW_HEIGHT),
		Rect2(source_x, source_y, ATLAS_TILE_SIZE, FINAL_SOURCE_HEIGHT),
		has_opening
	)

	if flip_h:
		draw_set_transform(Vector2.ZERO, 0.0, Vector2.ONE)

func _draw_clipped_segment(destination: Rect2, source: Rect2, has_opening: bool) -> void:
	if not has_opening:
		draw_texture_rect_region(atlas_texture, destination, source)
		return

	var segment_top := destination.position.y
	var segment_bottom := destination.end.y
	var opening_bottom := opening_top + opening_height
	var clipped_opening_top := maxf(opening_top, segment_top)
	var clipped_opening_bottom := minf(opening_bottom, segment_bottom)
	if clipped_opening_bottom <= segment_top or clipped_opening_top >= segment_bottom:
		draw_texture_rect_region(atlas_texture, destination, source)
		return

	var source_per_destination_y := source.size.y / destination.size.y
	if clipped_opening_top > segment_top:
		var upper_height := clipped_opening_top - segment_top
		draw_texture_rect_region(
			atlas_texture,
			Rect2(destination.position.x, segment_top, destination.size.x, upper_height),
			Rect2(source.position.x, source.position.y, source.size.x, upper_height * source_per_destination_y)
		)
	if clipped_opening_bottom < segment_bottom:
		var lower_height := segment_bottom - clipped_opening_bottom
		var lower_source_y := source.position.y + (clipped_opening_bottom - segment_top) * source_per_destination_y
		draw_texture_rect_region(
			atlas_texture,
			Rect2(destination.position.x, clipped_opening_bottom, destination.size.x, lower_height),
			Rect2(source.position.x, lower_source_y, source.size.x, lower_height * source_per_destination_y)
		)

## Porta reutilizável: controla alcance, apresentação e pedido de interação.

signal transition_requested(player: Player)

@export_enum("Exit", "Entrance") var door_role: int = 0
@export var interaction_enabled: bool = true

@onready var _door_visual: Node2D = $SideDoorVisual
@onready var _door_label: Label = $DoorLabel

var _players_in_range: Dictionary = {}
var _room_is_cleared: bool = false
var _transition_busy: bool = false
var _terminal_complete: bool = false
var _failure_message: String = ""
var _room_root: Node

func _ready() -> void:
	_room_root = get_parent()
	body_entered.connect(_on_body_entered)
	body_exited.connect(_on_body_exited)
	_update_presentation()

func set_room_cleared(cleared: bool) -> void:
	if door_role != 0 or _terminal_complete:
		return
	_room_is_cleared = cleared
	if not cleared:
		_transition_busy = false
		_failure_message = ""
	_update_presentation()

func set_transition_busy(busy: bool) -> void:
	if door_role != 0 or _terminal_complete:
		return
	if busy and not _room_is_cleared:
		return
	_transition_busy = busy
	if busy:
		_failure_message = ""
	_update_presentation()

func report_transition_failure(message: String) -> void:
	if door_role != 0 or _terminal_complete:
		return
	_transition_busy = false
	_failure_message = message.strip_edges()
	_update_presentation()

func _unhandled_input(event: InputEvent) -> void:
	if door_role != 0 or not interaction_enabled or _terminal_complete:
		return
	if not event is InputEventKey:
		return
	var key_event := event as InputEventKey
	if key_event.echo or not key_event.is_action_pressed("interact"):
		return
	if _room_root != null \
		and _room_root.has_method("is_gameplay_action_blocked") \
		and bool(_room_root.call("is_gameplay_action_blocked", &"interact")):
		get_viewport().set_input_as_handled()
		return
	if _transition_busy or not _room_is_cleared:
		return
	if not Input.is_action_just_pressed("interact"):
		return

	var player := _get_live_player_in_range()
	if player == null:
		return

	_transition_busy = true
	_update_presentation()
	transition_requested.emit(player)
	get_viewport().set_input_as_handled()

func _on_body_entered(body: Node2D) -> void:
	if body is Player and body.is_in_group("player"):
		_players_in_range[body] = true

func _on_body_exited(body: Node2D) -> void:
	_players_in_range.erase(body)

func is_player_in_range(player: Player) -> bool:
	if door_role != 0 or not interaction_enabled or _terminal_complete:
		return false
	if player == null or not is_instance_valid(player) or not _players_in_range.has(player):
		return false
	return get_overlapping_bodies().has(player)

func _get_live_player_in_range() -> Player:
	for candidate in _players_in_range.keys():
		if not is_instance_valid(candidate):
			continue
		var player := candidate as Player
		if player != null and player.is_in_group("player") and not player.is_dead and is_player_in_range(player):
			return player
	return null

func set_terminal_complete() -> void:
	_terminal_complete = true
	_room_is_cleared = false
	_transition_busy = false
	_failure_message = ""
	_update_presentation()

func _update_presentation() -> void:
	if not is_instance_valid(_door_visual) or not is_instance_valid(_door_label):
		return
	if door_role != 0:
		_door_visual.call("set_aperture_open", false)
		_door_label.text = "ENTRADA\nACESSO BLOQUEADO"
		return
	if _terminal_complete:
		_door_visual.call("set_aperture_open", true)
		_door_label.text = "ETAPA CONCLUÍDA"
		return

	var is_open := _room_is_cleared and not _transition_busy
	_door_visual.call("set_aperture_open", is_open)
	if _transition_busy:
		_door_label.text = "ABRINDO SAÍDA..."
	elif not _room_is_cleared:
		_door_label.text = "SAÍDA BLOQUEADA\nELIMINE OS INIMIGOS"
	elif not _failure_message.is_empty():
		_door_label.text = "%s\nE TENTAR DE NOVO" % _failure_message
	else:
		_door_label.text = "SAÍDA ABERTA\n[E] ENTRAR"

local function rect(img, x, y, w, h, color)
  for py = y, y + h - 1 do
    for px = x, x + w - 1 do
      if px >= 0 and py >= 0 and px < img.width and py < img.height then img:drawPixel(px, py, color) end
    end
  end

local function extract(src, sx, sy, w, h)
  local dst = Image(w, h)
  for y = 0, h - 1 do for x = 0, w - 1 do dst:drawPixel(x, y, src:getPixel(sx + x, sy + y)) end end

local function nearestScale(src, scale)
  local dst = Image(src.width * scale, src.height * scale)
  for y = 0, src.height - 1 do
    for x = 0, src.width - 1 do

      for dy = 0, scale - 1 do for dx = 0, scale - 1 do dst:drawPixel(x * scale + dx, y * scale + dy, pixel) end end

  end
  return dst
end

local function drawText(img, x, y, text, color, scale)
  local cursor = x
  for i = 1, #text do glyph(img, cursor, y, FONT[text:sub(i, i)] or FONT[" "], color, scale); cursor = cursor + 4 * scale end
end
local function drawSceneRow(dst, y0, open)
  rect(dst, 0, y0, 1152, 224, C.void)
  local atlas = Image{fromFile = atlasPath}
  local leftWall = flipHorizontal(nearestScale(extract(atlas, 0, 32, 32, 32), 2))
  local rightWall = flipHorizontal(nearestScale(extract(atlas, 64, 32, 32, 32), 2))
  for _, globalY in ipairs({384, 448, 512, 576}) do
    dst:drawImage(leftWall, 0, y0 + globalY - 376)
    dst:drawImage(rightWall, 1088, y0 + globalY - 376)
  end
  -- The cut is painted as a clear through-hole over the tiled wall, at the same 64x128 geometry as runtime.
  local door = makeDoor(open)
  dst:drawImage(flipHorizontal(door), 0, y0 + 56)
  dst:drawImage(door, 1088, y0 + 56)
  for x = 0, 1087, 64 do
    local tileCol = (x == 0) and 0 or 1
    dst:drawImage(nearestScale(extract(atlas, tileCol * 32, 0, 32, 32), 2), x, y0 + 184)
  end
  dst:drawImage(nearestScale(extract(atlas, 64, 0, 32, 32), 2), 1088, y0 + 184)
  local playerAtlas = Image{fromFile = playerPath}
  dst:drawImage(nearestScale(extract(playerAtlas, 0, 0, 64, 64), 2), 920, y0 + 70)
end
