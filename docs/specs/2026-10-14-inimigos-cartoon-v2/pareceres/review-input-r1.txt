local function rgba(r,g,b,a) return app.pixelColor.rgba(r,g,b,a or 255) end
local function colorComponents(pixel)
  return app.pixelColor.rgbaR(pixel), app.pixelColor.rgbaG(pixel), app.pixelColor.rgbaB(pixel), app.pixelColor.rgbaA(pixel)
end
local function nearestPalette(r,g,b)
  local best, bestDistance = palette[1], math.huge
  for _, c in ipairs(palette) do
    local dr, dg, db = r-c[1], g-c[2], b-c[3]
    local distance = 0.299*dr*dr + 0.587*dg*dg + 0.114*db*db
    if distance < bestDistance then best, bestDistance = c, distance end
  end
  return rgba(best[1],best[2],best[3],255)
end
local function isBackground(r,g,b,a)
  if a < 128 then return true end
  local high, low = math.max(r,g,b), math.min(r,g,b)
  local value = (r+g+b)/3
  -- O carvão da prancha fica perto de cinza #222; outlines azul-pretos
  -- têm cromaticidade maior e ficam preservados dentro da silhueta.
  return high-low <= 9 and value >= 22 and value <= 58
end

local function samplePose(pose)
  local image = Image(64,64,ColorMode.RGB)
  local sampled = {}
  local rx, ry, rw, rh = pose.rect[1], pose.rect[2], pose.rect[3], pose.rect[4]
  for y=0,63 do
    for x=0,63 do
      local sx = math.floor(pose.anchorX + (x-32)*pose.scale)
      local sy = math.floor(pose.baselineY + (y-56)*pose.scale)
      local inside = sx >= rx and sx < rx+rw and sy >= ry and sy < ry+rh and sx >= 0 and sx < source.width and sy >= 0 and sy < source.height
      local r,g,b,a = 0,0,0,0
      if inside then r,g,b,a = colorComponents(source:getPixel(sx,sy)) end
      local tr = pose.threatRect
      local inThreat = tr and sx>=tr[1] and sx<tr[1]+tr[3] and sy>=tr[2] and sy<tr[2]+tr[4]
      sampled[y*64+x] = {r,g,b,a,inThreat}
    end
  end

  -- Flood fill 4-conectado a partir das bordas: remove só carvão exterior.
  local seen, queue = {}, {}
  local head, tail = 1, 0
  local function enqueue(x,y)
    if x<0 or x>63 or y<0 or y>63 then return end
    local index = y*64+x
    if seen[index] then return end
    local c = sampled[index]
    if isBackground(c[1],c[2],c[3],c[4]) then
      seen[index] = true
      tail = tail+1
      queue[tail] = {x,y}
    end
  end
  for x=0,63 do enqueue(x,0); enqueue(x,63) end
  for y=1,62 do enqueue(0,y); enqueue(63,y) end
  while head<=tail do
    local cell = queue[head]; head=head+1
    enqueue(cell[1]-1,cell[2]); enqueue(cell[1]+1,cell[2])
    enqueue(cell[1],cell[2]-1); enqueue(cell[1],cell[2]+1)
  end

  for y=0,63 do
    for x=0,63 do
      local index = y*64+x
      local c = sampled[index]
      if c[4] >= 128 and not seen[index] then
        local orangeThreat = c[5] and c[1]>=150 and c[1]>c[2]*1.22 and c[2]>c[3]*1.15
        if orangeThreat then image:drawPixel(x,y,rgba(254,91,1,255))
        else image:drawPixel(x,y,nearestPalette(c[1],c[2],c[3])) end
      else
        image:drawPixel(x,y,rgba(0,0,0,0))
      end
    end
  end
  return image
end

local function buildAtlas(poses)
  local atlas = Image(64*#poses,64,ColorMode.RGB)
  for i,pose in ipairs(poses) do
    for y=0,63 do for x=0,63 do atlas:drawPixel((i-1)*64+x,y,pose.image:getPixel(x,y)) end end
  end
  return atlas
end
local function saveAseprite(poses,path,layerName)
  local sprite = Sprite(64,64,ColorMode.RGB)
  local layer = sprite.layers[1]
  layer.name = layerName
  for i,pose in ipairs(poses) do
    if i>1 then sprite:newEmptyFrame(i) end
    sprite:newCel(layer,i,pose.image,Point(0,0))
    sprite.frames[i].duration = pose.duration/1000
  end
  sprite:saveAs(path)
end
