# First-floor factory backgrounds

The production background is assembled in `scenes/main.tscn` from one of two 576×324 wall bases, one of two complete left-side machinery modules, and one of two complete right-side modules. The 2× nearest-neighbor render fills the fixed 1152×648 room. The three independent choices yield eight visual combinations while preserving the room’s existing layout and gameplay.

The anonymous visual script reads only `get_prepared_room_layout_seed()` at runtime in both `scenes/main.tscn` and `scenes/second_room.tscn`. Seed bit 0 selects the wall base, bit 1 selects the left module, and bit 2 selects the right module. This is a deterministic cosmetic mapping; combinations can repeat, and it does not change or consume RNG from the room generator. The editor uses exported fallback variants without calling the room script; `second_room.tscn` keeps the maintenance base, cabinets, and press as its fallback, also used if the getter is unavailable.

Production sources and previews:

- `background_floor1.lua` is the reproducible Aseprite source. It also retains the approved full V2 compositions as visual studies.
- `modular/factory-floor1-bases.aseprite` and `modular/factory-floor1-modules.aseprite` contain editable Aseprite layers.
- `modular/factory-floor1-bases-atlas.png` contains the two opaque 576×324 bases stacked vertically. `modular/factory-floor1-modules-atlas.png` contains the compact transparent module regions; exact atlas rectangles and room anchors are in `modular/regions-and-anchors.json`.
- The native ImageTexture and AtlasTexture resources under `resources/environment/factory_floor1_*` feed the scene script directly, without pending PNG import.
- `modular/factory-floor1-procedural-contact.png` is a 4×2 contact sheet of all eight base/left/right combinations; `modular/context-seed-*.png` contains one 1152×648 static context per seed. The sample platforms and character show hierarchy only, not a runtime capture or procedural gameplay promise. `modular/context-captions.txt` maps each cell to its seed and variants.
Re-run from the repository root with Aseprite batch mode. Pass each --script-param before --script because Aseprite executes the script when it encounters --script: Aseprite.exe -b --script-param "root=<repo>" --script-param "mode=bases-only" --script "assets/environment/factory_v1/background_floor1/background_floor1.lua". The bases-only mode refreshes both bases, the base atlas, its native texture resource, metadata, and all eight static contexts while reading existing modules. The default full mode rebuilds the studies and modular assets, then regenerates those same eight contexts.

- The original ImageGen reference is retained intact as `reference-imagegen.png`. Earlier approved V1 and full V2 source/art are preserved under `variants/v1/` and `variants/v2-complete/`; the earlier contact layout is preserved under `variants/modular-contact-before-base-fix/`.

local function pix(img,x,y,c)

local function rect(img,x,y,w,h,c)
  if w<=0 or h<=0 then return end
  local x0,y0=math.max(0,x),math.max(0,y)
  local x1,y1=math.min(img.width-1,x+w-1),math.min(img.height-1,y+h-1)
  for py=y0,y1 do for px=x0,x1 do img:drawPixel(px,py,c) end end
end
local function line(img,x0,y0,x1,y1,c,width)
  width=width or 1
  local dx,sx=math.abs(x1-x0),x0<x1 and 1 or -1
  local dy,sy=-math.abs(y1-y0),y0<y1 and 1 or -1
  local err=dx+dy
  while true do
    local a=math.floor((width-1)/2)
    rect(img,x0-a,y0-a,width,width,c)
    if x0==x1 and y0==y1 then break end
    local e2=2*err
    if e2>=dy then err=err+dy; x0=x0+sx end
    if e2<=dx then err=err+dx; y0=y0+sy end
  end
end
local function disk(img,cx,cy,r,c)
  for y=-r,r do

    rect(img,cx-lim,cy+y,lim*2+1,1,c)
  end
end
local function ring(img,cx,cy,outer,inner,c)
  local o2,i2=outer*outer,inner*inner
  for y=-outer,outer do for x=-outer,outer do
    local d=x*x+y*y
    if d<=o2 and d>=i2 then pix(img,cx+x,cy+y,c) end
  end end
end
local function arc(img,cx,cy,outer,inner,startDeg,endDeg,c)

  end end
end

local function makePalette(sprite)
  local ok,pal=pcall(function()
    local q=Palette(#HEX)
    for i,h in ipairs(HEX) do q:setColor(i-1,Color(tonumber(h:sub(1,2),16),tonumber(h:sub(3,4),16),tonumber(h:sub(5,6),16),255)) end
    return q
  end)
  if ok then pcall(function() sprite.palette=pal end) end
end
local function saveScene(name,layers)
  local spr=Sprite(W,H)
  while #spr.layers>1 do spr:deleteLayer(spr.layers[#spr.layers]) end
  for i,item in ipairs(layers) do
    local layer=i==1 and spr.layers[1] or spr:newLayer()
    layer.name=item.name; layer.opacity=255
    spr:newCel(layer,1).image=item.image
  end
  makePalette(spr)
  spr:saveAs(OUT.."/"..name..".aseprite")
  spr:saveCopyAs(OUT.."/"..name..".png")
end
local function scaled(src,n)
  local dst=Image(src.width*n,src.height*n)
  for y=0,src.height-1 do for x=0,src.width-1 do
    local v=src:getPixel(x,y)
    for dy=0,n-1 do for dx=0,n-1 do dst:drawPixel(x*n+dx,y*n+dy,v) end end
  end end
  return dst
end
local function extract(src,x,y,w,h)
  local dst=Image(w,h)
  for py=0,h-1 do for px=0,w-1 do dst:drawPixel(px,py,src:getPixel(x+px,y+py)) end end
  return dst
end
local function pasteNearest(dst,src,x,y,n) dst:drawImage(scaled(src,n),x,y) end
local function platformStrip(atlas,widthNative,heightNative)
  local strip=Image(widthNative,heightNative)
  local cap=math.min(32,math.floor(widthNative/4))
  strip:drawImage(extract(atlas,0,96,cap,heightNative),0,0)
  local mid=extract(atlas,32,96,32,heightNative)
  local cursor=cap
  while cursor<widthNative-cap do
    local useW=math.min(32,widthNative-cap-cursor)
    strip:drawImage(extract(mid,0,0,useW,heightNative),cursor,0)
    cursor=cursor+useW
  end
  strip:drawImage(extract(atlas,96-cap,96,cap,heightNative),widthNative-cap,0)
  return strip
end

local function cropAlpha(src)
  local x0,y0,x1,y1=src.width,src.height,-1,-1
  for y=0,src.height-1 do for x=0,src.width-1 do
    if pc.rgbaA(src:getPixel(x,y))>0 then
      if x<x0 then x0=x end; if y<y0 then y0=y end
      if x>x1 then x1=x end; if y>y1 then y1=y end
    end
  end end
  if x1<x0 or y1<y0 then return Image(1,1),0,0 end
  local out=Image(x1-x0+1,y1-y0+1)
  for y=0,out.height-1 do for x=0,out.width-1 do
    out:drawPixel(x,y,src:getPixel(x+x0,y+y0))
  end end
  return out,x0,y0
end

local function writeImageTextureTres(path,id,img)
  local f=assert(io.open(path,"w"))
  f:write('[gd_resource type="ImageTexture" load_steps=2 format=3]\n\n')
  f:write('[sub_resource type="Image" id="',id,'"]\n')
  f:write('data = { "data": PackedByteArray(\n')
  local total=img.width*img.height
  local n=0
  for y=0,img.height-1 do for x=0,img.width-1 do
    local c=img:getPixel(x,y)
    f:write(tostring(pc.rgbaR(c)),',',tostring(pc.rgbaG(c)),',',tostring(pc.rgbaB(c)),',',tostring(pc.rgbaA(c)))
    n=n+1
    if n<total then f:write(',') end
    if n%32==0 then f:write('\n') end
  end end
  f:write('\n), "format": "RGBA8", "height": ',tostring(img.height),
    ', "mipmaps": false, "width": ',tostring(img.width),' }\n\n')
  f:write('[resource]\nimage = SubResource("',id,'")\n')
  f:close()
end
