func _ready() -> void:
	motion_mode = CharacterBody2D.MOTION_MODE_FLOATING
	collision_mask = PATROL_MASK
	safe_margin = JUMP_SAFE_MARGIN
	health = maxi(max_health, 1)
	add_to_group("damageable")
	add_to_group("combat_enemy")
	_known_player = get_tree().get_first_node_in_group("player") as PhysicsBody2D
	_name_label.text = display_name
	if not _parent_room_allows_enemy_activation():
		process_mode = Node.PROCESS_MODE_DISABLED
		return

	_configure_physics_queries()
	_jump_helper = JumpSolver.new()
	_jump_helper.call("configure", self, _get_collision_rect())
	_pursuit_helper = PursuitSolver.new()
	_set_attack_visual(&"hidden", 0.0)
	_update_visuals()
	if authored_support_enabled:
		var room := get_parent()
		var binding: Variant = {}
		if room != null and room.has_method("get_prepared_authored_target_binding"):
			binding = room.call("get_prepared_authored_target_binding", self)
		var prepared_position_value: Variant = null
		if typeof(binding) == TYPE_DICTIONARY:
			prepared_position_value = binding.get("position")
		if typeof(prepared_position_value) == TYPE_VECTOR2:
			_placement_pending = true
			var prepared_position: Vector2 = prepared_position_value
			position = prepared_position
			var prepared_support := binding.get("support") as SolidPlatform
			_bind_patrol_support(
				prepared_support,
				float(binding.get("patrol_min_x", global_position.x)),
				float(binding.get("patrol_max_x", global_position.x)),
				bool(binding.get("patrol_enabled", false))
			)
			_placement_pending = false
		else:
			_bind_patrol_support(null, global_position.x, global_position.x, false)
	elif spawn_on_platform:
		# O movimento e a aquisição de combate ficam bloqueados até posição e suporte estarem prontos.
		_placement_pending = true
		_place_on_platform.call_deferred()
	else:
		_warn_patrol_once("nenhum suporte foi vinculado; o alvo permanecerá estacionário")

func _parent_room_allows_enemy_activation() -> bool:
	var room := get_parent()
	if room == null or not room.has_method("is_authored_layout_valid"):
		return true
	return bool(room.call("is_authored_layout_valid"))

func take_damage(amount: int, _source: Node = null) -> bool:
	if _defeated_emitted or health <= 0 or amount <= 0:
		return false

	health = maxi(health - amount, 0)
	_flash_remaining = maxf(flash_duration, 0.0)
	_update_visuals()
	if health == 0:
		_defeated_emitted = true
		_cancel_attack()
		defeated.emit()
		queue_free()
	return true

func _physics_process(delta: float) -> void:
	_update_hit_flash(delta)
	if _defeated_emitted:
		return
	if _placement_pending:
		_stop_patrol()
		return

	var scene_active := _is_inside_current_scene()
	var live_player: PhysicsBody2D = _refresh_live_player()
	var player_active := scene_active and live_player != null
	var physics_step := maxf(delta, 0.0)
	_decay_pursuit_blocks(physics_step)
	if not player_active:
		_cancel_attack()
	if _locomotion_state == LocomotionState.FLIGHT or _locomotion_state == LocomotionState.FALLING:
		_decisions_suspended = not scene_active
		_advance_airborne(physics_step)
		_sync_shockwave_progress()
		return
	if not scene_active:
		_decisions_suspended = true
		if _locomotion_state == LocomotionState.PREPARE:
			_abort_jump_prepare()
		_clear_pursuit_state()
		_stop_patrol()
		_sync_shockwave_progress()
		return
	_decisions_suspended = not scene_active
	if _attack_state != AttackState.IDLE and not _is_alive_player(_locked_player):
		_cancel_attack()
	if _locomotion_state == LocomotionState.LANDING:
		_jump_landing_remaining = maxf(_jump_landing_remaining - physics_step, 0.0)
		velocity = Vector2.ZERO
		_patrol_moving = false
		if _jump_landing_remaining <= 0.0:
			_locomotion_state = LocomotionState.GROUND
		_update_visuals()
		_sync_shockwave_progress()
		return
	if _locomotion_state == LocomotionState.PREPARE:
		_update_jump_prepare(physics_step)
		_sync_shockwave_progress()
		return
	if not _ground_support_is_current():
		_cancel_attack()
		_start_unplanned_fall()
		_advance_airborne(physics_step)
		_sync_shockwave_progress()
		return
	if _attack_state == AttackState.IDLE:
		var previous_cooldown := _jump_cooldown_remaining
		_jump_cooldown_remaining = maxf(_jump_cooldown_remaining - physics_step, 0.0)
		if previous_cooldown > 0.0 and _jump_cooldown_remaining <= 0.0:
			_pursuit_replan_remaining = 0.0
	match _attack_state:
		AttackState.IDLE:
			var detected_player := _find_player_in_attack_area(global_position + attack_center_offset)
			if detected_player != null:
				_known_player = detected_player
				_begin_windup(detected_player)
				return
			if live_player != null:
				_pursue_player(live_player, physics_step)
			elif not _patrol_preconditions_hold():
				_stop_patrol()
			else:
				_advance_patrol(physics_step)
			if live_player == null and _attack_state == AttackState.IDLE and _jump_cooldown_remaining <= 0.0:
				_try_begin_jump()
		AttackState.WINDUP:
			_update_windup(physics_step)
		AttackState.ACTIVE:
			_update_active(physics_step)
		AttackState.RECOVERY:
			_update_recovery(physics_step)
	_sync_shockwave_progress()

func _refresh_live_player() -> PhysicsBody2D:
	var previous_player := _known_player
	if _is_alive_player(_known_player) and _known_player.get_parent() == get_parent():
		return _known_player
	_known_player = null
	for node in get_tree().get_nodes_in_group("player"):
		var candidate := node as PhysicsBody2D
		if candidate == null or candidate.get_parent() != get_parent():
			continue
		if _is_alive_player(candidate):
			_known_player = candidate
			break
	if previous_player != _known_player or _known_player == null:
		_clear_pursuit_state()
	return _known_player

func _clear_pursuit_state() -> void:
	_pursuit_player_support = null

	_pursuit_replan_remaining = 0.0
	_pursuit_blocked_edges.clear()
	_pursuit_stuck_remaining = 0.0
	_pursuit_detour_active = false

func _clear_pursuit_route() -> void:
	_pursuit_goal_support = null

	_pursuit_replan_remaining = 0.0
	_pursuit_stuck_remaining = 0.0
	_pursuit_detour_active = false

func _decay_pursuit_blocks(delta: float) -> void:
	for edge in _pursuit_blocked_edges.keys():
		var remaining := maxf(float(_pursuit_blocked_edges[edge]) - delta, 0.0)
		if remaining <= 0.0:
			_pursuit_blocked_edges.erase(edge)
		else:
			_pursuit_blocked_edges[edge] = remaining

func _collect_pursuit_supports() -> Array[SolidPlatform]:
	var supports: Array[SolidPlatform] = []
	for node in get_tree().get_nodes_in_group("solid_platform"):
		var support := node as SolidPlatform
		if support == null or support.get_parent() != get_parent() or not _is_valid_landing_support(support):
			continue
		supports.append(support)
	return supports

func _pursue_player(player: PhysicsBody2D, delta: float) -> void:
	if _pursuit_helper == null or _jump_helper == null or not _ground_support_is_current():
		_stop_patrol()
		return
	var supports: Array[SolidPlatform] = _collect_pursuit_supports()
	var selected_support: SolidPlatform = _pursuit_helper.call(
		"choose_player_support",
		player,
		supports,
		_jump_helper,
		_pursuit_player_support
	)
	if selected_support != _pursuit_player_support:
		_pursuit_player_support = selected_support
		_pursuit_path.clear()
		_pursuit_goal_support = selected_support
		_pursuit_replan_remaining = 0.0
		_pursuit_stuck_remaining = 0.0
		_pursuit_detour_active = false
	if selected_support == null:
		_stop_patrol()
		return
	_pursuit_replan_remaining = maxf(_pursuit_replan_remaining - delta, 0.0)
	if selected_support == _bound_support and not _pursuit_detour_active:
		_pursuit_goal_support = selected_support
		_pursuit_path.clear()
		_pursue_on_current_support(player, delta, supports)
		return
	if selected_support != _pursuit_goal_support:
		_pursuit_goal_support = selected_support
		_pursuit_path.clear()
		_pursuit_replan_remaining = 0.0
		_pursuit_detour_active = false
	if selected_support == _bound_support and _pursuit_detour_active:
		if _pursuit_path.is_empty() or _pursuit_path[0] != _bound_support:
			if _pursuit_replan_remaining <= 0.0:
				_try_begin_pursuit_detour(selected_support, player, supports)
		if _pursuit_path.size() < 2:
			_stop_patrol()
			return
		_pursue_via_route(player, delta, supports)
		return
	if _pursuit_path.is_empty() or _pursuit_path[0] != _bound_support:
		if _pursuit_replan_remaining <= 0.0:
			var route: Array[SolidPlatform] = _pursuit_helper.call(
				"find_path",
				_bound_support,
				selected_support,
				supports,
				_jump_helper,
				_pursuit_blocked_edges
			)
			if route.size() >= 2:
				_pursuit_path = route
			else:
				_pursuit_path.clear()
				_pursuit_replan_remaining = PURSUIT_REPLAN_INTERVAL
	if _pursuit_path.size() < 2:
		_stop_patrol()
		return
	_pursue_via_route(player, delta, supports)
func _pursue_on_current_support(
	player: PhysicsBody2D,
	delta: float,
	supports: Array[SolidPlatform]
) -> void:
	var player_center := _get_player_collider_center(player)
	var attack_center := global_position + attack_center_offset
	var delta_x := player_center.x - attack_center.x
	var delta_y := player_center.y - attack_center.y
	if absf(delta_x) > PURSUIT_DIRECTION_DEADZONE:
		_patrol_direction = 1 if delta_x > 0.0 else -1
	var radius := _effective_attack_radius()
	var horizontal_attack_reach := sqrt(maxf(radius * radius - delta_y * delta_y, 0.0))
	var desired_x := global_position.x
	var clear_shot := _has_clear_line_of_sight(player, attack_center)
	if absf(delta_x) > maxf(horizontal_attack_reach - PURSUIT_STOP_PADDING, 0.0) or not clear_shot:
		var direction := signf(delta_x)
		if is_zero_approx(direction):
			direction = float(_patrol_direction)
		var stop_reach := horizontal_attack_reach - PURSUIT_STOP_PADDING if clear_shot else 0.0
		desired_x = player_center.x - direction * maxf(stop_reach, 0.0) - attack_center_offset.x
	var moved_x := _move_toward_pursuit_x(desired_x, delta, _bound_support)
	if absf(moved_x) >= 0.1:
		_pursuit_stuck_remaining = 0.0
	elif absf(delta_x) > maxf(horizontal_attack_reach - PURSUIT_STOP_PADDING, 0.0) + PURSUIT_DIRECTION_DEADZONE:
		_pursuit_stuck_remaining += delta
	else:
		_pursuit_stuck_remaining = 0.0
	if _pursuit_stuck_remaining >= PURSUIT_STUCK_TIMEOUT and _pursuit_replan_remaining <= 0.0:
		_try_begin_pursuit_detour(_bound_support, player, supports)
	if _attack_state == AttackState.IDLE:
		_update_visuals()

func _try_begin_pursuit_detour(
	goal_support: SolidPlatform,
	player: PhysicsBody2D,
	supports: Array[SolidPlatform]
) -> bool:
	var toward_player := signf(_get_player_collider_center(player).x - global_position.x)
	if is_zero_approx(toward_player):
		toward_player = float(_patrol_direction)
	var best_score: float = INF
	var best_route: Array[SolidPlatform] = []
	for candidate in supports:
		if candidate == _bound_support or candidate == goal_support or candidate.name == &"Floor":
			continue
		if not bool(_jump_helper.call("can_reach_support", candidate, goal_support)):
			continue
		var data := _get_support_rect(candidate)
		if not bool(data.get("valid", false)):
			continue
		var rect: Rect2 = data.get("rect", Rect2())
		var progress := (rect.get_center().x - global_position.x) * toward_player
		if progress <= 16.0:
			continue
		var approach: Array[SolidPlatform] = _pursuit_helper.call(
			"find_path",
			_bound_support,
			candidate,
			supports,
			_jump_helper,
			_pursuit_blocked_edges
		)
		if approach.size() < 2:
			continue
		var score := absf(_get_player_collider_center(player).x - rect.get_center().x) + float(approach.size()) * 80.0
		if score < best_score:
			best_score = score
			approach.append(goal_support)
			best_route = approach
	if best_route.is_empty():
		_pursuit_replan_remaining = PURSUIT_REPLAN_INTERVAL
		return false
	_pursuit_path = best_route
	_pursuit_goal_support = goal_support
	_pursuit_detour_active = true
	_pursuit_stuck_remaining = 0.0
	_pursuit_replan_remaining = 0.0
	return true

func _pursue_via_route(
	player: PhysicsBody2D,
	delta: float,
	_supports: Array[SolidPlatform]
) -> void:
	var source := _bound_support
	var destination := _pursuit_path[1]
	if source == null or destination == null or not is_instance_valid(destination):
		_pursuit_path.clear()
		_pursuit_replan_remaining = 0.0
		_stop_patrol()
		return
	var target_x := _get_player_collider_center(player).x
	if _pursuit_path.size() > 2:
		var next_rect: Rect2 = _jump_helper.call("support_rect", _pursuit_path[2])
		target_x = next_rect.get_center().x
	var launch_x := _pursuit_launch_x(source, destination, target_x)
	var moved_x := _move_toward_pursuit_x(launch_x, delta, source)
	var launch_distance := absf(global_position.x - launch_x)
	if launch_distance > 2.0:
		if absf(moved_x) >= 0.1:
			_pursuit_stuck_remaining = 0.0
		else:
			_pursuit_stuck_remaining += delta
		if _pursuit_stuck_remaining >= PURSUIT_STUCK_TIMEOUT:
			var blocked_edge: String = _pursuit_helper.call("edge_key", source, destination)
			_pursuit_blocked_edges[blocked_edge] = PURSUIT_EDGE_BLOCK_TIME
			_pursuit_path.clear()
			_pursuit_replan_remaining = 0.0
			_pursuit_stuck_remaining = 0.0
			_stop_patrol()
		return
	_pursuit_stuck_remaining = 0.0
	if _jump_cooldown_remaining > 0.0:
		return
	if _pursuit_replan_remaining > 0.0:
		return
	var landing_target_x := _get_player_collider_center(player).x
	if _pursuit_path.size() > 2:
		var landing_next_rect: Rect2 = _jump_helper.call("support_rect", _pursuit_path[2])
		landing_target_x = landing_next_rect.get_center().x
	var plan: Dictionary = _jump_helper.call("plan_jump_to", source, destination, landing_target_x)
	_pursuit_replan_remaining = PURSUIT_REPLAN_INTERVAL
	if plan.is_empty():
		var edge: String = _pursuit_helper.call("edge_key", source, destination)
		_pursuit_blocked_edges[edge] = PURSUIT_EDGE_BLOCK_TIME
		_pursuit_path.clear()
		_pursuit_replan_remaining = 0.0
		_stop_patrol()
		return
	_start_jump_prepare(plan)

func _pursuit_launch_x(source: SolidPlatform, destination: SolidPlatform, target_x: float) -> float:
	var source_data := _get_support_rect(source)
	var destination_data := _get_support_rect(destination)
	if not bool(source_data.get("valid", false)) or not bool(destination_data.get("valid", false)):
		return global_position.x
	var source_rect: Rect2 = source_data.get("rect", Rect2())
	var destination_rect: Rect2 = destination_data.get("rect", Rect2())
	var bounds := _pursuit_bounds_for_support(source)
	if destination.name == &"Floor":
		var direction := signf(target_x - source_rect.get_center().x)
		if is_zero_approx(direction):
			var left_distance := absf(global_position.x - bounds.x) + absf(target_x - source_rect.position.x)
			var right_distance := absf(global_position.x - bounds.y) + absf(target_x - source_rect.end.x)
			direction = -1.0 if left_distance <= right_distance else 1.0
		return bounds.y if direction > 0.0 else bounds.x
	if destination_rect.position.y < source_rect.position.y - 1.0:
		var body := _get_collision_rect()
		var left_standoff := destination_rect.position.x - body.end.x - PURSUIT_EDGE_CLEARANCE
		var right_standoff := destination_rect.end.x - body.position.x + PURSUIT_EDGE_CLEARANCE
		var toward_destination := signf(target_x - destination_rect.get_center().x)
		if toward_destination > 0.0 and right_standoff >= bounds.x and right_standoff <= bounds.y:
			return right_standoff
		if toward_destination < 0.0 and left_standoff >= bounds.x and left_standoff <= bounds.y:
			return left_standoff
		var left_valid := left_standoff >= bounds.x and left_standoff <= bounds.y
		var right_valid := right_standoff >= bounds.x and right_standoff <= bounds.y
		if left_valid and right_valid:
			return left_standoff if absf(global_position.x - left_standoff) <= absf(global_position.x - right_standoff) else right_standoff
		if left_valid:
			return left_standoff
		if right_valid:
			return right_standoff
	var direction := signf(destination_rect.get_center().x - global_position.x)
	if is_zero_approx(direction):
		direction = signf(target_x - global_position.x)
	if is_zero_approx(direction):
		direction = float(_patrol_direction)
	return bounds.y if direction > 0.0 else bounds.x

func _pursuit_bounds_for_support(support: SolidPlatform) -> Vector2:
	var data := _get_support_rect(support)
	if not bool(data.get("valid", false)):
		return Vector2(global_position.x, global_position.x)
	var rect: Rect2 = data.get("rect", Rect2())
	var body := _get_collision_rect()
	var minimum_x := rect.position.x + edge_margin - body.position.x
	var maximum_x := rect.end.x - edge_margin - body.end.x
	if maximum_x < minimum_x:
		return Vector2(global_position.x, global_position.x)
	return Vector2(minimum_x, maximum_x)

func _move_toward_pursuit_x(target_x: float, delta: float, support: SolidPlatform) -> float:
	velocity = Vector2.ZERO
	_patrol_moving = false
	if delta <= 0.0 or not is_instance_valid(support):
		return 0.0
	var bounds := _pursuit_bounds_for_support(support)
	var destination_x := clampf(target_x, bounds.x, bounds.y)
	var dx := destination_x - global_position.x
	if absf(dx) <= MOVEMENT_EPSILON:
		_update_visuals()
		return 0.0
	var direction := signf(dx)
	if absf(dx) > PURSUIT_DIRECTION_DEADZONE:
		_patrol_direction = int(direction)
	var requested_dx := direction * minf(absf(dx), PATROL_SPEED * delta)
	var collision: KinematicCollision2D = move_and_collide(Vector2(requested_dx, 0.0), true, 0.0)
	var safe_dx := requested_dx
	if collision != null:
		safe_dx = collision.get_travel().x
	if signf(safe_dx) != signf(requested_dx):
		safe_dx = 0.0
	safe_dx = signf(requested_dx) * minf(absf(safe_dx), absf(requested_dx))
	if absf(safe_dx) >= MOVEMENT_EPSILON:
		global_position.x += safe_dx
		_patrol_moving = true
		velocity.x = safe_dx / delta
	_update_visuals()
	return safe_dx
func _sync_shockwave_progress() -> void:
	if not is_instance_valid(_attack_visual) or not _attack_visual.has_method("set_shockwave_progress"):
		return
	var progress := 0.0
	if _attack_state == AttackState.ACTIVE:
		progress = 1.0 - _state_remaining / maxf(active_window, 0.05)
	_attack_visual.call("set_shockwave_progress", clampf(progress, 0.0, 1.0))

func _ground_support_is_current() -> bool:
	return _bound_support != null and is_instance_valid(_bound_support) \
		and _is_valid_landing_support(_bound_support) and _support_feet_match(_bound_support)

func _support_feet_match(support: SolidPlatform) -> bool:
	var support_data := _get_support_rect(support)
	if not bool(support_data.get("valid", false)):
		return false
	var rect: Rect2 = support_data.get("rect", Rect2())
	return absf(global_position.y + _get_collision_rect().end.y - rect.position.y) <= 1.0

func _try_begin_jump() -> void:
	_jump_cooldown_remaining = JUMP_COOLDOWN
	if _jump_helper == null or not _ground_support_is_current() or _attack_state != AttackState.IDLE:
		return
	var plan: Dictionary = _jump_helper.call("plan_jump", _bound_support, _patrol_direction)
	if plan.is_empty():
		return
	_start_jump_prepare(plan)
func _start_jump_prepare(plan: Dictionary) -> void:
	if plan.is_empty() or _attack_state != AttackState.IDLE or not _ground_support_is_current():
		return
	_jump_plan = plan
	_jump_cooldown_remaining = JUMP_COOLDOWN
	_jump_facing_direction = 1 if float(_jump_plan["landing_x"]) >= global_position.x else -1
	_locomotion_state = LocomotionState.PREPARE
	_state_remaining = JUMP_PREPARE_TIME
	_patrol_moving = false
	velocity = Vector2.ZERO
	_update_visuals()
func _update_jump_prepare(delta: float) -> void:
	var detected_player := _find_player_in_attack_area(global_position + attack_center_offset)
	if detected_player != null:
		_abort_jump_prepare()
		_known_player = detected_player
		_begin_windup(detected_player)
		return
	if not _ground_support_is_current():
		_abort_jump_prepare()
		return
	_state_remaining = maxf(_state_remaining - delta, 0.0)
	if _state_remaining > 0.0:
		_update_visuals()
		return
	detected_player = _find_player_in_attack_area(global_position + attack_center_offset)
	if detected_player != null:
		_abort_jump_prepare()
		_known_player = detected_player
		_begin_windup(detected_player)
		return
	var refreshed: Dictionary = _jump_helper.call("revalidate_plan", _jump_plan, _bound_support)
	if refreshed.is_empty():
		_abort_jump_prepare()
		return
	_jump_plan = refreshed
	_jump_source = _bound_support
	var source_data := _get_support_rect(_jump_source)
	var source_rect: Rect2 = source_data.get("rect", Rect2())
	_jump_source_top = source_rect.position.y
	_jump_facing_direction = 1 if float(_jump_plan["landing_x"]) >= global_position.x else -1
	_patrol_direction = _jump_facing_direction
	_jump_escape_direction = 0
	_jump_elapsed = 0.0
	_jump_impulse_remaining = JUMP_IMPULSE_TIME
	velocity = _jump_plan["launch_velocity"]
	_locomotion_state = LocomotionState.FLIGHT
	_update_visuals()

func _abort_jump_prepare() -> void:
	_jump_plan.clear()
	_state_remaining = 0.0
	_jump_facing_direction = _patrol_direction
	_jump_cooldown_remaining = JUMP_COOLDOWN
	_locomotion_state = LocomotionState.GROUND
	velocity = Vector2.ZERO
	_patrol_moving = false
	_update_visuals()
func _start_unplanned_fall() -> void:
	if _bound_support != null and is_instance_valid(_bound_support):
		remove_collision_exception_with(_bound_support)
	_bound_support = null
	_patrol_enabled = false
	_patrol_moving = false
	_jump_plan.clear()
	_jump_source = null
	_jump_elapsed = 0.0
	_jump_escape_direction = 0
	velocity = Vector2.ZERO
	_locomotion_state = LocomotionState.FALLING

func _advance_airborne(delta: float) -> void:
	if delta <= 0.0:
		return
	_release_jump_source_if_clear()
	if _jump_elapsed >= JUMP_MAX_FLIGHT_TIME and _jump_escape_direction == 0:
		velocity.x = 0.0
	var remaining := delta
	while remaining > 0.00001:
		var final_velocity := velocity + Vector2(0.0, JUMP_GRAVITY * remaining)
		var max_speed := maxf(velocity.length(), final_velocity.length())
		var step_time := minf(remaining, JUMP_MAX_SEGMENT / maxf(max_speed, 1.0))
		# End the source-support exception at the same analytic clearance used by route casts.
		if _locomotion_state == LocomotionState.FLIGHT and is_instance_valid(_jump_source) and velocity.y < 0.0:
			var feet_y := global_position.y + _get_collision_rect().end.y
			var clearance := feet_y - (_jump_source_top - JumpSolver.EXIT_CLEARANCE)
			var discriminant := velocity.y * velocity.y - 2.0 * JUMP_GRAVITY * clearance
			if clearance > 0.0 and discriminant >= 0.0:
				var clear_time := (-velocity.y - sqrt(discriminant)) / JUMP_GRAVITY
				if clear_time > 0.000001:
					step_time = minf(step_time, clear_time)
		var motion := velocity * step_time + Vector2(0.0, 0.5 * JUMP_GRAVITY * step_time * step_time)
		while motion.length() > JUMP_MAX_SEGMENT + 0.0001:
			step_time *= 0.5
			motion = velocity * step_time + Vector2(0.0, 0.5 * JUMP_GRAVITY * step_time * step_time)
		var collision := move_and_collide(motion, false, JUMP_SAFE_MARGIN)
		if collision == null:
			# The executed displacement and velocity now use the same substep duration.
			velocity += Vector2(0.0, JUMP_GRAVITY * step_time)
			remaining -= step_time
			_jump_elapsed += step_time
			if _jump_impulse_remaining > 0.0:
				_jump_impulse_remaining = maxf(_jump_impulse_remaining - step_time, 0.0)
		else:
			var impact_ratio := 1.0
			if motion.length_squared() > 0.000001:
				impact_ratio = clampf(collision.get_travel().length() / motion.length(), 0.0, 1.0)
			var impact_time := maxf(step_time * impact_ratio, minf(step_time, 0.0001))
			impact_time = minf(impact_time, step_time)
			velocity += Vector2(0.0, JUMP_GRAVITY * impact_time)
			var normal := collision.get_normal()
			var into_surface := velocity.dot(normal)
			if into_surface < 0.0:
				velocity -= normal * into_surface
			if normal.dot(Vector2.UP) >= 0.7:
				var collider := collision.get_collider()
				var landing_support := collider as SolidPlatform
				if _try_bind_landing(landing_support):
					return
				if _is_player_collider(collider):
					_begin_player_top_escape(collider as PhysicsBody2D)
				else:
					if _jump_escape_direction == 0:
						_jump_escape_direction = signi(velocity.x) if absf(velocity.x) > 1.0 else _jump_facing_direction
					velocity.x = float(_jump_escape_direction) * PLAYER_ESCAPE_SPEED
					velocity.y = 0.0
				# A diagonal retry at TOI zero can pin against a body top. Move sideways physically for
				# the rest of this tick, counting each part of the tick exactly once.
				_jump_elapsed += impact_time
				if _jump_impulse_remaining > 0.0:
					_jump_impulse_remaining = maxf(_jump_impulse_remaining - impact_time, 0.0)
				_advance_escape_tangent(maxf(remaining - impact_time, 0.0))
				_update_visuals()
				return
			if _jump_escape_direction != 0 and absf(normal.x) >= 0.7:
				_jump_escape_direction *= -1
				velocity.x = float(_jump_escape_direction) * PLAYER_ESCAPE_SPEED
			remaining -= impact_time
			_jump_elapsed += impact_time
			if _jump_impulse_remaining > 0.0:
				_jump_impulse_remaining = maxf(_jump_impulse_remaining - impact_time, 0.0)
		if _jump_elapsed >= JUMP_MAX_FLIGHT_TIME and _jump_escape_direction == 0:
			velocity.x = 0.0
		_release_jump_source_if_clear()
		_update_visuals()

func _advance_escape_tangent(delta: float) -> void:
	var remaining := maxf(delta, 0.0)
	if remaining <= 0.0:
		return
	if _jump_escape_direction == 0:
		_jump_escape_direction = signi(velocity.x) if absf(velocity.x) > 1.0 else _jump_facing_direction
	velocity.x = float(_jump_escape_direction) * PLAYER_ESCAPE_SPEED
	velocity.y = 0.0
	while remaining > 0.00001:
		var step_time := minf(remaining, JUMP_MAX_SEGMENT / PLAYER_ESCAPE_SPEED)
		var tangent_motion := Vector2(velocity.x * step_time, 0.0)
		var collision := move_and_collide(tangent_motion, false, JUMP_SAFE_MARGIN)
		if collision != null and Vector2(velocity.x, 0.0).dot(collision.get_normal()) < -0.0001:
			_jump_escape_direction *= -1
			velocity.x = float(_jump_escape_direction) * PLAYER_ESCAPE_SPEED
		remaining -= step_time
		_jump_elapsed += step_time
		if _jump_impulse_remaining > 0.0:
			_jump_impulse_remaining = maxf(_jump_impulse_remaining - step_time, 0.0)
		if _jump_elapsed >= JUMP_MAX_FLIGHT_TIME and _jump_escape_direction == 0:
			velocity.x = 0.0
		_release_jump_source_if_clear()
	_update_visuals()
func _release_jump_source_if_clear() -> void:
	if _locomotion_state != LocomotionState.FLIGHT or not is_instance_valid(_jump_source):
		return
	var feet_y := global_position.y + _get_collision_rect().end.y
	if feet_y <= _jump_source_top - JumpSolver.EXIT_CLEARANCE or velocity.y >= 0.0:
		remove_collision_exception_with(_jump_source)
		_jump_source = null
func _is_player_collider(collider: Object) -> bool:
	return collider is PhysicsBody2D and (collider as PhysicsBody2D).is_in_group("player")

func _begin_player_top_escape(player_body: PhysicsBody2D) -> void:
	if _jump_escape_direction == 0:
		var away := global_position.x - player_body.global_position.x
		_jump_escape_direction = signi(away) if absf(away) > 1.0 else _jump_facing_direction
	velocity.x = float(_jump_escape_direction) * PLAYER_ESCAPE_SPEED
	if velocity.y > 0.0:
		velocity.y = 0.0

func _try_bind_landing(support: SolidPlatform) -> bool:
	if support == null or not is_instance_valid(support) or support.get_parent() != get_parent() or not _is_compatible_support(support):
		return false
	var data := _get_support_rect(support)
	if not bool(data.get("valid", false)):
		return false
	var support_rect: Rect2 = data.get("rect", Rect2())
	var body := _get_collision_rect()
	var feet_y := global_position.y + body.end.y
	if absf(feet_y - support_rect.position.y) > 1.0:
		return false
	if global_position.x + body.position.x < support_rect.position.x - 0.01 \
		or global_position.x + body.end.x > support_rect.end.x + 0.01:
		return false
	if _decisions_suspended:
		if _bound_support != null and is_instance_valid(_bound_support):
			remove_collision_exception_with(_bound_support)
		_bound_support = null
		_patrol_enabled = false
		_jump_plan.clear()
		_jump_source = null
		_jump_landing_remaining = 0.0
		_locomotion_state = LocomotionState.LANDING
		velocity = Vector2.ZERO
		_patrol_moving = false
		_clear_pursuit_route()
		return true
	if not _is_valid_landing_support(support):
		return false
	var minimum_x := support_rect.position.x + edge_margin - body.position.x
	var maximum_x := support_rect.end.x - edge_margin - body.end.x
	var can_patrol := maximum_x - minimum_x >= MIN_PATROL_DISTANCE \
		and global_position.x >= minimum_x - 0.01 and global_position.x <= maximum_x + 0.01
	var sweep := _make_body_sweep(minimum_x, maximum_x, body, global_position.y)
	if can_patrol and not _is_world_rect_clear(sweep, support):
		can_patrol = false
	if not can_patrol:
		var stationary := _make_body_sweep(global_position.x, global_position.x, body, global_position.y)
		if not _is_world_rect_clear(stationary, support):
			return false
		minimum_x = global_position.x
		maximum_x = global_position.x
	_patrol_direction = _jump_facing_direction
	_bind_patrol_support(support, minimum_x, maximum_x, can_patrol)
	_clear_pursuit_route()
	_jump_source = null
	_jump_plan.clear()
	_jump_landing_remaining = JUMP_LANDING_TIME
	_jump_cooldown_remaining = JUMP_COOLDOWN
	_jump_elapsed = 0.0
	_jump_escape_direction = 0
	_locomotion_state = LocomotionState.LANDING
	velocity = Vector2.ZERO
	_patrol_moving = false
	_update_visuals()
	return true
func _is_valid_landing_support(support: SolidPlatform) -> bool:
	return support != null and is_instance_valid(support) \
		and support.get_parent() == get_parent() and _node_is_inside_current_scene(support) \
		and _is_compatible_support(support)
func _configure_physics_queries() -> void:
	_area_shape = CircleShape2D.new()
	_area_shape.radius = _effective_attack_radius()
	_area_query = PhysicsShapeQueryParameters2D.new()
	_area_query.shape = _area_shape
	_area_query.collision_mask = PLAYER_MASK
	_area_query.collide_with_bodies = true
	_area_query.collide_with_areas = false
	_area_query.margin = 0.0
	_area_query.exclude = [get_rid()]

	_line_of_sight_query = PhysicsRayQueryParameters2D.new()
	_line_of_sight_query.collision_mask = WORLD_MASK
	_line_of_sight_query.collide_with_bodies = true
	_line_of_sight_query.collide_with_areas = false
	_line_of_sight_query.exclude = [get_rid()]

	_patrol_query_shape = RectangleShape2D.new()
	_patrol_query = PhysicsShapeQueryParameters2D.new()
	_patrol_query.shape = _patrol_query_shape
	_patrol_query.collision_mask = JumpSolver.QUERY_MASK
	_patrol_query.collide_with_bodies = true
	_patrol_query.collide_with_areas = false
	_patrol_query.margin = 0.0

func _effective_attack_radius() -> float:
	return clampf(attack_radius, 16.0, 400.0)

func _find_player_in_attack_area(area_center: Vector2) -> PhysicsBody2D:
	if _area_query == null or _area_shape == null:
		return null
	_area_shape.radius = _effective_attack_radius()
	_area_query.transform = Transform2D(0.0, area_center)
	var overlaps: Array[Dictionary] = get_world_2d().direct_space_state.intersect_shape(
		_area_query,
		MAX_QUERY_RESULTS
	)
	for overlap in overlaps:
		var collider: Variant = overlap.get("collider")
		if not collider is PhysicsBody2D:
			continue
		var player_body := collider as PhysicsBody2D
		if not _is_alive_player(player_body):
			continue
		if _has_clear_line_of_sight(player_body, area_center):
			return player_body
	return null

func _is_alive_player(candidate: PhysicsBody2D) -> bool:
	if not is_instance_valid(candidate):
		return false
	if not candidate.is_in_group("player") or candidate.get_parent() != get_parent() or not candidate.has_method("take_damage"):
		return false
	var dead_value: Variant = candidate.get("is_dead")
	if typeof(dead_value) != TYPE_BOOL or bool(dead_value):
		return false
	return _node_is_inside_current_scene(candidate)

func _is_known_player_defeated() -> bool:
	if _known_player == null or not is_instance_valid(_known_player):
		return false
	var dead_value: Variant = _known_player.get("is_dead")
	return typeof(dead_value) == TYPE_BOOL and bool(dead_value)

func _has_clear_line_of_sight(player_body: PhysicsBody2D, area_center: Vector2) -> bool:
	_line_of_sight_query.from = area_center
	_line_of_sight_query.to = _get_player_collider_center(player_body)
	var obstruction: Dictionary = get_world_2d().direct_space_state.intersect_ray(
		_line_of_sight_query
	)
	return obstruction.is_empty()

func _get_player_collider_center(player_body: PhysicsBody2D) -> Vector2:
	var player_shape := player_body.get_node_or_null("CollisionShape2D") as CollisionShape2D
	if player_shape != null and not player_shape.disabled:
		return player_shape.global_position
	return player_body.global_position

func _begin_windup(player_body: PhysicsBody2D) -> void:
	velocity = Vector2.ZERO
	_patrol_moving = false
	_locked_player = player_body
	_damaged_player_ids.clear()
	_attack_state = AttackState.WINDUP
	_state_remaining = maxf(windup_duration, 0.05)
	_recovery_fade_remaining = 0.0
	_body_sprite.pause()
	_locked_attack_center = global_position + attack_center_offset
	_select_attack_frame(&"windup", _state_remaining, _state_remaining)
	_update_attack_area_visual()

func _update_windup(delta: float) -> void:
	var effective_duration := maxf(windup_duration, 0.05)
	var before_step := _state_remaining
	_state_remaining = maxf(_state_remaining - delta, 0.0)
	if _state_remaining <= 0.0:
		var overshoot := maxf(delta - before_step, 0.0)
		_enter_active()
		_apply_attack_area()
		if overshoot >= maxf(active_window, 0.05):
			_enter_recovery()
		else:
			_state_remaining = maxf(active_window, 0.05) - overshoot
			_select_attack_frame(&"active", _state_remaining, maxf(active_window, 0.05))
		return
	_select_attack_frame(&"windup", _state_remaining, effective_duration)
	_update_attack_area_visual()

func _enter_active() -> void:
	_attack_state = AttackState.ACTIVE
	_state_remaining = maxf(active_window, 0.05)
	var feet_origin := Vector2(global_position.x, global_position.y + _get_collision_rect().end.y)
	if is_instance_valid(_attack_dust) and _attack_dust.has_method("burst"):
		_attack_dust.call("burst", feet_origin)
	_select_attack_frame(&"active", _state_remaining, _state_remaining)
	_update_attack_area_visual()
	_update_visuals()

func _update_active(delta: float) -> void:
	_apply_attack_area()
	var effective_duration := maxf(active_window, 0.05)
	_state_remaining = maxf(_state_remaining - delta, 0.0)
	if _state_remaining <= 0.0:
		_enter_recovery()
		return
	_select_attack_frame(&"active", _state_remaining, effective_duration)

func _apply_attack_area() -> void:
	if _defeated_emitted or _attack_state != AttackState.ACTIVE:
		return
	var overlaps: Array[Dictionary] = _find_overlaps_at_locked_center()
	for overlap in overlaps:
		var collider: Variant = overlap.get("collider")
		if not collider is PhysicsBody2D:
			continue
		var player_body := collider as PhysicsBody2D
		if not _is_alive_player(player_body):
			continue
		var player_id := player_body.get_instance_id()
		if _damaged_player_ids.has(player_id):
			continue
		if not _has_clear_line_of_sight(player_body, _locked_attack_center):
			continue
		_damaged_player_ids[player_id] = true
		var damage_result: Variant = player_body.call(
			"take_damage",
			maxi(attack_damage, 1),
			self
		)
		if typeof(damage_result) == TYPE_BOOL and bool(damage_result):
			_set_attack_visual(&"impact", 1.0)

		MAX_QUERY_RESULTS
	)

func _select_attack_frame(animation_name: StringName, remaining: float, effective_duration: float) -> void:
	if _body_sprite.sprite_frames == null or not _body_sprite.sprite_frames.has_animation(animation_name):
		return
	var frames := _body_sprite.sprite_frames.get_frame_count(animation_name)
	if frames <= 0:
		return
	if _body_sprite.animation != animation_name:
		_body_sprite.animation = animation_name
	_body_sprite.pause()
	var normalized_progress := clampf(1.0 - remaining / maxf(effective_duration, 0.0001), 0.0, 1.0)
	var total_weight := 0.0
	for frame_index in frames:
		total_weight += _body_sprite.sprite_frames.get_frame_duration(animation_name, frame_index)
	var elapsed := normalized_progress * total_weight
	var cumulative := 0.0
	var selected_frame := frames - 1
	for frame_index in frames:
		cumulative += _body_sprite.sprite_frames.get_frame_duration(animation_name, frame_index)
		if elapsed < cumulative:
			selected_frame = frame_index
			break
	_body_sprite.pause()
	_body_sprite.frame = selected_frame

func _advance_patrol(delta: float) -> void:
	velocity = Vector2.ZERO
	if _patrol_pause_remaining > 0.0:
		_patrol_pause_remaining = maxf(_patrol_pause_remaining - delta, 0.0)
		_patrol_moving = false
		_update_visuals()
		return
	var distance_to_bound := _patrol_max_x - global_position.x if _patrol_direction > 0 else global_position.x - _patrol_min_x
	if distance_to_bound <= MOVEMENT_EPSILON:
		_turn_patrol_around()
		return
	var requested_dx := float(_patrol_direction) * minf(PATROL_SPEED * delta, distance_to_bound)
	if absf(requested_dx) < MOVEMENT_EPSILON:
		_patrol_moving = false
		_update_visuals()
		return

	var collision: KinematicCollision2D = move_and_collide(Vector2(requested_dx, 0.0), true, 0.0)
	var travel_x := requested_dx
	if collision != null:
		travel_x = collision.get_travel().x
	if signf(travel_x) != signf(requested_dx):
		travel_x = 0.0
	travel_x = signf(requested_dx) * minf(absf(travel_x), absf(requested_dx))
	var candidate_x := global_position.x + travel_x
	var clamped_x := clampf(candidate_x, _patrol_min_x, _patrol_max_x)
	var safe_dx := clamped_x - global_position.x
	var postcondition_valid := (
		candidate_x >= _patrol_min_x - MOVEMENT_EPSILON
		and candidate_x <= _patrol_max_x + MOVEMENT_EPSILON
		and signf(safe_dx) == signf(requested_dx)
		and absf(safe_dx) <= absf(requested_dx) + MOVEMENT_EPSILON
	)
	if not postcondition_valid:
		safe_dx = 0.0
	if absf(safe_dx) >= MOVEMENT_EPSILON:
		global_position.x += safe_dx
	var hit_bound := absf(_patrol_max_x - global_position.x) <= MOVEMENT_EPSILON if _patrol_direction > 0 else absf(global_position.x - _patrol_min_x) <= MOVEMENT_EPSILON
	var blocked := collision != null or absf(safe_dx - requested_dx) >= MOVEMENT_EPSILON
	if hit_bound or blocked:
		_turn_patrol_around()
		return
	_patrol_moving = absf(safe_dx) >= MOVEMENT_EPSILON
	_update_visuals()

	_patrol_direction *= -1
	_patrol_pause_remaining = PATROL_PAUSE
	_update_visuals()

func _bind_patrol_support(support: SolidPlatform, minimum_x: float, maximum_x: float, enable_patrol: bool) -> void:
	var previous_support := _bound_support
	var valid_support := support != null and is_instance_valid(support) and _is_compatible_support(support)
	if previous_support != null and is_instance_valid(previous_support) \
		and (not valid_support or previous_support != support):
		remove_collision_exception_with(previous_support)
	_patrol_enabled = false
	_patrol_moving = false
	if not valid_support:
		_bound_support = null
		_warn_patrol_once("suporte ausente ou incompatível; o alvo permanecerá estacionário")
		return
	_bound_support = support
	_patrol_min_x = minf(minimum_x, maximum_x)
	_patrol_max_x = maxf(minimum_x, maximum_x)
	# Reafirma a exceção após pousos, inclusive quando voltamos ao mesmo suporte.
	add_collision_exception_with(support)
	if enable_patrol and _patrol_max_x - _patrol_min_x >= MIN_PATROL_DISTANCE:
		_patrol_enabled = true
		_patrol_direction = _jump_facing_direction if _locomotion_state == LocomotionState.FLIGHT else 1
		_patrol_pause_remaining = 0.0
		_body_sprite.flip_h = _patrol_direction < 0
	else:
		_warn_patrol_once("nenhum percurso livre e útil de 8 px; o alvo permanecerá estacionário")
func _is_compatible_support(support: SolidPlatform) -> bool:
	if not is_instance_valid(support) or not support.is_inside_tree():
		return false
	if absf(support.global_rotation) > 0.001 or not support.global_scale.is_equal_approx(Vector2.ONE):
		return false
	var shape_count := 0
	for child in support.get_children():
		var collision := child as CollisionShape2D
		if collision == null:
			continue
		shape_count += 1
		if collision.disabled or not collision.shape is RectangleShape2D:
			return false
		if collision.position.length() > 0.01 or absf(collision.rotation) > 0.001 or not collision.scale.is_equal_approx(Vector2.ONE):
			return false
	return shape_count == 1

func _get_collision_rect() -> Rect2:
	for child in get_children():
		var collision := child as CollisionShape2D
		if collision != null and not collision.disabled and collision.shape != null:
			var rect := collision.shape.get_rect()
			rect.position += collision.position
			return rect
	# Sem CollisionShape2D: assume que a origem está no centro da base.
	return Rect2(-16.0, -32.0, 32.0, 32.0)

func _get_support_rect(support: SolidPlatform) -> Dictionary:
	if not _is_compatible_support(support):
		return {"valid": false}
	for child in support.get_children():
		var collision := child as CollisionShape2D
		if collision == null:
			continue
		var rectangle := collision.shape as RectangleShape2D
		var top_left := collision.global_position - rectangle.size * 0.5
		return {"valid": true, "rect": Rect2(top_left, rectangle.size)}
	return {"valid": false}

		support_rect.end.x - margin - body_rect.end.x
	)

func _find_first_safe_stationary_x(minimum_x: float, maximum_x: float, body_rect: Rect2, body_global_y: float, support: SolidPlatform) -> Dictionary:
	var span := maxf(maximum_x - minimum_x, 0.0)
	var sample_count := maxi(1, int(ceil(span / 8.0)))
	for sample_index in range(sample_count + 1):
		var x := lerpf(minimum_x, maximum_x, float(sample_index) / float(sample_count))
		var bounds := _make_body_sweep(x, x, body_rect, body_global_y)

	return {"valid": false}

func _place_on_platform() -> void:
	if not _placement_pending:
		return
	var body := _get_collision_rect()
	var needed_width := body.size.x + edge_margin * 2.0

	# Prefere plataformas livres; o Floor só entra se não sobrar nenhuma,
	# e plataformas já ocupadas por outro inimigo ficam por último.
	var free: Array[SolidPlatform] = []
	var floors: Array[SolidPlatform] = []
	var taken: Array[SolidPlatform] = []
	for node in get_tree().get_nodes_in_group(&"solid_platform"):
		var candidate := node as SolidPlatform
		if candidate == null or candidate.size.x < needed_width or not _is_compatible_support(candidate):
			continue
		if candidate.name == &"Floor":
			floors.append(candidate)
		elif candidate.has_meta(CLAIMED_META):
			taken.append(candidate)
		else:
			free.append(candidate)

	var pool: Array[SolidPlatform] = free
	if pool.is_empty():
		pool = floors
	if pool.is_empty():
		pool = taken
	if pool.is_empty():
		_placement_pending = false
		_disable_patrol_with_warning("nenhuma plataforma compatível; o alvo permanecerá estacionário")
		return

	var rng := RandomNumberGenerator.new()
	if SolidPlatform.room_seed == 0:
		rng.randomize()
	else:
		rng.seed = hash([SolidPlatform.room_seed, get_index(), "training_target"])

	var platform := pool[rng.randi_range(0, pool.size() - 1)]
	platform.set_meta(CLAIMED_META, true)
	var support_data := _get_support_rect(platform)
	var support_rect: Rect2 = support_data.get("rect", Rect2())
	var patrol_bounds := _patrol_bounds_for_support(support_rect, body, edge_margin)
	var minimum_x := patrol_bounds.x
	var maximum_x := patrol_bounds.y
	var platform_top := support_rect.position.y
	var body_global_y := platform_top - body.end.y
	var sweep := _make_body_sweep(minimum_x, maximum_x, body, body_global_y)
	var has_useful_route := maximum_x - minimum_x >= MIN_PATROL_DISTANCE
	var sweep_is_clear := has_useful_route and _is_world_rect_clear(sweep, platform)
	if sweep_is_clear:
		global_position = Vector2(rng.randf_range(minimum_x, maximum_x), body_global_y)
		_bind_patrol_support(platform, minimum_x, maximum_x, true)
	else:
		var stationary := _find_first_safe_stationary_x(minimum_x, maximum_x, body, body_global_y, platform)
		if bool(stationary.get("valid", false)):
			global_position = Vector2(float(stationary.get("x", minimum_x)), body_global_y)
			_bind_patrol_support(platform, minimum_x, maximum_x, false)
			_warn_patrol_once("sweep bloqueado ou curto; usando posição estacionária segura")
		else:
			_bind_patrol_support(null, global_position.x, global_position.x, false)
			_warn_patrol_once("nenhum spawn seguro; mantendo a posição original")
	_placement_pending = false
	_update_visuals()

func _floor_landing_xs(source_rect: Rect2, minimum_x: float, maximum_x: float, preferred_x: float) -> Array[float]:
	var candidates := _candidate_landing_xs(preferred_x, minimum_x, maximum_x)
	var left_clear_x := source_rect.position.x - _body_rect.end.x - 4.0
	var right_clear_x := source_rect.end.x - _body_rect.position.x + 4.0
	if left_clear_x >= minimum_x and left_clear_x <= maximum_x:
		candidates.append(left_clear_x)
	if right_clear_x >= minimum_x and right_clear_x <= maximum_x:
		candidates.append(right_clear_x)
	var result: Array[float] = []
	for landing_x in candidates:
		var body_left := landing_x + _body_rect.position.x
		var body_right := landing_x + _body_rect.end.x
		if body_right > source_rect.position.x + 0.01 and body_left < source_rect.end.x - 0.01:
			continue
		var unique := true
		for existing in result:
			if absf(existing - landing_x) <= POSITION_EPSILON:
				unique = false
				break
		if unique:
			result.append(landing_x)
	result.sort_custom(func(a: float, b: float) -> bool:
		return absf(a - preferred_x) < absf(b - preferred_x)
	)
	return result

func _build_route(
	source: SolidPlatform,
	source_rect: Rect2,
	destination: SolidPlatform,
	destination_rect: Rect2,
	start_position: Vector2,
	landing_x: float
) -> Dictionary:
	if not _support_is_valid(source) or not _support_is_valid(destination):
		return {}
	var landing_min_x := destination_rect.position.x + EDGE_MARGIN - _body_rect.position.x
	var landing_max_x := destination_rect.end.x - EDGE_MARGIN - _body_rect.end.x
	if landing_x < landing_min_x - POSITION_EPSILON or landing_x > landing_max_x + POSITION_EPSILON:
		return {}
	var dx := landing_x - start_position.x
	if absf(dx) > MAX_HORIZONTAL:
		return {}
	var rise := source_rect.position.y - destination_rect.position.y
	if rise > MAX_RISE or rise < -MAX_DROP:
		return {}

	var apex_height := minf(maxf(MIN_APEX, rise + MIN_APEX_CLEARANCE), MAX_APEX)
	var launch_vy := -sqrt(2.0 * GRAVITY * apex_height)
	var landing_y := destination_rect.position.y - 0.5 - _body_rect.end.y
	var dy := landing_y - start_position.y
	var discriminant := launch_vy * launch_vy + 2.0 * GRAVITY * dy
	if discriminant < 0.0:
		return {}
	var final_vy := sqrt(discriminant)
	var flight_time := (-launch_vy + final_vy) / GRAVITY
	if flight_time <= 0.0 or flight_time > MAX_FLIGHT_TIME:
		return {}
	if apex_height + 0.01 < rise + MIN_APEX_CLEARANCE:
		return {}
	var launch_vx := dx / flight_time
	if absf(launch_vx) > MAX_HORIZONTAL_SPEED:
		return {}
	if not _trajectory_is_clear(
		start_position,
		launch_vx,
		launch_vy,
		flight_time,
		source_rect.position.y,
		source,
		destination
	):
		return {}
	return {
		"source": source,
		"destination": destination,
		"start_position": start_position,
		"landing_position": Vector2(landing_x, landing_y),
		"landing_x": landing_x,
		"launch_velocity": Vector2(launch_vx, launch_vy),
		"gravity": GRAVITY,
		"flight_time": flight_time
	}

func _trajectory_is_clear(
	start_position: Vector2,
	launch_vx: float,
	launch_vy: float,
	flight_time: float,
	source_top: float,
	source: SolidPlatform,
	_destination: SolidPlatform
) -> bool:
	var final_vy := launch_vy + GRAVITY * flight_time
	var maximum_speed := maxf(
		Vector2(launch_vx, launch_vy).length(),
		Vector2(launch_vx, final_vy).length()
	)
	var steps := maxi(1, int(ceil(maximum_speed * flight_time / MAX_SEGMENT_LENGTH)))
	var sample_times: Array[float] = [0.0]
	for step_index in range(1, steps):
		sample_times.append(flight_time * float(step_index) / float(steps))
	if is_instance_valid(source) and launch_vy < 0.0:
		var start_bottom := start_position.y + _body_rect.end.y
		var clearance := start_bottom - (source_top - EXIT_CLEARANCE)
		var discriminant := launch_vy * launch_vy - 2.0 * GRAVITY * clearance
		if clearance > 0.0 and discriminant >= 0.0:
			var clear_time := (-launch_vy - sqrt(discriminant)) / GRAVITY
			if clear_time > 0.000001 and clear_time < flight_time - 0.000001:
				sample_times.append(clear_time)
	sample_times.append(flight_time)
	sample_times.sort()

	var unique_times: Array[float] = []
	for t in sample_times:
		if unique_times.is_empty() or t - unique_times.back() > 0.000001:
			unique_times.append(t)
	var space := _owner.get_world_2d().direct_space_state
	var previous_center := start_position + _body_rect.get_center()
	var previous_source_exception := is_instance_valid(source) and launch_vy < 0.0 \
		and start_position.y + _body_rect.end.y > source_top - EXIT_CLEARANCE
	for step_index in range(unique_times.size()):
		var t: float = unique_times[step_index]
		var node_position := start_position + Vector2(launch_vx * t, launch_vy * t + 0.5 * GRAVITY * t * t)
		var center := node_position + _body_rect.get_center()
		var vy := launch_vy + GRAVITY * t
		var bottom := node_position.y + _body_rect.end.y
		var allow_source_exit := (
			bottom > source_top - EXIT_CLEARANCE
			and vy < 0.0
			and is_instance_valid(source)
		)
		if _overlaps_at(space, center, source, allow_source_exit):
			return false
		if step_index > 0:
			_query.transform = Transform2D(0.0, previous_center)
			_query.motion = center - previous_center
			_query.exclude = _exclusions(source, previous_source_exception)
			var cast := space.cast_motion(_query)
			if cast.size() > 0 and cast[0] < 0.9999:
				return false
		previous_center = center
		previous_source_exception = allow_source_exit
	return true
func _overlaps_at(

func _candidate_landing_xs(start_x: float, minimum_x: float, maximum_x: float) -> Array[float]:
	var raw: Array[float] = [
		clampf(start_x, minimum_x, maximum_x),
		minimum_x,
		maximum_x,
		lerpf(minimum_x, maximum_x, 0.5),
		lerpf(minimum_x, maximum_x, 0.25),
		lerpf(minimum_x, maximum_x, 0.75)
	]
	raw.sort_custom(func(a: float, b: float) -> bool:
		return absf(a - start_x) < absf(b - start_x)
	)
	var result: Array[float] = []
	for x in raw:
		var unique := true
		for existing in result:
			if absf(existing - x) <= POSITION_EPSILON:
				unique = false
				break
		if unique:
			result.append(x)
	return result

func _support_is_valid(support: SolidPlatform) -> bool:
	if support == null or not is_instance_valid(support) or not support.is_inside_tree():
		return false
	if support.get_parent() != _owner.get_parent() or not _node_is_in_current_scene(support):
		return false
	if absf(support.global_rotation) > 0.001 or not support.global_scale.is_equal_approx(Vector2.ONE):
		return false
	var shape_count := 0
	for child in support.get_children():
		var collision := child as CollisionShape2D
		if collision == null:
			continue
		shape_count += 1
		if collision.disabled or not collision.shape is RectangleShape2D:
			return false
		if collision.position.length() > 0.01 or absf(collision.rotation) > 0.001 or not collision.scale.is_equal_approx(Vector2.ONE):
			return false
	return shape_count == 1

func _support_rect(support: SolidPlatform) -> Rect2:
	for child in support.get_children():
		var collision := child as CollisionShape2D
		if collision == null:
			continue
		var rectangle := collision.shape as RectangleShape2D
		if rectangle == null:
			return Rect2()
		return Rect2(collision.global_position - rectangle.size * 0.5, rectangle.size)
	return Rect2()

func _standing_on(support: SolidPlatform) -> bool:
	if not _support_is_valid(support):
		return false
	var rect := _support_rect(support)
	var feet_y := _owner.global_position.y + _body_rect.end.y
	return absf(feet_y - rect.position.y) <= 1.0

		and _query != null and _query_shape != null

	var scene_root := _owner.get_tree().current_scene
	return is_instance_valid(scene_root) and (node == scene_root or scene_root.is_ancestor_of(node))

func find_path(
	source: SolidPlatform,
	destination: SolidPlatform,
	supports: Array[SolidPlatform],
	jump_solver: Object,
	blocked_edges: Dictionary
) -> Array[SolidPlatform]:
	if source == destination:
		return [source]
	if not is_instance_valid(source) or not is_instance_valid(destination):
		return []
	var costs: Dictionary = {source.get_instance_id(): 0.0}
	var previous: Dictionary = {}
	var supports_by_id: Dictionary = {}
	var visited: Dictionary = {}
	var open: Array[SolidPlatform] = [source]
	supports_by_id[source.get_instance_id()] = source
	supports_by_id[destination.get_instance_id()] = destination
	for support in supports:
		supports_by_id[support.get_instance_id()] = support
	while not open.is_empty():
		var current_index: int = -1
		var current: SolidPlatform
		var current_cost: float = INF
		for index in range(open.size()):
			var candidate: SolidPlatform = open[index]
			var candidate_cost: float = float(costs.get(candidate.get_instance_id(), INF))
			if candidate_cost < current_cost:
				current_index = index
				current = candidate
				current_cost = candidate_cost
		if current_index < 0 or current == null:
			break
		open.remove_at(current_index)
		var current_id: int = current.get_instance_id()
		if visited.has(current_id):
			continue
		visited[current_id] = true
		if current == destination:
			break
		for neighbor in supports:
			if neighbor == current or not bool(jump_solver.call("can_reach_support", current, neighbor)):
				continue
			var neighbor_id: int = neighbor.get_instance_id()
			if visited.has(neighbor_id):
				continue
			if float(blocked_edges.get(edge_key(current, neighbor), 0.0)) > 0.0:
				continue
			var source_rect: Rect2 = jump_solver.call("support_rect", current)
			var destination_rect: Rect2 = jump_solver.call("support_rect", neighbor)
			var horizontal_cost := absf(source_rect.get_center().x - destination_rect.get_center().x) / EDGE_HORIZONTAL_SCALE
			var vertical_cost := absf(source_rect.position.y - destination_rect.position.y) / EDGE_VERTICAL_SCALE
			var edge_cost := 1.0 + horizontal_cost * 0.2 + vertical_cost * 0.25
			var next_cost: float = current_cost + edge_cost
			if next_cost >= float(costs.get(neighbor_id, INF)):
				continue
			costs[neighbor_id] = next_cost
			previous[neighbor_id] = current_id
			supports_by_id[neighbor_id] = neighbor
			if not open.has(neighbor):
				open.append(neighbor)
	var cursor_id: int = destination.get_instance_id()
	if not previous.has(cursor_id):
		return []
	var path: Array[SolidPlatform] = [destination]
	while cursor_id != source.get_instance_id():
		if not previous.has(cursor_id):
			return []
		cursor_id = int(previous[cursor_id])
		var step: SolidPlatform = supports_by_id.get(cursor_id) as SolidPlatform
		if step == null:
			return []
		path.append(step)
	path.reverse()
	return path

func edge_key(source: SolidPlatform, destination: SolidPlatform) -> String:
	if not is_instance_valid(source) or not is_instance_valid(destination):
		return ""
	return "%d>%d" % [source.get_instance_id(), destination.get_instance_id()]

	_rivet_query.collision_mask = WORLD_MASK
	_rivet_query.collide_with_bodies = true
	_rivet_query.collide_with_areas = false
	_rivet_query.margin = 0.08
	_prepare_warning_sound()
	_update_visuals()
	_apply_facing_direction()
	_set_warning_visible(false)

func take_damage(amount: int, _source: Node = null) -> bool:
	if _defeated_emitted or health <= 0 or amount <= 0:
		return false

	health = maxi(health - amount, 0)
	_flash_remaining = maxf(flash_duration, 0.0)
	_update_visuals()
	if health == 0:
		_defeated_emitted = true
		_set_warning_visible(false)
		defeated.emit()
		queue_free()
	return true

func _process(delta: float) -> void:
	if _shot_visual_remaining > 0.0:
		_shot_visual_remaining = maxf(_shot_visual_remaining - delta, 0.0)
	_update_body_animation()

func _physics_process(delta: float) -> void:
	_patrol_step_pending = false
	if _defeated_emitted or health <= 0:
		velocity = Vector2.ZERO
		_set_warning_visible(false)
		return
	if _flash_remaining > 0.0:
		_flash_remaining = maxf(_flash_remaining - delta, 0.0)
		_update_visuals()
	_shot_cooldown_remaining = maxf(_shot_cooldown_remaining - delta, 0.0)

	if not is_on_floor():
		velocity.y = minf(velocity.y + 1600.0 * delta, 900.0)
	else:
		velocity.y = 0.0

	if _player == null or not is_instance_valid(_player):
		_player = get_tree().get_first_node_in_group("player") as Node2D
	if _player == null or not is_instance_valid(_player):
		_patrol_for_angle = false
		_set_warning_visible(false)
		if _state == AttackState.WINDUP:
			_state = AttackState.APPROACH
			_windup_remaining = 0.0
		elif _state == AttackState.RETREAT:
			_state = AttackState.APPROACH
			_retreat_remaining = 0.0
		var previous_x := global_position.x
		if _state == AttackState.RECOVERY:
			_process_recovery(delta)
			velocity.x = 0.0
		else:
			_process_patrol(delta)
		move_and_slide()
		_finish_patrol_step(previous_x)
		return
	if _player.get("is_dead") == true:
		_patrol_for_angle = false
		velocity.x = 0.0
		_set_warning_visible(false)
		move_and_slide()
		return

	var was_retreating := _state == AttackState.RETREAT
	var previous_x := global_position.x
	match _state:
		AttackState.WINDUP:
			_process_windup(delta)
			velocity.x = 0.0
		AttackState.RECOVERY:
			_process_recovery(delta)
			velocity.x = 0.0
		AttackState.RETREAT:
			_process_retreat(delta)
		AttackState.APPROACH:
			_decide_ready_state(delta)
			_apply_ready_motion(delta)

	move_and_slide()
	if was_retreating and _state == AttackState.RETREAT:
		var travelled := absf(global_position.x - previous_x)
		_retreat_remaining = maxf(_retreat_remaining - travelled, 0.0)
		if travelled < 0.01 or _retreat_remaining <= POSITION_EPSILON:
			velocity.x = 0.0
			_state = AttackState.APPROACH
	_finish_patrol_step(previous_x)

func _finish_patrol_step(previous_x: float) -> void:
	if not _patrol_step_pending:
		return
	_patrol_step_pending = false
	if absf(global_position.x - previous_x) < 0.01:
		_patrol_direction *= -1.0
		_patrol_pause_remaining = maxf(patrol_end_pause, 0.0)
		velocity.x = 0.0
func _configure_patrol_bounds() -> void:
	var fallback_center := global_position.x
	var fallback_half_width := maxf(patrol_half_width, 0.0)
	_patrol_min_x = fallback_center - fallback_half_width
	_patrol_max_x = fallback_center + fallback_half_width
	if not patrol_area_path.is_empty():
		var patrol_area := get_node_or_null(patrol_area_path) as Node2D
		var left_marker: Marker2D = null
		var right_marker: Marker2D = null
		if patrol_area != null:
			left_marker = patrol_area.get_node_or_null("LeftLimit") as Marker2D
			right_marker = patrol_area.get_node_or_null("RightLimit") as Marker2D
		if left_marker != null and right_marker != null:
			_patrol_min_x = minf(left_marker.global_position.x, right_marker.global_position.x)
			_patrol_max_x = maxf(left_marker.global_position.x, right_marker.global_position.x)
		else:
			push_warning("RivetWorker: patrol_area_path inválido ou sem Marker2D LeftLimit/RightLimit; usando o fallback centrado no spawn.")

	if _patrol_max_x - _patrol_min_x <= BODY_HALF_WIDTH * 2.0 + POSITION_EPSILON * 2.0:
		_patrol_area_usable = false
		push_warning("RivetWorker: a área de patrulha não deixa espaço útil para o corpo; o inimigo ficará imóvel.")

func _get_safe_center_bounds() -> Vector2:
	return Vector2(_patrol_min_x + BODY_HALF_WIDTH, _patrol_max_x - BODY_HALF_WIDTH)

func _move_toward_area_x(target_x: float, speed: float, delta: float) -> bool:
	if not _patrol_area_usable or delta <= 0.0:
		velocity.x = 0.0
		return false
	var bounds := _get_safe_center_bounds()
	var destination_x := clampf(target_x, bounds.x, bounds.y)
	if global_position.x < bounds.x:
		destination_x = bounds.x
	elif global_position.x > bounds.y:
		destination_x = bounds.y
	var dx := destination_x - global_position.x
	var outside_bounds := global_position.x < bounds.x or global_position.x > bounds.y
	if absf(dx) <= POSITION_EPSILON and not outside_bounds:
		velocity.x = 0.0
		return true
	var direction := signf(dx)
	if not _has_ground_ahead(direction):
		velocity.x = 0.0
		return false
	_facing_direction = int(direction)
	_apply_facing_direction()
	var step_distance := minf(absf(dx), maxf(speed, 0.0) * delta)
	if step_distance <= 0.0:
		velocity.x = 0.0
		return true
	velocity.x = direction * step_distance / delta
	return true

func _process_patrol(delta: float) -> void:
	_set_warning_visible(false)
	if not _patrol_area_usable or delta <= 0.0:
		velocity.x = 0.0
		return
	var bounds := _get_safe_center_bounds()
	if global_position.x < bounds.x:
		_patrol_direction = 1.0
		_move_toward_area_x(bounds.x, patrol_speed, delta)
		return
	if global_position.x > bounds.y:
		_patrol_direction = -1.0
		_move_toward_area_x(bounds.y, patrol_speed, delta)
		return
	if _patrol_pause_remaining > 0.0:
		_patrol_pause_remaining = maxf(_patrol_pause_remaining - delta, 0.0)
		velocity.x = 0.0
		return
	if (_patrol_direction > 0.0 and global_position.x >= bounds.y - POSITION_EPSILON) or (_patrol_direction < 0.0 and global_position.x <= bounds.x + POSITION_EPSILON):
		_patrol_direction *= -1.0
		_patrol_pause_remaining = maxf(patrol_end_pause, 0.0)
		velocity.x = 0.0
		return
	var endpoint_x := bounds.y if _patrol_direction > 0.0 else bounds.x
	if not _move_toward_area_x(endpoint_x, patrol_speed, delta):
		_patrol_direction *= -1.0
		_patrol_pause_remaining = maxf(patrol_end_pause, 0.0)
		velocity.x = 0.0
		return
	_patrol_step_pending = not is_zero_approx(velocity.x)
func _process_windup(delta: float) -> void:
	_windup_remaining = maxf(_windup_remaining - delta, 0.0)
	_update_warning_line()
	if _windup_remaining <= 0.0:
		_fire_locked_rivet()
		_state = AttackState.RECOVERY
		_recovery_remaining = maxf(recovery_duration, 0.0)
		_shot_cooldown_remaining = maxf(minimum_shot_interval, 0.0)
		_retreat_used_this_cycle = false

func _process_recovery(delta: float) -> void:
	_recovery_remaining = maxf(_recovery_remaining - delta, 0.0)
	if _recovery_remaining <= 0.0:
		_state = AttackState.APPROACH

func _decide_ready_state(delta: float) -> void:
	if _patrol_area_usable:
		var bounds := _get_safe_center_bounds()
		if global_position.x < bounds.x or global_position.x > bounds.y:
			return
	var target := _get_player_aim_point()
	var desired_facing := _get_facing_for_target(target)
	var muzzle := _get_muzzle_position_at(global_position.x, desired_facing)
	var distance := muzzle.distance_to(target)
	if distance < minimum_attack_range:
		if not _retreat_used_this_cycle:
			_patrol_for_angle = false
			_begin_close_range_retreat(delta)
		return

	if _shot_cooldown_remaining > 0.0:
		return
	if distance > maximum_attack_range:
		return
	if not _has_line_of_sight(muzzle, target):
		return

	_patrol_for_angle = false
	_locked_direction = (target - muzzle).normalized()
	if _locked_direction.is_zero_approx():
		_locked_direction = Vector2.LEFT
	_facing_direction = desired_facing
	_apply_facing_direction()
	_locked_muzzle_position = muzzle
	velocity.x = 0.0
	_windup_remaining = maxf(windup_duration, 0.0)
	_state = AttackState.WINDUP
	_set_warning_visible(true)
	_warning_sound.play()
	_update_warning_line()

func _apply_ready_motion(delta: float) -> void:
	if _state != AttackState.APPROACH:
		return
	if not _patrol_area_usable:
		velocity.x = 0.0
		return
	var bounds := _get_safe_center_bounds()
	if global_position.x < bounds.x or global_position.x > bounds.y:
		_patrol_for_angle = false
		_move_toward_area_x(global_position.x, move_speed, delta)
		return

	var target := _get_player_aim_point()
	var desired_facing := _get_facing_for_target(target)
	var muzzle := _get_muzzle_position_at(global_position.x, desired_facing)
	var distance := muzzle.distance_to(target)
	if distance < minimum_attack_range:
		# Depois do único recuo permitido, espera o jogador abrir espaço.
		velocity.x = 0.0
		return

	if _shot_cooldown_remaining > 0.0:
		_process_patrol(delta)
		return
	if _patrol_for_angle:
		var clear_position := _find_nearest_clear_x(target)
		if is_finite(clear_position):
			if _move_toward_area_x(clear_position, move_speed, delta):
				_patrol_for_angle = false
			else:
				_process_patrol(delta)
		else:
			_process_patrol(delta)
		return
	var target_visible := _has_line_of_sight(muzzle, target)
	if distance > maximum_attack_range:
		_move_toward_area_x(target.x, move_speed, delta)
		return
	if not target_visible:
		var reposition_x := _find_nearest_clear_x(target)
		if is_finite(reposition_x):
			_move_toward_area_x(reposition_x, move_speed, delta)
		else:
			_patrol_for_angle = true
			_process_patrol(delta)
		return
	velocity.x = 0.0
func _find_nearest_clear_x(target: Vector2) -> float:
	var current_x := global_position.x
	var bounds := _get_safe_center_bounds()
	var best_x := INF
	var best_distance := INF
	var steps := int(REPOSITION_SEARCH_DISTANCE / REPOSITION_SEARCH_STEP)
	for step in range(1, steps + 1):
		for direction_value in [-1.0, 1.0]:
			var direction := float(direction_value)
			var candidate_x: float = current_x + direction * float(step) * REPOSITION_SEARCH_STEP
			if candidate_x < bounds.x or candidate_x > bounds.y:
				continue
			var candidate_facing := 1 if target.x > candidate_x else -1
			var candidate_muzzle := _get_muzzle_position_at(candidate_x, candidate_facing)
			var candidate_distance := candidate_muzzle.distance_to(target)
			if candidate_distance < minimum_attack_range or candidate_distance > maximum_attack_range:
				continue
			if not _has_line_of_sight(candidate_muzzle, target):
				continue
			if not _has_ground_at_x(candidate_x):
				continue
			var movement_distance := absf(candidate_x - current_x)
			if movement_distance < best_distance:
				best_distance = movement_distance
				best_x = candidate_x
	return best_x

func _has_line_of_sight(from: Vector2, to: Vector2) -> bool:
	if from.distance_squared_to(to) <= 0.01:
		return false
	_rivet_query.transform = Transform2D(0.0, from)
	_rivet_query.motion = Vector2.ZERO
	var space_state := get_world_2d().direct_space_state
	if not space_state.intersect_shape(_rivet_query, 1).is_empty():
		return false
	_rivet_query.motion = to - from
	var fractions := space_state.cast_motion(_rivet_query)
	return fractions.size() >= 2 and fractions[0] >= 1.0

func _has_ground_ahead(direction: float) -> bool:
	return _has_ground_at_x(global_position.x + direction * (BODY_HALF_WIDTH + 7.0))

func _has_ground_at_x(x_position: float) -> bool:
	var space_state := get_world_2d().direct_space_state
	_ground_query.from = Vector2(x_position, global_position.y - 12.0)
	_ground_query.to = Vector2(x_position, global_position.y + 24.0)
	_ground_query.exclude = [get_rid()]
	var result := space_state.intersect_ray(_ground_query)
	if result.is_empty():
		return false
	var surface_normal: Vector2 = result.get("normal", Vector2.ZERO)
	var surface_position: Vector2 = result.get("position", Vector2.ZERO)
	var surface_y := surface_position.y
	return surface_normal.y <= -0.7 and absf(surface_y - global_position.y) <= 24.0

func _fire_locked_rivet() -> void:
	_set_warning_visible(false)
	if _defeated_emitted or health <= 0 or rivet_scene == null or _player == null or not is_instance_valid(_player):
		return
	var muzzle := _locked_muzzle_position
	if muzzle.distance_to(_get_player_aim_point()) < MINIMUM_SAFE_SHOT_DISTANCE:
		return
	if _active_rivet != null and is_instance_valid(_active_rivet):
		return
	var parent := get_tree().get_first_node_in_group("hostile_projectile_container")
	if parent == null:
		parent = get_parent()
	if parent == null:
		return
	if not _is_clear_rivet_spawn(muzzle, _locked_direction):
		return
	var rivet := rivet_scene.instantiate() as HostileRivet
	if rivet == null:
		return
	parent.add_child(rivet)
	rivet.global_position = muzzle + _locked_direction * RIVET_SPAWN_OFFSET
	rivet.launch(_locked_direction, self, rivet_speed, rivet_damage, rivet_lifetime)
	_active_rivet = rivet
	rivet.finished.connect(_on_rivet_finished.bind(rivet), CONNECT_ONE_SHOT)
	_shot_visual_remaining = 0.12
	_set_body_animation(&"fire")

func _on_rivet_finished(_reason: String, rivet: HostileRivet) -> void:
	if rivet == _active_rivet:
		_active_rivet = null

func _get_muzzle_position_at(x_position: float, facing: int) -> Vector2:
	return Vector2(x_position + float(facing) * 22.0, global_position.y - 36.0)

func _get_muzzle_position() -> Vector2:
	return _get_muzzle_position_at(global_position.x, _facing_direction)

func _get_facing_for_target(target: Vector2) -> int:

	if is_zero_approx(horizontal_delta):
		return _facing_direction
	return 1 if horizontal_delta > 0.0 else -1

func _update_warning_line() -> void:
	if _state != AttackState.WINDUP or _warning_line == null:
		return
	var start := _locked_muzzle_position
	var motion := _locked_direction * maximum_attack_range
	_rivet_query.transform = Transform2D(0.0, start)
	_rivet_query.motion = motion
	var fractions := get_world_2d().direct_space_state.cast_motion(_rivet_query)
	var end := start + motion
	if fractions.size() >= 2 and fractions[0] < 1.0:
		end = start + motion * fractions[0]
	_warning_line.points = PackedVector2Array([to_local(start), to_local(end)])
	_warning_mark.modulate.a = 0.65 + 0.35 * (0.5 + 0.5 * sin(Time.get_ticks_msec() * 0.025))

func _is_clear_rivet_spawn(muzzle: Vector2, direction: Vector2) -> bool:
	var space_state := get_world_2d().direct_space_state
	_rivet_query.transform = Transform2D(0.0, muzzle)
	_rivet_query.motion = Vector2.ZERO
	if not space_state.intersect_shape(_rivet_query, 1).is_empty():
		return false
	var motion := direction.normalized() * RIVET_SPAWN_OFFSET
	_rivet_query.motion = motion
	var fractions := space_state.cast_motion(_rivet_query)
	if fractions.size() < 2 or fractions[0] < 1.0:
		return false
	var spawn_position := muzzle + motion
	_rivet_query.transform = Transform2D(0.0, spawn_position)
	_rivet_query.motion = Vector2.ZERO

func _process_retreat(delta: float) -> void:
	_set_warning_visible(false)
	if not _patrol_area_usable:
		velocity.x = 0.0
		_retreat_remaining = 0.0
		_state = AttackState.APPROACH
		return
	var bounds := _get_safe_center_bounds()
	_retreat_target_x = clampf(_retreat_target_x, bounds.x, bounds.y)
	var away_direction := signf(_retreat_target_x - global_position.x)
	if _retreat_remaining <= POSITION_EPSILON or is_zero_approx(away_direction) or delta <= 0.0 or not _has_ground_ahead(away_direction):
		velocity.x = 0.0
		_retreat_remaining = 0.0
		_state = AttackState.APPROACH
		return
	var allowed_step := minf(move_speed * delta, minf(_retreat_remaining, absf(_retreat_target_x - global_position.x)))
	velocity.x = away_direction * allowed_step / delta

func _begin_close_range_retreat(delta: float) -> void:
	_retreat_used_this_cycle = true
	if not _patrol_area_usable:
		velocity.x = 0.0
		return
	var toward_player := signf(_player.global_position.x - global_position.x)
	if is_zero_approx(toward_player):
		toward_player = float(-_facing_direction)
	var away_direction := -toward_player
	var bounds := _get_safe_center_bounds()
	var destination_x := clampf(
		global_position.x + away_direction * maxf(close_range_retreat_step, 0.0),
		bounds.x,
		bounds.y
	)
	_retreat_target_x = destination_x
	_retreat_remaining = absf(destination_x - global_position.x)
	var retreat_direction := signf(destination_x - global_position.x)
	if _retreat_remaining <= POSITION_EPSILON or is_zero_approx(retreat_direction) or not _has_ground_ahead(retreat_direction):
		velocity.x = 0.0
		return
	_facing_direction = int(retreat_direction)
	_apply_facing_direction()
	_state = AttackState.RETREAT
	_set_warning_visible(false)
	if delta > 0.0:
		var allowed_step := minf(move_speed * delta, _retreat_remaining)
		velocity.x = retreat_direction * allowed_step / delta
		_retreat_remaining = maxf(_retreat_remaining - allowed_step, 0.0)
func _prepare_warning_sound() -> void:
	if _warning_sound.stream != null:
		return
	const sample_rate: int = 22050
	const duration: float = 0.09
	var sample_count := int(sample_rate * duration)
	var pcm := PackedByteArray()
	pcm.resize(sample_count * 2)
	for sample_index in range(sample_count):
		var time := float(sample_index) / float(sample_rate)
		var envelope := minf(time / 0.006, 1.0) * clampf((duration - time) / 0.022, 0.0, 1.0)
		var sample_value := int(sin(TAU * 880.0 * time) * envelope * 9000.0)
		pcm.encode_s16(sample_index * 2, sample_value)
	var warning_stream := AudioStreamWAV.new()
	warning_stream.format = AudioStreamWAV.FORMAT_16_BITS
	warning_stream.mix_rate = sample_rate
	warning_stream.stereo = false
	warning_stream.data = pcm
	_warning_sound.stream = warning_stream
	_warning_sound.volume_db = warning_volume_db

func _update_visuals() -> void:
	if not is_instance_valid(_body_sprite) or not is_instance_valid(_health_fill):
		return
	var hit_flash := _body_sprite.material as ShaderMaterial
	if hit_flash != null:
		hit_flash.set_shader_parameter("flash_amount", 1.0 if _flash_remaining > 0.0 else 0.0)
	var health_ratio := clampf(float(health) / float(maxi(max_health, 1)), 0.0, 1.0)
	_health_fill.scale.x = health_ratio

func _update_body_animation() -> void:
	if not is_instance_valid(_body_sprite):
		return
	var desired_animation: StringName = &"idle"
	if _shot_visual_remaining > 0.0:
		desired_animation = &"fire"
	elif _state == AttackState.WINDUP:
		desired_animation = &"aim"
	elif _state == AttackState.RECOVERY:
		desired_animation = &"recovery"
	elif absf(velocity.x) > 1.0:
		desired_animation = &"walk"
	_set_body_animation(desired_animation)

func _set_body_animation(animation: StringName) -> void:
	if not is_instance_valid(_body_sprite):
		return
	if _body_sprite.animation != animation or not _body_sprite.is_playing():
		_body_sprite.play(animation)

func _apply_facing_direction() -> void:
	if is_instance_valid(_visuals):
		_visuals.scale.x = float(_facing_direction)

local function write_atlas(spec)
  local image = Image{ fromFile = spec.input }
  assert(image.width == spec.width and image.height == spec.height,
    "Dimensões inesperadas: " .. spec.input)
  assert(image.colorMode == ColorMode.RGB,
    "Esperado PNG RGBA/RGB de 8 bits sem indexação: " .. spec.input)

  local output = assert(io.open(spec.output, "w"), "Não foi possível gravar " .. spec.output)
  output:write("[gd_resource type=\"ImageTexture\" load_steps=2 format=3]\n\n")
  output:write("[sub_resource type=\"Image\" id=\"" .. spec.image_id .. "\"]\n")
  output:write("data = { \"data\": PackedByteArray(\n")

  local chunk = {}
  local emitted = 0
  local total = spec.width * spec.height * 4
  local function flush()
    if #chunk == 0 then return end
    local text = table.concat(chunk, ", ")
    if emitted + #chunk < total then text = text .. "," end
    output:write(text, "\n")
    emitted = emitted + #chunk
    chunk = {}
  end

  for y = 0, spec.height - 1 do
    for x = 0, spec.width - 1 do
      local value = image:getPixel(x, y)
      chunk[#chunk + 1] = tostring(pixel.rgbaR(value))
      chunk[#chunk + 1] = tostring(pixel.rgbaG(value))
      chunk[#chunk + 1] = tostring(pixel.rgbaB(value))
      chunk[#chunk + 1] = tostring(pixel.rgbaA(value))
      if #chunk >= 4096 then flush() end
    end
  end
  flush()

  output:write("), \"format\": \"RGBA8\", \"height\": ", tostring(spec.height),
    ", \"mipmaps\": false, \"width\": ", tostring(spec.width), " }\n\n")
  output:write("[resource]\nimage = SubResource(\"", spec.image_id, "\")\n")
  output:close()
  print(string.format("Empacotado: %s (%dx%d RGBA8; %d bytes)",
    spec.output, spec.width, spec.height, total))
end
