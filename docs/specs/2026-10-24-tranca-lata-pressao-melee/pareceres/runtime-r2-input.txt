func _pursue_on_current_support(
	player: PhysicsBody2D,
	delta: float,
	supports: Array[SolidPlatform],
	allow_detour: bool = true
) -> void:
	var player_center := _get_player_collider_center(player)
	var attack_center := global_position + attack_center_offset
	var delta_x := player_center.x - attack_center.x
	var delta_y := player_center.y - attack_center.y
	if absf(delta_x) > PURSUIT_DIRECTION_DEADZONE:
		_patrol_direction = 1 if delta_x > 0.0 else -1
	var preferred_distance := minf(PREFERRED_MELEE_DISTANCE, _effective_attack_radius())
	var preferred_horizontal_reach := sqrt(maxf(
		preferred_distance * preferred_distance - delta_y * delta_y,
		0.0
	))
	var desired_x := global_position.x
	var clear_shot := _has_clear_line_of_sight(player, attack_center)
	var center_distance := attack_center.distance_to(player_center)
	var needs_horizontal_approach := center_distance > preferred_distance \
		or (not clear_shot and absf(delta_x) > PURSUIT_DIRECTION_DEADZONE)
	if needs_horizontal_approach:
		var direction := signf(delta_x)
		if absf(delta_x) <= PURSUIT_DIRECTION_DEADZONE or is_zero_approx(direction):
			direction = float(_patrol_direction)
		var stop_reach := maxf(preferred_horizontal_reach - PURSUIT_STOP_PADDING, 0.0) if clear_shot else 0.0
		desired_x = player_center.x - direction * stop_reach - attack_center_offset.x
	var moved_x := _move_toward_pursuit_x(desired_x, delta, _bound_support)
	if absf(moved_x) >= 0.1:
		_pursuit_stuck_remaining = 0.0
	elif needs_horizontal_approach:
		_pursuit_stuck_remaining += delta
	else:
		_pursuit_stuck_remaining = 0.0
	if _pursuit_stuck_remaining >= PURSUIT_STUCK_TIMEOUT \
		and _pursuit_replan_remaining <= 0.0:
		var stalled_player := _find_melee_engagement_player(global_position + attack_center_offset, true)
		if stalled_player != null:
			_known_player = stalled_player
			_begin_windup(stalled_player)
			return
		if allow_detour:
			_try_begin_pursuit_detour(_bound_support, player, supports)
		else:
			_pursuit_replan_remaining = PURSUIT_REPLAN_INTERVAL
	if _attack_state == AttackState.IDLE:
		_update_visuals()

func _find_melee_engagement_player(
	area_center: Vector2,
	allow_stuck_fallback: bool = false
) -> PhysicsBody2D:
	var player := _find_player_in_attack_area(area_center)
	if player == null or allow_stuck_fallback:
		return player
	var preferred_distance := minf(PREFERRED_MELEE_DISTANCE, _effective_attack_radius())
	if area_center.distance_to(_get_player_collider_center(player)) <= preferred_distance:
		return player
	return null

func _find_player_in_attack_area(area_center: Vector2) -> PhysicsBody2D:
	if _area_query == null or _area_shape == null:
		return null
	_area_shape.radius = _effective_attack_radius()
	_area_query.transform = Transform2D(0.0, area_center)
	var overlaps: Array[Dictionary] = get_world_2d().direct_space_state.intersect_shape(
		_area_query,
		MAX_QUERY_RESULTS
	)
	for overlap in overlaps:
		var collider: Variant = overlap.get("collider")
		if not collider is PhysicsBody2D:
			continue
		var player_body := collider as PhysicsBody2D
		if not _is_alive_player(player_body):
			continue
		if _has_clear_line_of_sight(player_body, area_center):
			return player_body
	return null
