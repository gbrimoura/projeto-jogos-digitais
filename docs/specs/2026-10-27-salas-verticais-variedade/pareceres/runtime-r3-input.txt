func _managed_can_jump_between(source: Rect2, target: Rect2, rects: Array[Rect2], source_index: int, top_open: bool) -> bool:
	var required_rise := source.position.y - target.position.y
	if required_rise < 0.0 or required_rise > 128.0:
		return false
	var movement := _managed_movement_values()
	var max_speed := float(movement.get("move_speed", 260.0))
	var acceleration := float(movement.get("acceleration", 1600.0))
	var deceleration := float(movement.get("deceleration", 2200.0))
	var gravity := float(movement.get("gravity", 1600.0))
	var jump_velocity := float(movement.get("jump_velocity", 640.0))
	var max_fall_speed := float(movement.get("max_fall_speed", 900.0))
	var dt := 1.0 / float(ProjectSettings.get_setting("physics/common/physics_ticks_per_second", 60))
	var launch_min := source.position.x + 14.0
	var launch_max := source.end.x - 14.0
	if launch_max < launch_min:
		return false
	var launch_points := _managed_launch_points(source, target, launch_min, launch_max)
	var speed_candidates: Array[float] = [0.0, -100.0, 100.0, -180.0, 180.0, -260.0, 260.0]
	for launch_x in launch_points:
		var approach_axis := 0.0
		if launch_x < target.position.x + 22.0:
			approach_axis = 1.0
		elif launch_x > target.end.x - 22.0:
			approach_axis = -1.0
		for initial_speed in speed_candidates:
			if not is_zero_approx(initial_speed) and signf(initial_speed) != approach_axis:
				continue
			var runway := initial_speed * initial_speed / maxf(2.0 * acceleration, 1.0)
			if initial_speed > 0.0 and launch_x - launch_min + 0.01 < runway:
				continue
			if initial_speed < 0.0 and launch_max - launch_x + 0.01 < runway:
				continue
			var root := Vector2(launch_x, source.position.y)
			var vx := initial_speed
			var vy := 0.0
			var candidate_blocked := false
			for frame in 120:
				vy = minf(vy + gravity * dt, max_fall_speed)
				if frame == 0:
					vy = -jump_velocity
				var horizontal_rate := acceleration if not is_zero_approx(approach_axis) else deceleration
				vx = move_toward(vx, approach_axis * max_speed, horizontal_rate * dt)
				var next := root + Vector2(vx, vy) * dt
				var body := Rect2(Vector2(next.x - 14.0, next.y - 76.0), Vector2(28.0, 76.0))
				if body.position.x < 64.0 or body.end.x > 1088.0:
					candidate_blocked = true
					break
				var descending_landing := vy > 0.0 and root.y <= target.position.y and next.y >= target.position.y
				if descending_landing:
					var crossing_fraction := clampf((target.position.y - root.y) / (next.y - root.y), 0.0, 1.0)
					var landing_x := lerpf(root.x, next.x, crossing_fraction)
					if landing_x >= target.position.x + 22.0 and landing_x <= target.end.x - 22.0:
						return true
				for blocker_index in rects.size():
					if blocker_index == source_index:
						continue
					if _positive_overlap_area(body, rects[blocker_index]) <= 0.01:
						continue
					# Plataformas sólidas também bloqueiam subida e colisões laterais.
					# Só a aterrissagem descendente no topo do alvo é uma rota válida.
					candidate_blocked = true
					break
				if candidate_blocked:
					break
				if body.position.y < 64.0 and body.end.y > -32.0:
					var portal_x := float(_managed_room_descriptor.get("exit_offset", -1.0))
					var inside_open_hatch := top_open \
						and _managed_room_descriptor.get("exit_dir", Vector2i.ZERO) == Vector2i(0, -1) \
						and body.position.x >= portal_x - 128.0 \
						and body.end.x <= portal_x + 128.0
					if not inside_open_hatch:
						candidate_blocked = true
						break
				root = next
				if root.y > source.position.y + 180.0:
					break
			# Um bloqueio invalida apenas este lançamento; os demais são testados.
			if candidate_blocked:
				continue
	return false

func _managed_can_drop_between(source: Rect2, target: Rect2, rects: Array[Rect2], source_index: int) -> bool:
	if target.position.y <= source.position.y or target.position.y - source.position.y > 400.0:
		return false
	var movement := _managed_movement_values()
	var max_speed := float(movement.get("move_speed", 260.0))
	var acceleration := float(movement.get("acceleration", 1600.0))
	var gravity := float(movement.get("gravity", 1600.0))
	var max_fall_speed := float(movement.get("max_fall_speed", 900.0))
	var dt := 1.0 / float(ProjectSettings.get_setting("physics/common/physics_ticks_per_second", 60))
	var launch_min := source.position.x - 14.0
	var launch_max := source.end.x + 14.0
	var launch_points := _managed_launch_points(source, target, launch_min, launch_max)
	for launch_x in launch_points:
		for direction in [-1.0, 1.0]:
			var root := Vector2(launch_x, source.position.y)
			var vx := 0.0
			var vertical_speed := 0.0
			for _frame in 120:
				var previous := root
				vx = move_toward(vx, direction * max_speed, acceleration * dt)
				root.x += vx * dt
				var still_supported := root.x + 14.0 > source.position.x and root.x - 14.0 < source.end.x
				if still_supported:
					root.y = source.position.y
					vertical_speed = 0.0
				else:
					vertical_speed = minf(vertical_speed + gravity * dt, max_fall_speed)
					root.y += vertical_speed * dt
				var body := Rect2(Vector2(root.x - 14.0, root.y - 76.0), Vector2(28.0, 76.0))
				if body.position.x < 64.0 or body.end.x > 1088.0:
					break
				var descending_landing := not still_supported and previous.y <= target.position.y and root.y >= target.position.y
				if descending_landing:
					var crossing_fraction := clampf((target.position.y - previous.y) / (root.y - previous.y), 0.0, 1.0)
					var landing_x := lerpf(previous.x, root.x, crossing_fraction)
					if landing_x >= target.position.x + 22.0 and landing_x <= target.end.x - 22.0:
						return true
				var blocked := false
				for blocker_index in rects.size():
					if blocker_index == source_index:
						continue
					if _positive_overlap_area(body, rects[blocker_index]) > 0.01:
						blocked = true
						break
				if blocked:
					break
				if root.y > target.position.y + 80.0:
					break
	return false

func _managed_launch_points(source: Rect2, target: Rect2, launch_min: float, launch_max: float) -> Array[float]:
	var points: Array[float] = []
	var candidates: Array[float] = [
		launch_min + 8.0,
		launch_min + 32.0,
		source.get_center().x,
		launch_max - 32.0,
		launch_max - 8.0,
		target.position.x - 64.0,
		target.position.x - 32.0,
		target.position.x + 22.0,
		target.get_center().x,
		target.end.x - 22.0,
		target.end.x + 32.0,
		target.end.x + 64.0,
	]
	for candidate in candidates:
		var point := clampf(snappedf(candidate, 8.0), launch_min, launch_max)
		if not points.has(point):
			points.append(point)
	return points

	if force_canonical:
		var canonical_entries := _canonical_managed_entries(family, entry_dir, exit_dir, entry_offset, exit_offset)
		if _managed_entries_are_safe(canonical_entries, family, entry_dir, exit_dir, entry_offset, exit_offset):
			return _managed_layout_result(canonical_entries, family, seed_used, entry_dir, exit_dir, entry_offset, exit_offset, true)
		return _invalid_result(seed_used, "fallback canônico da família %s falhou no contrato geométrico" % String(family))
	for _attempt in 12:
		var entries := _build_managed_entries(family, rng, entry_dir, exit_dir, entry_offset, exit_offset)
		if _managed_entries_are_safe(entries, family, entry_dir, exit_dir, entry_offset, exit_offset):
			return _managed_layout_result(entries, family, seed_used, entry_dir, exit_dir, entry_offset, exit_offset, false)
	var canonical := _canonical_managed_entries(family, entry_dir, exit_dir, entry_offset, exit_offset)
	if _managed_entries_are_safe(canonical, family, entry_dir, exit_dir, entry_offset, exit_offset):
		return _managed_layout_result(canonical, family, seed_used, entry_dir, exit_dir, entry_offset, exit_offset, true)
	return _invalid_result(seed_used, "fallback canônico da família %s falhou no contrato geométrico" % String(family))


===== room_layout_generator.gd build =====
static func _build_managed_entries(
	family: StringName,
	rng: RandomNumberGenerator,
	entry_dir: Vector2i,
	exit_dir: Vector2i,
	entry_offset: float,
	exit_offset: float
) -> Array[Dictionary]:
	var count := 3 if family == &"catwalk" else (5 if family == &"roof_five" else 4)
	var centers: Array[float] = []
	var tops: Array[float] = []
	var widths: Array[float] = []
	centers.resize(count)
	tops.resize(count)
	widths.resize(count)
	var is_roof := family == &"roof_four" or family == &"roof_five"
	if is_roof:
		var portal_x := entry_offset if entry_dir == Vector2i(0, -1) else exit_offset
		var horizontal_sign := -1.0 if portal_x == 248.0 else 1.0
		var allowed_steps: Array[int] = [64, 72, 80] if family == &"roof_five" else [80, 88, 96, 104, 112]
		var rises: Array[int] = []
		for _tries in 40:
			rises.clear()
			var total := 0
			for _step in count - 1:
				var rise: int = allowed_steps[rng.randi_range(0, allowed_steps.size() - 1)]
				rises.append(rise)
				total += rise
			if total == 288:
				break
		var rise_sum := 0
		for rise in rises:
			rise_sum += rise
		if rises.size() != count - 1 or rise_sum != 288:
			rises.clear()
			rises.resize(count - 1)
			for index in rises.size():
				rises[index] = 72 if family == &"roof_five" else 96
		tops[0] = 448.0
		for index in count - 1:
			tops[index + 1] = tops[index] - float(rises[index])
		for index in count:
			widths[index] = 104.0 + float(rng.randi_range(0, 3) * 8) if family == &"roof_five" else 144.0 + float(rng.randi_range(0, 4) * 8)
		centers[count - 1] = portal_x
		for index in range(count - 1, 0, -1):
			var gap := float(40 + rng.randi_range(0, 5) * 8)
			var center_step := widths[index] * 0.5 + widths[index - 1] * 0.5 + gap
			centers[index - 1] = centers[index] - horizontal_sign * center_step
	else:
		var base_tops: Array[float]
		if family == &"catwalk":
			base_tops = [448.0, 400.0, 448.0]
		elif family == &"ridge":
			base_tops = [448.0, 384.0, 352.0, 416.0]
		else:
			base_tops = [400.0, 448.0, 448.0, 400.0]
		for index in count:
			var offset_steps := rng.randi_range(-2, 2)
			if index == 0:
				offset_steps = rng.randi_range(-2, 0) if family != &"valley" else rng.randi_range(-1, 1)
			tops[index] = clampf(base_tops[index] + float(offset_steps * 8), 376.0, 448.0)
			widths[index] = float(144 + rng.randi_range(0, 5) * 8)
		var spawn_x := 160.0 if entry_dir == Vector2i(-1, 0) else (992.0 if entry_dir == Vector2i(1, 0) else entry_offset)
		var horizontal_sign := -1.0 if entry_dir == Vector2i(1, 0) or (entry_dir.y != 0 and spawn_x > 576.0) else 1.0
		var center_min := 240.0
		var center_max := 304.0
		if horizontal_sign < 0.0:
			center_min = 848.0
			center_max = 912.0
		if entry_dir.y != 0:
			center_min = spawn_x - 24.0
			center_max = spawn_x + 24.0
		centers[0] = float(rng.randi_range(roundi(center_min / 8.0), roundi(center_max / 8.0)) * 8)
		for index in range(1, count):
			var gap := float(48 + rng.randi_range(0, 6) * 8)
			centers[index] = centers[index - 1] + horizontal_sign * (widths[index - 1] * 0.5 + widths[index] * 0.5 + gap)
	var entries: Array[Dictionary] = []
	for index in count:
		centers[index] = snappedf(centers[index], 8.0)
		var size := Vector2(widths[index], 16.0)
		var position := Vector2(centers[index], tops[index] + 8.0)
		var rect := Rect2(position - size * 0.5, size)
		entries.append({"position": position, "size": size, "rect": rect})
	return entries


===== room_layout_generator.gd geometry safety =====
static func _managed_entries_are_safe(
	entries: Array[Dictionary],
	family: StringName,
	entry_dir: Vector2i,
	exit_dir: Vector2i,
	entry_offset: float,
	exit_offset: float
) -> bool:
	var expected_count := 3 if family == &"catwalk" else (5 if family == &"roof_five" else 4)
	if entries.size() != expected_count or entry_dir == exit_dir:
		return false
	if exit_dir.x == 0 and absf(exit_offset - 248.0) > 0.01 and absf(exit_offset - 904.0) > 0.01:
		return false
	if entry_dir.x == 0 and absf(entry_offset - 248.0) > 0.01 and absf(entry_offset - 904.0) > 0.01:
		return false
	var is_roof := family == &"roof_four" or family == &"roof_five"
	if (entry_dir == Vector2i(0, -1) or exit_dir == Vector2i(0, -1)) and not is_roof:
		return false
	var previous := Rect2()
	for index in entries.size():
		var rect: Rect2 = entries[index].get("rect", Rect2())
		if rect.size.y != 16.0 or rect.size.x < MIN_FALLBACK_WIDTH or rect.size.x > MAX_PLATFORM_WIDTH:
			return false
		if rect.position.x < 64.0 or rect.end.x > 1088.0 \
			or fposmod(rect.get_center().x, 8.0) > 0.01 or fposmod(rect.size.x, 8.0) > 0.01 \
			or fposmod(rect.position.y, 8.0) > 0.01:
			return false
		if rect.position.y < 152.0 or rect.position.y > 448.0 or 560.0 - rect.end.y < 96.0:
			return false
		if index > 0:
			if rect.intersects(previous):
				return false
			var gap := maxf(rect.position.x - previous.end.x, previous.position.x - rect.end.x)
			var rise := previous.position.y - rect.position.y
			if is_roof:
				if gap < 40.0 or gap > 80.0 or rise < 64.0 or rise > 112.0:
					return false
			elif gap < 40.0 or gap > 96.0 or absf(rise) > 64.0:
				return false
		previous = rect
	if is_roof:
		var last_rect: Rect2 = entries.back().get("rect", Rect2())
		var expected := entry_offset if entry_dir == Vector2i(0, -1) else exit_offset
		if (entry_dir == Vector2i(0, -1) or exit_dir == Vector2i(0, -1)) \
			and absf(last_rect.get_center().x - expected) > 0.01:
			return false
	elif family == &"catwalk":
		if entries.size() != 3 or tops_are_not_catwalk(entries):
			return false
	elif family == &"ridge":
		if entries.size() != 4 or tops_are_not_ridge(entries):
			return false
	elif family == &"valley":
		if entries.size() != 4 or tops_are_not_valley(entries):
			return false
	return true
