## Subida na perseguição
O problema confirmado por código é acesso/alternância da decolagem, não MIN_APEX72 como teto. Corpo72x96; MAX_RISE144, MAX_DROP200, DX360, VX480, gravity1400, ápice max192 e clearance48 preservados. A aproximação deve continuar por movimento físico seguro sobre apoio atual; plano válido não garante que caminhada esteja livre, então manter stuck timeout/blockededge/replan existentes. Retry deve ser limitado por aresta, não loop por frame: considerar posição atual (inclusive parada real junto à parede), deslocamentos curtos ±16/±32 nativos se úteis e standoffs esquerdo/direito do destino, deduplicados dentro dos bounds do suporte. Não exigir standoff exato dentro da parede, nem fazer retry somente nos extremos doFloor. Novos valores de candidatos são locais de busca, não alcance/física de jogo. Objetivo principal é perseguir Player em apoios superiores; não criar sistema novo de waypoints para patrulha autônoma.

func _refresh_live_player() -> PhysicsBody2D:
	var previous_player := _known_player
	if _is_alive_player(_known_player) and _known_player.get_parent() == get_parent():
		return _known_player
	_known_player = null
	for node in get_tree().get_nodes_in_group("player"):
		var candidate := node as PhysicsBody2D
		if candidate == null or candidate.get_parent() != get_parent():
			continue
		if _is_alive_player(candidate):
			_known_player = candidate
			break
	if previous_player != _known_player or _known_player == null:
		_clear_pursuit_state()
	return _known_player

func _pursue_player(player: PhysicsBody2D, delta: float) -> void:

		_pursue_fallback(player, delta, supports)

		"choose_player_support",
		player,
		supports,
		_jump_helper,
		_pursuit_player_support
	)

		_clear_pursuit_launch_retry()
		_pursuit_player_support = selected_support
		_pursuit_path.clear()

		_pursuit_replan_remaining = 0.0
		_pursuit_stuck_remaining = 0.0
		_pursuit_detour_active = false

		_clear_pursuit_launch_retry()
		_pursuit_goal_support = null
		_pursuit_path.clear()
		_pursuit_detour_active = false
		_pursue_fallback(player, delta, supports)
		return
	if selected_support == _bound_support and not _pursuit_detour_active:
		_clear_pursuit_launch_retry()
		_pursuit_goal_support = selected_support
		_pursuit_path.clear()
		_pursue_on_current_support(player, delta, supports)
		return
	if selected_support != _pursuit_goal_support:
		_clear_pursuit_launch_retry()
		_pursuit_goal_support = selected_support
		_pursuit_path.clear()
		_pursuit_replan_remaining = 0.0
		_pursuit_detour_active = false
	if selected_support == _bound_support and _pursuit_detour_active:
		if not _pursuit_path.is_empty() and _pursuit_path[0] != _bound_support:
			_pursuit_path.clear()
		if _pursuit_path.size() < 2:
			if _pursuit_replan_remaining <= 0.0 \
				and not _try_begin_pursuit_detour(selected_support, player, supports):
				_pursuit_detour_active = false
				_pursuit_path.clear()
				_pursuit_goal_support = selected_support
				_pursue_fallback(player, delta, supports)
				return
		if _pursuit_path.size() < 2:
			_pursuit_detour_active = false
			_pursuit_goal_support = selected_support
			_pursue_fallback(player, delta, supports)
			return
		_pursue_via_route(player, delta, supports)
		return
	if _pursuit_path.size() < 2 or _pursuit_path[0] != _bound_support:
		if _pursuit_replan_remaining <= 0.0:
			var route: Array[SolidPlatform] = _pursuit_helper.call(
				"find_path",
				_bound_support,
				selected_support,
				supports,
				_jump_helper,
				_pursuit_blocked_edges
			)
			if route.size() >= 2:
				_clear_pursuit_launch_retry()
				_pursuit_path = route
			else:
				_pursuit_path.clear()
				_pursuit_detour_active = false
				_pursuit_replan_remaining = PURSUIT_REPLAN_INTERVAL
	if _pursuit_path.size() < 2:
		_pursuit_detour_active = false
		_pursue_fallback(player, delta, supports)
		return
	_pursue_via_route(player, delta, supports)

func _pursue_fallback(
	player: PhysicsBody2D,
	delta: float,
	supports: Array[SolidPlatform]
) -> void:
	if not _is_alive_player(player) or not _ground_support_is_current():
		_stop_patrol()
		return
	_clear_pursuit_launch_retry()
	_pursuit_path.clear()
	_pursuit_detour_active = false
	_pursue_on_current_support(player, delta, supports, false)

func _pursue_on_current_support(
	player: PhysicsBody2D,
	delta: float,
	supports: Array[SolidPlatform],
	allow_detour: bool = true
) -> void:
	var player_center := _get_player_collider_center(player)
	var attack_center := global_position + attack_center_offset
	var delta_x := player_center.x - attack_center.x
	var delta_y := player_center.y - attack_center.y
	if absf(delta_x) > PURSUIT_DIRECTION_DEADZONE:
		_patrol_direction = 1 if delta_x > 0.0 else -1
	var radius := _effective_attack_radius()
	var horizontal_attack_reach := sqrt(maxf(radius * radius - delta_y * delta_y, 0.0))
	var stop_distance := maxf(horizontal_attack_reach - PURSUIT_STOP_PADDING, PURSUIT_DIRECTION_DEADZONE)
	var desired_x := global_position.x
	var clear_shot := _has_clear_line_of_sight(player, attack_center)
	var needs_horizontal_approach := absf(delta_x) > stop_distance \
		or (not clear_shot and absf(delta_x) > PURSUIT_DIRECTION_DEADZONE)
	if needs_horizontal_approach:
		var direction := signf(delta_x)
		if absf(delta_x) <= PURSUIT_DIRECTION_DEADZONE or is_zero_approx(direction):
			direction = float(_patrol_direction)
		var stop_reach := horizontal_attack_reach - PURSUIT_STOP_PADDING if clear_shot else 0.0
		desired_x = player_center.x - direction * maxf(stop_reach, 0.0) - attack_center_offset.x
	var moved_x := _move_toward_pursuit_x(desired_x, delta, _bound_support)
	if not allow_detour:
		_pursuit_stuck_remaining = 0.0
	elif absf(moved_x) >= 0.1:
		_pursuit_stuck_remaining = 0.0
	elif needs_horizontal_approach:
		_pursuit_stuck_remaining += delta
	else:
		_pursuit_stuck_remaining = 0.0
	if allow_detour and _pursuit_stuck_remaining >= PURSUIT_STUCK_TIMEOUT \
		and _pursuit_replan_remaining <= 0.0:
		_try_begin_pursuit_detour(_bound_support, player, supports)
	if _attack_state == AttackState.IDLE:
		_update_visuals()

func _try_begin_pursuit_detour(
	goal_support: SolidPlatform,
	player: PhysicsBody2D,
	supports: Array[SolidPlatform]
) -> bool:
	var toward_player := signf(_get_player_collider_center(player).x - global_position.x)
	if is_zero_approx(toward_player):
		toward_player = float(_patrol_direction)
	var best_score: float = INF
	var best_route: Array[SolidPlatform] = []
	for candidate in supports:
		if candidate == _bound_support or candidate == goal_support or candidate.name == &"Floor":
			continue
		if not bool(_jump_helper.call("can_reach_support", candidate, goal_support)):
			continue
		var data := _get_support_rect(candidate)
		if not bool(data.get("valid", false)):
			continue
		var rect: Rect2 = data.get("rect", Rect2())
		var progress := (rect.get_center().x - global_position.x) * toward_player
		if progress <= 16.0:
			continue
		var approach: Array[SolidPlatform] = _pursuit_helper.call(
			"find_path",
			_bound_support,
			candidate,
			supports,
			_jump_helper,
			_pursuit_blocked_edges
		)
		if approach.size() < 2:
			continue
		var score := absf(_get_player_collider_center(player).x - rect.get_center().x) + float(approach.size()) * 80.0
		if score < best_score:
			best_score = score
			approach.append(goal_support)
			best_route = approach
	if best_route.is_empty():
		_pursuit_replan_remaining = PURSUIT_REPLAN_INTERVAL
		return false
	_pursuit_path = best_route
	_pursuit_goal_support = goal_support
	_pursuit_detour_active = true
	_pursuit_stuck_remaining = 0.0
	_pursuit_replan_remaining = 0.0
	return true

func _pursue_via_route(
	player: PhysicsBody2D,
	delta: float,
	_supports: Array[SolidPlatform]
) -> void:
	var source := _bound_support
	var destination := _pursuit_path[1]
	if source == null or destination == null or not is_instance_valid(destination):
		_clear_pursuit_launch_retry()
		_pursuit_path.clear()
		_pursuit_detour_active = false
		_pursuit_replan_remaining = 0.0
		_stop_patrol()
		return
	var target_x := _get_player_collider_center(player).x
	if _pursuit_path.size() > 2:
		var next_rect: Rect2 = _jump_helper.call("support_rect", _pursuit_path[2])
		target_x = next_rect.get_center().x
	var launch_x := _pursuit_launch_x(source, destination, target_x)
	var moved_x := _move_toward_pursuit_x(launch_x, delta, source)
	var launch_distance := absf(global_position.x - launch_x)
	if launch_distance > 2.0:
		if absf(moved_x) >= 0.1:
			_pursuit_stuck_remaining = 0.0
		else:
			_pursuit_stuck_remaining += delta
		if _pursuit_stuck_remaining >= PURSUIT_STUCK_TIMEOUT:
			var blocked_edge: String = _pursuit_helper.call("edge_key", source, destination)
			if _try_alternate_pursuit_launch(source, destination, target_x, launch_x):
				return
			_clear_pursuit_launch_retry()
			_pursuit_blocked_edges[blocked_edge] = PURSUIT_EDGE_BLOCK_TIME
			_pursuit_path.clear()
			_pursuit_detour_active = false
			_pursuit_replan_remaining = 0.0
			_pursuit_stuck_remaining = 0.0
			_stop_patrol()
		return
	_pursuit_stuck_remaining = 0.0
	if _jump_cooldown_remaining > 0.0:
		return
	if _pursuit_replan_remaining > 0.0:
		return
	var landing_target_x := _get_player_collider_center(player).x
	if _pursuit_path.size() > 2:
		var landing_next_rect: Rect2 = _jump_helper.call("support_rect", _pursuit_path[2])
		landing_target_x = landing_next_rect.get_center().x
	var plan: Dictionary = _jump_helper.call("plan_jump_to", source, destination, landing_target_x)
	_pursuit_replan_remaining = PURSUIT_REPLAN_INTERVAL
	if plan.is_empty():
		var edge: String = _pursuit_helper.call("edge_key", source, destination)
		if _try_alternate_pursuit_launch(source, destination, landing_target_x, launch_x):
			return
		_clear_pursuit_launch_retry()
		_pursuit_blocked_edges[edge] = PURSUIT_EDGE_BLOCK_TIME
		_pursuit_path.clear()
		_pursuit_detour_active = false
		_pursuit_replan_remaining = 0.0
		_stop_patrol()
		return
	_clear_pursuit_launch_retry()
	_start_jump_prepare(plan)

func _try_alternate_pursuit_launch(
	source: SolidPlatform,
	destination: SolidPlatform,
	landing_target_x: float,
	current_launch_x: float
) -> bool:
	if _jump_helper == null or _pursuit_helper == null:
		return false
	if source == null or destination == null or not is_instance_valid(source) or not is_instance_valid(destination):
		return false
	var edge: String = _pursuit_helper.call("edge_key", source, destination)
	if _pursuit_retry_launch_edge == edge:
		_pursuit_retry_launch_index += 1
		if _pursuit_retry_launch_index >= _pursuit_retry_launch_candidates.size():
			return false
		_pursuit_stuck_remaining = 0.0
		return true

	var source_rect: Rect2 = _jump_helper.call("support_rect", source)
	var destination_rect: Rect2 = _jump_helper.call("support_rect", destination)
	var bounds := _pursuit_bounds_for_support(source)
	var raw_candidates: Array[float] = []
	if destination_rect.position.y > source_rect.position.y + 1.0:
		# Keep the existing single opposite-edge retry for descents.
		var alternate_x := bounds.x if absf(current_launch_x - bounds.y) <= 2.0 else bounds.y
		raw_candidates.append(alternate_x)
	else:
		# Source extremes can lead into a room wall. Try nearby positions and
		# actual destination standoffs, with the current physical position first.
		var current_x := global_position.x
		raw_candidates.append(current_x)
		var toward_destination := signf(destination_rect.get_center().x - current_x)
		if is_zero_approx(toward_destination):
			toward_destination = float(_patrol_direction)
		for offset in [16.0, 32.0]:
			raw_candidates.append(current_x + toward_destination * float(offset))
			raw_candidates.append(current_x - toward_destination * float(offset))
		var body := _get_collision_rect()
		var left_standoff := destination_rect.position.x - body.end.x - PURSUIT_EDGE_CLEARANCE
		var right_standoff := destination_rect.end.x - body.position.x + PURSUIT_EDGE_CLEARANCE
		if absf(current_x - left_standoff) <= absf(current_x - right_standoff):
			raw_candidates.append(left_standoff)
			raw_candidates.append(right_standoff)
		else:
			raw_candidates.append(right_standoff)
			raw_candidates.append(left_standoff)

	var checked_candidates: Array[float] = []
	var validated_candidates: Array[float] = []
	for candidate_x in raw_candidates:
		if candidate_x < bounds.x - 0.01 or candidate_x > bounds.y + 0.01:
			continue
		if absf(candidate_x - current_launch_x) <= 2.0:
			continue
		var unique := true
		for existing_x in checked_candidates:
			if absf(candidate_x - existing_x) <= 2.0:
				unique = false
				break
		if not unique:
			continue
		checked_candidates.append(candidate_x)
		var alternate_plan: Dictionary = _jump_helper.call(
			"plan_jump_to_from",
			source,
			destination,
			landing_target_x,
			candidate_x
		)
		if not alternate_plan.is_empty():
			validated_candidates.append(candidate_x)
	if validated_candidates.is_empty():
		return false
	_pursuit_retry_launch_edge = edge
	_pursuit_retry_launch_candidates = validated_candidates
	_pursuit_retry_launch_index = 0
	_pursuit_stuck_remaining = 0.0
	return true

func _pursuit_launch_x(source: SolidPlatform, destination: SolidPlatform, target_x: float) -> float:
	var edge: String = _pursuit_helper.call("edge_key", source, destination)
	if _pursuit_retry_launch_edge == edge:
		if _pursuit_retry_launch_index >= 0 \
			and _pursuit_retry_launch_index < _pursuit_retry_launch_candidates.size():
			return _pursuit_retry_launch_candidates[_pursuit_retry_launch_index]
		_clear_pursuit_launch_retry()
	if not _pursuit_retry_launch_edge.is_empty():
		_clear_pursuit_launch_retry()
	var source_data := _get_support_rect(source)
	var destination_data := _get_support_rect(destination)
	if not bool(source_data.get("valid", false)) or not bool(destination_data.get("valid", false)):
		return global_position.x
	var source_rect: Rect2 = source_data.get("rect", Rect2())
	var destination_rect: Rect2 = destination_data.get("rect", Rect2())
	var bounds := _pursuit_bounds_for_support(source)
	if destination.name == &"Floor":
		var direction := signf(target_x - source_rect.get_center().x)
		if is_zero_approx(direction):
			var left_distance := absf(global_position.x - bounds.x) + absf(target_x - source_rect.position.x)
			var right_distance := absf(global_position.x - bounds.y) + absf(target_x - source_rect.end.x)
			direction = -1.0 if left_distance <= right_distance else 1.0
		return bounds.y if direction > 0.0 else bounds.x
	if destination_rect.position.y < source_rect.position.y - 1.0:
		var body := _get_collision_rect()
		var left_standoff := destination_rect.position.x - body.end.x - PURSUIT_EDGE_CLEARANCE
		var right_standoff := destination_rect.end.x - body.position.x + PURSUIT_EDGE_CLEARANCE
		var left_valid := left_standoff >= bounds.x and left_standoff <= bounds.y
		var right_valid := right_standoff >= bounds.x and right_standoff <= bounds.y
		if left_valid and right_valid:
			var left_distance := absf(global_position.x - left_standoff)
			var right_distance := absf(global_position.x - right_standoff)
			if absf(left_distance - right_distance) > 1.0:
				return left_standoff if left_distance < right_distance else right_standoff
			var toward_goal := signf(target_x - global_position.x)
			if is_zero_approx(toward_goal):
				toward_goal = signf(target_x - destination_rect.get_center().x)
			if is_zero_approx(toward_goal):
				toward_goal = float(_patrol_direction)
			return right_standoff if toward_goal > 0.0 else left_standoff
		if left_valid:
			return left_standoff
		if right_valid:
			return right_standoff
	var direction := signf(destination_rect.get_center().x - global_position.x)
	if is_zero_approx(direction):
		direction = signf(target_x - global_position.x)
	if is_zero_approx(direction):
		direction = float(_patrol_direction)
	return bounds.y if direction > 0.0 else bounds.x

func _pursuit_bounds_for_support(support: SolidPlatform) -> Vector2:
	var data := _get_support_rect(support)
	if not bool(data.get("valid", false)):
		return Vector2(global_position.x, global_position.x)
	var rect: Rect2 = data.get("rect", Rect2())
	var body := _get_collision_rect()
	var minimum_x := rect.position.x + edge_margin - body.position.x
	var maximum_x := rect.end.x - edge_margin - body.end.x
	if maximum_x < minimum_x:
		return Vector2(global_position.x, global_position.x)
	return Vector2(minimum_x, maximum_x)

func _move_toward_pursuit_x(target_x: float, delta: float, support: SolidPlatform) -> float:
	velocity = Vector2.ZERO
	_patrol_moving = false
	if delta <= 0.0 or not is_instance_valid(support):
		return 0.0
	var bounds := _pursuit_bounds_for_support(support)
	var destination_x := clampf(target_x, bounds.x, bounds.y)
	var dx := destination_x - global_position.x
	if absf(dx) <= MOVEMENT_EPSILON:
		_update_visuals()
		return 0.0
	var direction := signf(dx)
	if absf(dx) > PURSUIT_DIRECTION_DEADZONE:
		_patrol_direction = int(direction)
	var requested_dx := direction * minf(absf(dx), PATROL_SPEED * delta)
	var collision: KinematicCollision2D = move_and_collide(Vector2(requested_dx, 0.0), true, 0.0)
	var safe_dx := requested_dx
	if collision != null:
		safe_dx = collision.get_travel().x
	if signf(safe_dx) != signf(requested_dx):
		safe_dx = 0.0
	safe_dx = signf(requested_dx) * minf(absf(safe_dx), absf(requested_dx))
	if absf(safe_dx) >= MOVEMENT_EPSILON:
		global_position.x += safe_dx
		_patrol_moving = true
		velocity.x = safe_dx / delta
	_update_visuals()
	return safe_dx
func _sync_shockwave_progress() -> void:
	if not is_instance_valid(_attack_visual) or not _attack_visual.has_method("set_shockwave_progress"):
		return
	var progress := 0.0
	if _attack_state == AttackState.ACTIVE:
		var active_progress := 1.0 - _state_remaining / maxf(active_window, 0.05)
		progress = active_progress * (SHOCKWAVE_VISUAL_DURATION - SHOCKWAVE_RECOVERY_TAIL) \
			/ SHOCKWAVE_VISUAL_DURATION
	elif _attack_state == AttackState.RECOVERY:
		var tail_elapsed := clampf(
			SHOCKWAVE_RECOVERY_TAIL - _shockwave_tail_remaining,
			0.0,
			SHOCKWAVE_RECOVERY_TAIL
		)
		progress = (SHOCKWAVE_VISUAL_DURATION - SHOCKWAVE_RECOVERY_TAIL + tail_elapsed) \
			/ SHOCKWAVE_VISUAL_DURATION
	_attack_visual.call("set_shockwave_progress", clampf(progress, 0.0, 1.0))

func _ground_support_is_current() -> bool:
	return _bound_support != null and is_instance_valid(_bound_support) \
		and _is_valid_landing_support(_bound_support) and _support_feet_match(_bound_support)

func _support_feet_match(support: SolidPlatform) -> bool:
	var support_data := _get_support_rect(support)
	if not bool(support_data.get("valid", false)):
		return false
	var rect: Rect2 = support_data.get("rect", Rect2())
	return absf(global_position.y + _get_collision_rect().end.y - rect.position.y) <= 1.0

func _try_begin_jump() -> void:
	_jump_cooldown_remaining = JUMP_COOLDOWN
	if _jump_helper == null or not _ground_support_is_current() or _attack_state != AttackState.IDLE:
		return
	var plan: Dictionary = _jump_helper.call("plan_jump", _bound_support, _patrol_direction)
	if plan.is_empty():
		return
	_start_jump_prepare(plan)
func _start_jump_prepare(plan: Dictionary) -> void:
	if plan.is_empty() or _attack_state != AttackState.IDLE or not _ground_support_is_current():
		return
	_jump_plan = plan
	_jump_cooldown_remaining = JUMP_COOLDOWN
	_jump_facing_direction = 1 if float(_jump_plan["landing_x"]) >= global_position.x else -1
	_locomotion_state = LocomotionState.PREPARE
	_state_remaining = JUMP_PREPARE_TIME
	_patrol_moving = false
	velocity = Vector2.ZERO
	_update_visuals()
func _update_jump_prepare(delta: float) -> void:
	var detected_player := _find_player_in_attack_area(global_position + attack_center_offset)
	if detected_player != null:
		_abort_jump_prepare()
		_known_player = detected_player
		_begin_windup(detected_player)
		return
	if not _ground_support_is_current():
		_abort_jump_prepare()
		return
	_state_remaining = maxf(_state_remaining - delta, 0.0)
	if _state_remaining > 0.0:
		_update_visuals()
		return
	detected_player = _find_player_in_attack_area(global_position + attack_center_offset)
	if detected_player != null:
		_abort_jump_prepare()
		_known_player = detected_player
		_begin_windup(detected_player)
		return
	var refreshed: Dictionary = _jump_helper.call("revalidate_plan", _jump_plan, _bound_support)
	if refreshed.is_empty():
		_abort_jump_prepare()
		return
	_jump_plan = refreshed
	_jump_source = _bound_support
	var source_data := _get_support_rect(_jump_source)
	var source_rect: Rect2 = source_data.get("rect", Rect2())
	_jump_source_top = source_rect.position.y
	_jump_facing_direction = 1 if float(_jump_plan["landing_x"]) >= global_position.x else -1
	_patrol_direction = _jump_facing_direction
	_jump_escape_direction = 0
	_jump_elapsed = 0.0
	_jump_impulse_remaining = JUMP_IMPULSE_TIME
	velocity = _jump_plan["launch_velocity"]
	_locomotion_state = LocomotionState.FLIGHT
	_update_visuals()

func _abort_jump_prepare() -> void:
	_jump_plan.clear()
	_state_remaining = 0.0
	_jump_facing_direction = _patrol_direction
	_jump_cooldown_remaining = JUMP_COOLDOWN
	_locomotion_state = LocomotionState.GROUND
	velocity = Vector2.ZERO
	_patrol_moving = false
	_update_visuals()
func _start_unplanned_fall() -> void:
	if _bound_support != null and is_instance_valid(_bound_support):
		remove_collision_exception_with(_bound_support)
	_bound_support = null
	_patrol_enabled = false
	_patrol_moving = false
	_jump_plan.clear()
	_jump_source = null
	_jump_elapsed = 0.0
	_jump_escape_direction = 0
	velocity = Vector2.ZERO
	_locomotion_state = LocomotionState.FALLING

func _advance_airborne(delta: float) -> void:
	if delta <= 0.0:
		return
	_release_jump_source_if_clear()
	if _jump_elapsed >= JUMP_MAX_FLIGHT_TIME and _jump_escape_direction == 0:
		velocity.x = 0.0
	var remaining := delta
	while remaining > 0.00001:
		var final_velocity := velocity + Vector2(0.0, JUMP_GRAVITY * remaining)
		var max_speed := maxf(velocity.length(), final_velocity.length())
		var step_time := minf(remaining, JUMP_MAX_SEGMENT / maxf(max_speed, 1.0))
		# End the source-support exception at the same analytic clearance used by route casts.
		if _locomotion_state == LocomotionState.FLIGHT and is_instance_valid(_jump_source) and velocity.y < 0.0:
			var feet_y := global_position.y + _get_collision_rect().end.y
			var clearance := feet_y - (_jump_source_top - JumpSolver.EXIT_CLEARANCE)
			var discriminant := velocity.y * velocity.y - 2.0 * JUMP_GRAVITY * clearance
			if clearance > 0.0 and discriminant >= 0.0:
				var clear_time := (-velocity.y - sqrt(discriminant)) / JUMP_GRAVITY
				if clear_time > 0.000001:
					step_time = minf(step_time, clear_time)
		var motion := velocity * step_time + Vector2(0.0, 0.5 * JUMP_GRAVITY * step_time * step_time)
		while motion.length() > JUMP_MAX_SEGMENT + 0.0001:
			step_time *= 0.5
			motion = velocity * step_time + Vector2(0.0, 0.5 * JUMP_GRAVITY * step_time * step_time)
		var collision := move_and_collide(motion, false, JUMP_SAFE_MARGIN)
		if collision == null:
			# The executed displacement and velocity now use the same substep duration.
			velocity += Vector2(0.0, JUMP_GRAVITY * step_time)
			remaining -= step_time
			_jump_elapsed += step_time
			if _jump_impulse_remaining > 0.0:
				_jump_impulse_remaining = maxf(_jump_impulse_remaining - step_time, 0.0)
		else:
			var impact_ratio := 1.0
			if motion.length_squared() > 0.000001:
				impact_ratio = clampf(collision.get_travel().length() / motion.length(), 0.0, 1.0)
			var impact_time := maxf(step_time * impact_ratio, minf(step_time, 0.0001))
			impact_time = minf(impact_time, step_time)
			velocity += Vector2(0.0, JUMP_GRAVITY * impact_time)
			var normal := collision.get_normal()
			var into_surface := velocity.dot(normal)
			if into_surface < 0.0:
				velocity -= normal * into_surface
			if normal.dot(Vector2.UP) >= 0.7:
				var collider := collision.get_collider()
				var landing_support := collider as SolidPlatform
				if _try_bind_landing(landing_support):
					return
				if _is_player_collider(collider):
					_begin_player_top_escape(collider as PhysicsBody2D)
				else:
					if _jump_escape_direction == 0:
						_jump_escape_direction = signi(velocity.x) if absf(velocity.x) > 1.0 else _jump_facing_direction
					velocity.x = float(_jump_escape_direction) * PLAYER_ESCAPE_SPEED
					velocity.y = 0.0
				# A diagonal retry at TOI zero can pin against a body top. Move sideways physically for
				# the rest of this tick, counting each part of the tick exactly once.
				_jump_elapsed += impact_time
				if _jump_impulse_remaining > 0.0:
					_jump_impulse_remaining = maxf(_jump_impulse_remaining - impact_time, 0.0)
				_advance_escape_tangent(maxf(remaining - impact_time, 0.0))
				_update_visuals()
				return
			if _jump_escape_direction != 0 and absf(normal.x) >= 0.7:
				_jump_escape_direction *= -1
				velocity.x = float(_jump_escape_direction) * PLAYER_ESCAPE_SPEED
			remaining -= impact_time
			_jump_elapsed += impact_time
			if _jump_impulse_remaining > 0.0:
				_jump_impulse_remaining = maxf(_jump_impulse_remaining - impact_time, 0.0)
		if _jump_elapsed >= JUMP_MAX_FLIGHT_TIME and _jump_escape_direction == 0:
			velocity.x = 0.0
		_release_jump_source_if_clear()
		_update_visuals()

func _advance_escape_tangent(delta: float) -> void:
	var remaining := maxf(delta, 0.0)
	if remaining <= 0.0:
		return
	if _jump_escape_direction == 0:
		_jump_escape_direction = signi(velocity.x) if absf(velocity.x) > 1.0 else _jump_facing_direction
	velocity.x = float(_jump_escape_direction) * PLAYER_ESCAPE_SPEED
	velocity.y = 0.0
	while remaining > 0.00001:
		var step_time := minf(remaining, JUMP_MAX_SEGMENT / PLAYER_ESCAPE_SPEED)
		var tangent_motion := Vector2(velocity.x * step_time, 0.0)
		var collision := move_and_collide(tangent_motion, false, JUMP_SAFE_MARGIN)
		if collision != null and Vector2(velocity.x, 0.0).dot(collision.get_normal()) < -0.0001:
			_jump_escape_direction *= -1
			velocity.x = float(_jump_escape_direction) * PLAYER_ESCAPE_SPEED
		remaining -= step_time
		_jump_elapsed += step_time
		if _jump_impulse_remaining > 0.0:
			_jump_impulse_remaining = maxf(_jump_impulse_remaining - step_time, 0.0)
		if _jump_elapsed >= JUMP_MAX_FLIGHT_TIME and _jump_escape_direction == 0:
			velocity.x = 0.0
		_release_jump_source_if_clear()
	_update_visuals()
func _release_jump_source_if_clear() -> void:
	if _locomotion_state != LocomotionState.FLIGHT or not is_instance_valid(_jump_source):
		return
	var feet_y := global_position.y + _get_collision_rect().end.y
	if feet_y <= _jump_source_top - JumpSolver.EXIT_CLEARANCE or velocity.y >= 0.0:
		remove_collision_exception_with(_jump_source)
		_jump_source = null
func _is_player_collider(collider: Object) -> bool:
	return collider is PhysicsBody2D and (collider as PhysicsBody2D).is_in_group("player")

func _begin_player_top_escape(player_body: PhysicsBody2D) -> void:
	if _jump_escape_direction == 0:
		var away := global_position.x - player_body.global_position.x
		_jump_escape_direction = signi(away) if absf(away) > 1.0 else _jump_facing_direction
	velocity.x = float(_jump_escape_direction) * PLAYER_ESCAPE_SPEED
	if velocity.y > 0.0:
		velocity.y = 0.0

func _try_bind_landing(support: SolidPlatform) -> bool:
	if support == null or not is_instance_valid(support) or support.get_parent() != get_parent() or not _is_compatible_support(support):
		return false
	var data := _get_support_rect(support)
	if not bool(data.get("valid", false)):
		return false
	var support_rect: Rect2 = data.get("rect", Rect2())
	var body := _get_collision_rect()
	var feet_y := global_position.y + body.end.y
	if absf(feet_y - support_rect.position.y) > 1.0:
		return false
	if global_position.x + body.position.x < support_rect.position.x - 0.01 \
		or global_position.x + body.end.x > support_rect.end.x + 0.01:
		return false
	if _decisions_suspended:
		if _bound_support != null and is_instance_valid(_bound_support):
			remove_collision_exception_with(_bound_support)
		_bound_support = null
		_patrol_enabled = false
		_jump_plan.clear()
		_jump_source = null
		_jump_landing_remaining = 0.0
		_locomotion_state = LocomotionState.LANDING
		velocity = Vector2.ZERO
		_patrol_moving = false
		_clear_pursuit_route()
		return true
	if not _is_valid_landing_support(support):
		return false
	var minimum_x := support_rect.position.x + edge_margin - body.position.x
	var maximum_x := support_rect.end.x - edge_margin - body.end.x
	var can_patrol := maximum_x - minimum_x >= MIN_PATROL_DISTANCE \
		and global_position.x >= minimum_x - 0.01 and global_position.x <= maximum_x + 0.01
	var sweep := _make_body_sweep(minimum_x, maximum_x, body, global_position.y)
	if can_patrol and not _is_world_rect_clear(sweep, support):
		can_patrol = false
	if not can_patrol:
		var stationary := _make_body_sweep(global_position.x, global_position.x, body, global_position.y)
		if not _is_world_rect_clear(stationary, support):
			return false
		minimum_x = global_position.x
		maximum_x = global_position.x
	_patrol_direction = _jump_facing_direction
	_bind_patrol_support(support, minimum_x, maximum_x, can_patrol)
	_clear_pursuit_route()
	_jump_source = null
	_jump_plan.clear()
	_jump_landing_remaining = JUMP_LANDING_TIME
	_jump_cooldown_remaining = JUMP_COOLDOWN
	_jump_elapsed = 0.0
	_jump_escape_direction = 0
	_locomotion_state = LocomotionState.LANDING
	velocity = Vector2.ZERO
	_patrol_moving = false
	_update_visuals()
	return true
func _is_valid_landing_support(support: SolidPlatform) -> bool:
	return support != null and is_instance_valid(support) \
		and support.get_parent() == get_parent() and _node_is_inside_current_scene(support) \
		and _is_compatible_support(support)
func _configure_physics_queries() -> void:
	_area_shape = CircleShape2D.new()
	_area_shape.radius = _effective_attack_radius()
	_area_query = PhysicsShapeQueryParameters2D.new()
	_area_query.shape = _area_shape
	_area_query.collision_mask = PLAYER_MASK
	_area_query.collide_with_bodies = true
	_area_query.collide_with_areas = false
	_area_query.margin = 0.0
	_area_query.exclude = [get_rid()]

	_line_of_sight_query = PhysicsRayQueryParameters2D.new()
	_line_of_sight_query.collision_mask = WORLD_MASK
	_line_of_sight_query.collide_with_bodies = true
	_line_of_sight_query.collide_with_areas = false
	_line_of_sight_query.exclude = [get_rid()]

	_patrol_query_shape = RectangleShape2D.new()
	_patrol_query = PhysicsShapeQueryParameters2D.new()
	_patrol_query.shape = _patrol_query_shape
	_patrol_query.collision_mask = JumpSolver.QUERY_MASK
	_patrol_query.collide_with_bodies = true
	_patrol_query.collide_with_areas = false
	_patrol_query.margin = 0.0

func _effective_attack_radius() -> float:
	return clampf(attack_radius, 16.0, 400.0)

func _find_player_in_attack_area(area_center: Vector2) -> PhysicsBody2D:
	if _area_query == null or _area_shape == null:
		return null
	_area_shape.radius = _effective_attack_radius()
	_area_query.transform = Transform2D(0.0, area_center)
	var overlaps: Array[Dictionary] = get_world_2d().direct_space_state.intersect_shape(
		_area_query,
		MAX_QUERY_RESULTS
	)
	for overlap in overlaps:
		var collider: Variant = overlap.get("collider")
		if not collider is PhysicsBody2D:
			continue
		var player_body := collider as PhysicsBody2D
		if not _is_alive_player(player_body):
			continue
		if _has_clear_line_of_sight(player_body, area_center):
			return player_body
	return null

func _is_alive_player(candidate: PhysicsBody2D) -> bool:
	if not is_instance_valid(candidate):
		return false
	if not candidate.is_in_group("player") or candidate.get_parent() != get_parent() or not candidate.has_method("take_damage"):
		return false
	var dead_value: Variant = candidate.get("is_dead")
	if typeof(dead_value) != TYPE_BOOL or bool(dead_value):
		return false
	return _node_is_inside_current_scene(candidate)

func _is_known_player_defeated() -> bool:
	if _known_player == null or not is_instance_valid(_known_player):
		return false
	var dead_value: Variant = _known_player.get("is_dead")
	return typeof(dead_value) == TYPE_BOOL and bool(dead_value)

func _has_clear_line_of_sight(player_body: PhysicsBody2D, area_center: Vector2) -> bool:
	_line_of_sight_query.from = area_center
	_line_of_sight_query.to = _get_player_collider_center(player_body)
	var obstruction: Dictionary = get_world_2d().direct_space_state.intersect_ray(
		_line_of_sight_query
	)
	return obstruction.is_empty()

func _get_player_collider_center(player_body: PhysicsBody2D) -> Vector2:
	var player_shape := player_body.get_node_or_null("CollisionShape2D") as CollisionShape2D
	if player_shape != null and not player_shape.disabled:
		return player_shape.global_position
	return player_body.global_position

func _begin_windup(player_body: PhysicsBody2D) -> void:
	_shockwave_tail_remaining = 0.0
	velocity = Vector2.ZERO
	_patrol_moving = false
	_locked_player = player_body
	_damaged_player_ids.clear()
	_attack_state = AttackState.WINDUP
	_state_remaining = maxf(windup_duration, 0.05)
	_recovery_fade_remaining = 0.0
	_body_sprite.pause()
	_locked_attack_center = global_position + attack_center_offset
	_select_attack_frame(&"windup", _state_remaining, _state_remaining)
	_update_attack_area_visual()

func _update_windup(delta: float) -> void:
	var effective_duration := maxf(windup_duration, 0.05)
	var before_step := _state_remaining
	_state_remaining = maxf(_state_remaining - delta, 0.0)
	if _state_remaining <= 0.0:
		var overshoot := maxf(delta - before_step, 0.0)
		_enter_active()
		_apply_attack_area()
		if overshoot >= maxf(active_window, 0.05):
			_enter_recovery()
		else:
			_state_remaining = maxf(active_window, 0.05) - overshoot
			_select_attack_frame(&"active", _state_remaining, maxf(active_window, 0.05))
		return
	_select_attack_frame(&"windup", _state_remaining, effective_duration)
	_update_attack_area_visual()

func _enter_active() -> void:
	_attack_state = AttackState.ACTIVE
	_state_remaining = maxf(active_window, 0.05)
	_shockwave_tail_remaining = 0.0
	var feet_origin := Vector2(global_position.x, global_position.y + _get_collision_rect().end.y)
	if is_instance_valid(_attack_dust) and _attack_dust.has_method("burst"):
		_attack_dust.call("burst", feet_origin)
	_select_attack_frame(&"active", _state_remaining, _state_remaining)
	_update_attack_area_visual()
	_update_visuals()

func _update_active(delta: float) -> void:
	_apply_attack_area()
	var effective_duration := maxf(active_window, 0.05)
	_state_remaining = maxf(_state_remaining - delta, 0.0)
	if _state_remaining <= 0.0:
		_enter_recovery()
		return
	_select_attack_frame(&"active", _state_remaining, effective_duration)

func _apply_attack_area() -> void:
	if _defeated_emitted or _attack_state != AttackState.ACTIVE:
		return
	var overlaps: Array[Dictionary] = _find_overlaps_at_locked_center()
	for overlap in overlaps:
		var collider: Variant = overlap.get("collider")
		if not collider is PhysicsBody2D:
			continue
		var player_body := collider as PhysicsBody2D
		if not _is_alive_player(player_body):
			continue
		var player_id := player_body.get_instance_id()
		if _damaged_player_ids.has(player_id):
			continue
		if not _has_clear_line_of_sight(player_body, _locked_attack_center):
			continue
		_damaged_player_ids[player_id] = true
		var damage_result: Variant = player_body.call(
			"take_damage",
			maxi(attack_damage, 1),
			self
		)
		if typeof(damage_result) == TYPE_BOOL and bool(damage_result):
			_set_attack_visual(&"impact", 1.0)

func _find_overlaps_at_locked_center() -> Array[Dictionary]:
	_area_shape.radius = _effective_attack_radius()
	_area_query.transform = Transform2D(0.0, _locked_attack_center)
	return get_world_2d().direct_space_state.intersect_shape(
		_area_query,
		MAX_QUERY_RESULTS
	)

func _enter_recovery() -> void:
	var active_just_ended := _attack_state == AttackState.ACTIVE
	_attack_state = AttackState.RECOVERY
	_state_remaining = maxf(recovery_duration, 0.05)
	_recovery_fade_remaining = maxf(recovery_fade_duration, 0.0)
	_shockwave_tail_remaining = SHOCKWAVE_RECOVERY_TAIL if active_just_ended else 0.0
	_select_attack_frame(&"recovery", _state_remaining, _state_remaining)
	_update_attack_area_visual()
	_update_visuals()

func _update_recovery(delta: float) -> void:
	var effective_duration := maxf(recovery_duration, 0.05)
	_state_remaining = maxf(_state_remaining - delta, 0.0)
	_recovery_fade_remaining = maxf(_recovery_fade_remaining - delta, 0.0)
	_shockwave_tail_remaining = maxf(_shockwave_tail_remaining - delta, 0.0)
	_update_attack_area_visual()
	if _state_remaining <= 0.0:
		_set_attack_visual(&"hidden", 0.0)
		_attack_state = AttackState.IDLE
		_state_remaining = 0.0
		_recovery_fade_remaining = 0.0
		_locked_player = null
		_locked_attack_center = Vector2.ZERO
		_damaged_player_ids.clear()
		_update_visuals()
		return
	_select_attack_frame(&"recovery", _state_remaining, effective_duration)

func _select_attack_frame(animation_name: StringName, remaining: float, effective_duration: float) -> void:
	if _body_sprite.sprite_frames == null or not _body_sprite.sprite_frames.has_animation(animation_name):
		return
	var frames := _body_sprite.sprite_frames.get_frame_count(animation_name)
	if frames <= 0:
		return
	if _body_sprite.animation != animation_name:
		_body_sprite.animation = animation_name
	_body_sprite.pause()
	var normalized_progress := clampf(1.0 - remaining / maxf(effective_duration, 0.0001), 0.0, 1.0)
	var total_weight := 0.0
	for frame_index in frames:
		total_weight += _body_sprite.sprite_frames.get_frame_duration(animation_name, frame_index)
	var elapsed := normalized_progress * total_weight
	var cumulative := 0.0
	var selected_frame := frames - 1
	for frame_index in frames:
		cumulative += _body_sprite.sprite_frames.get_frame_duration(animation_name, frame_index)
		if elapsed < cumulative:
			selected_frame = frame_index
			break
	_body_sprite.pause()
	_body_sprite.frame = selected_frame

func _update_attack_area_visual() -> void:
	match _attack_state:
		AttackState.WINDUP:
			var windup_progress := 1.0 - clampf(
				_state_remaining / maxf(windup_duration, 0.05),
				0.0,
				1.0
			)
			_set_attack_visual(&"windup", 0.58 + windup_progress * 0.34)
		AttackState.ACTIVE:
			_set_attack_visual(&"active", 1.0)
		AttackState.RECOVERY:
			var fade_alpha := 0.0
			if recovery_fade_duration > 0.0:
				fade_alpha = _recovery_fade_remaining / recovery_fade_duration
			_set_attack_visual(&"recovery", clampf(fade_alpha, 0.0, 1.0))
		_:
			_set_attack_visual(&"hidden", 0.0)

func _set_attack_visual(phase: StringName, alpha: float) -> void:
	if not is_instance_valid(_attack_visual) or not _attack_visual.has_method("set_area_state"):
		return
	_attack_visual.call(
		"set_area_state",
		_effective_attack_radius(),
		attack_center_offset if _attack_state == AttackState.IDLE else _locked_attack_center - global_position,
		phase,
		alpha
	)

func _update_hit_flash(delta: float) -> void:
	if _flash_remaining <= 0.0:
		return
	_flash_remaining = maxf(_flash_remaining - maxf(delta, 0.0), 0.0)
	_update_visuals()

func _update_visuals() -> void:
	if not is_instance_valid(_body_sprite) or not is_instance_valid(_health_fill):
		return
	match _locomotion_state:
		LocomotionState.PREPARE:
			_select_jump_animation(&"jump_crouch")
			_body_sprite.flip_h = _jump_facing_direction < 0
		LocomotionState.FLIGHT, LocomotionState.FALLING:
			if _jump_impulse_remaining > 0.0:
				_select_jump_animation(&"jump_impulse")
			elif velocity.y < -80.0:
				_select_jump_animation(&"jump_ascent")
			elif absf(velocity.y) <= 80.0:
				_select_jump_animation(&"jump_apex")
			else:
				_select_jump_animation(&"jump_fall")
			_body_sprite.flip_h = _jump_facing_direction < 0
		LocomotionState.LANDING:
			_select_jump_animation(&"jump_land")
			_body_sprite.flip_h = _jump_facing_direction < 0
		LocomotionState.GROUND:
			if _attack_state == AttackState.IDLE:
				var locomotion_clip: StringName = &"walk" if _patrol_moving else &"idle"
				if _body_sprite.animation != locomotion_clip:
					_body_sprite.play(locomotion_clip)
			_body_sprite.flip_h = _patrol_direction < 0
	var hit_flash := _body_sprite.material as ShaderMaterial
	if hit_flash != null:
		hit_flash.set_shader_parameter("flash_amount", 1.0 if _flash_remaining > 0.0 else 0.0)
	var health_ratio := clampf(float(health) / float(maxi(max_health, 1)), 0.0, 1.0)
	_health_fill.scale.x = health_ratio

func _select_jump_animation(animation_name: StringName) -> void:
	if _body_sprite.sprite_frames == null or not _body_sprite.sprite_frames.has_animation(animation_name):
		return
	if _body_sprite.animation != animation_name:
		_body_sprite.animation = animation_name
	_body_sprite.pause()
	_body_sprite.frame = 0
func _cancel_attack() -> void:
	if is_instance_valid(_attack_dust) and _attack_dust.has_method("clear"):
		_attack_dust.call("clear")
	_set_attack_visual(&"hidden", 0.0)
	_attack_state = AttackState.IDLE
	_state_remaining = 0.0
	_recovery_fade_remaining = 0.0
	_shockwave_tail_remaining = 0.0
	_locked_player = null
	_locked_attack_center = Vector2.ZERO
	_damaged_player_ids.clear()
	_update_visuals()

func _is_inside_current_scene() -> bool:
	return _node_is_inside_current_scene(self)

func _node_is_inside_current_scene(node: Node) -> bool:
	if not is_instance_valid(node):
		return false
	var scene_root := get_tree().current_scene
	return is_instance_valid(scene_root) and (scene_root == node or scene_root.is_ancestor_of(node))

func _patrol_preconditions_hold() -> bool:
	if not _patrol_enabled:
		return false
	if _defeated_emitted or health <= 0 or _placement_pending or not _is_inside_current_scene():
		return false
	if _bound_support == null or not is_instance_valid(_bound_support) \
		or not _bound_support.is_inside_tree() or not _is_compatible_support(_bound_support):
		_disable_patrol_with_warning("o suporte vinculado deixou de ser válido")
		return false
	return true

func _advance_patrol(delta: float) -> void:
	velocity = Vector2.ZERO
	if _patrol_pause_remaining > 0.0:
		_patrol_pause_remaining = maxf(_patrol_pause_remaining - delta, 0.0)
		_patrol_moving = false
		_update_visuals()
		return
	var distance_to_bound := _patrol_max_x - global_position.x if _patrol_direction > 0 else global_position.x - _patrol_min_x
	if distance_to_bound <= MOVEMENT_EPSILON:
		_turn_patrol_around()
		return
	var requested_dx := float(_patrol_direction) * minf(PATROL_SPEED * delta, distance_to_bound)
	if absf(requested_dx) < MOVEMENT_EPSILON:
		_patrol_moving = false
		_update_visuals()
		return

	var collision: KinematicCollision2D = move_and_collide(Vector2(requested_dx, 0.0), true, 0.0)
	var travel_x := requested_dx
	if collision != null:
		travel_x = collision.get_travel().x
	if signf(travel_x) != signf(requested_dx):
		travel_x = 0.0
	travel_x = signf(requested_dx) * minf(absf(travel_x), absf(requested_dx))
	var candidate_x := global_position.x + travel_x
	var clamped_x := clampf(candidate_x, _patrol_min_x, _patrol_max_x)
	var safe_dx := clamped_x - global_position.x
	var postcondition_valid := (
		candidate_x >= _patrol_min_x - MOVEMENT_EPSILON
		and candidate_x <= _patrol_max_x + MOVEMENT_EPSILON
		and signf(safe_dx) == signf(requested_dx)
		and absf(safe_dx) <= absf(requested_dx) + MOVEMENT_EPSILON
	)
	if not postcondition_valid:
		safe_dx = 0.0
	if absf(safe_dx) >= MOVEMENT_EPSILON:
		global_position.x += safe_dx
	var hit_bound := absf(_patrol_max_x - global_position.x) <= MOVEMENT_EPSILON if _patrol_direction > 0 else absf(global_position.x - _patrol_min_x) <= MOVEMENT_EPSILON
	var blocked := collision != null or absf(safe_dx - requested_dx) >= MOVEMENT_EPSILON
	if hit_bound or blocked:
		_turn_patrol_around()
		return
	_patrol_moving = absf(safe_dx) >= MOVEMENT_EPSILON
	_update_visuals()

func _bind_patrol_support(support: SolidPlatform, minimum_x: float, maximum_x: float, enable_patrol: bool) -> void:
	var previous_support := _bound_support
	var valid_support := support != null and is_instance_valid(support) and _is_compatible_support(support)
	if previous_support != null and is_instance_valid(previous_support) \
		and (not valid_support or previous_support != support):
		remove_collision_exception_with(previous_support)
	_patrol_enabled = false
	_patrol_moving = false
	if not valid_support:
		_bound_support = null
		_warn_patrol_once("suporte ausente ou incompatível; o alvo permanecerá estacionário")
		return
	_bound_support = support
	_patrol_min_x = minf(minimum_x, maximum_x)
	_patrol_max_x = maxf(minimum_x, maximum_x)
	# Reafirma a exceção após pousos, inclusive quando voltamos ao mesmo suporte.
	add_collision_exception_with(support)
	if enable_patrol and _patrol_max_x - _patrol_min_x >= MIN_PATROL_DISTANCE:
		_patrol_enabled = true
		_patrol_direction = _jump_facing_direction if _locomotion_state == LocomotionState.FLIGHT else 1
		_patrol_pause_remaining = 0.0
		_body_sprite.flip_h = _patrol_direction < 0
	else:
		_warn_patrol_once("nenhum percurso livre e útil de 8 px; o alvo permanecerá estacionário")
func _is_compatible_support(support: SolidPlatform) -> bool:
	if not is_instance_valid(support) or not support.is_inside_tree():
		return false
	if absf(support.global_rotation) > 0.001 or not support.global_scale.is_equal_approx(Vector2.ONE):
		return false
	var shape_count := 0
	for child in support.get_children():
		var collision := child as CollisionShape2D
		if collision == null:
			continue
		shape_count += 1
		if collision.disabled or not collision.shape is RectangleShape2D:
			return false
		if collision.position.length() > 0.01 or absf(collision.rotation) > 0.001 or not collision.scale.is_equal_approx(Vector2.ONE):
			return false
	return shape_count == 1

func _place_on_platform() -> void:
	if not _placement_pending:
		return
	var body := _get_collision_rect()
	var needed_width := body.size.x + edge_margin * 2.0

	# Prefere plataformas livres; o Floor só entra se não sobrar nenhuma,
	# e plataformas já ocupadas por outro inimigo ficam por último.
	var free: Array[SolidPlatform] = []
	var floors: Array[SolidPlatform] = []
	var taken: Array[SolidPlatform] = []
	for node in get_tree().get_nodes_in_group(&"solid_platform"):
		var candidate := node as SolidPlatform
		if candidate == null or candidate.size.x < needed_width or not _is_compatible_support(candidate):
			continue
		if candidate.name == &"Floor":
			floors.append(candidate)
		elif candidate.has_meta(CLAIMED_META):
			taken.append(candidate)
		else:
			free.append(candidate)

	var pool: Array[SolidPlatform] = free
	if pool.is_empty():
		pool = floors
	if pool.is_empty():
		pool = taken
	if pool.is_empty():
		_placement_pending = false
		_disable_patrol_with_warning("nenhuma plataforma compatível; o alvo permanecerá estacionário")
		return

	var rng := RandomNumberGenerator.new()
	if SolidPlatform.room_seed == 0:
		rng.randomize()
	else:
		rng.seed = hash([SolidPlatform.room_seed, get_index(), "training_target"])

	var platform := pool[rng.randi_range(0, pool.size() - 1)]
	platform.set_meta(CLAIMED_META, true)
	var support_data := _get_support_rect(platform)
	var support_rect: Rect2 = support_data.get("rect", Rect2())
	var patrol_bounds := _patrol_bounds_for_support(support_rect, body, edge_margin)
	var minimum_x := patrol_bounds.x
	var maximum_x := patrol_bounds.y
	var platform_top := support_rect.position.y
	var body_global_y := platform_top - body.end.y
	var sweep := _make_body_sweep(minimum_x, maximum_x, body, body_global_y)
	var has_useful_route := maximum_x - minimum_x >= MIN_PATROL_DISTANCE
	var sweep_is_clear := has_useful_route and _is_world_rect_clear(sweep, platform)
	if sweep_is_clear:
		global_position = Vector2(rng.randf_range(minimum_x, maximum_x), body_global_y)
		_bind_patrol_support(platform, minimum_x, maximum_x, true)
	else:
		var stationary := _find_first_safe_stationary_x(minimum_x, maximum_x, body, body_global_y, platform)
		if bool(stationary.get("valid", false)):
			global_position = Vector2(float(stationary.get("x", minimum_x)), body_global_y)
			_bind_patrol_support(platform, minimum_x, maximum_x, false)
			_warn_patrol_once("sweep bloqueado ou curto; usando posição estacionária segura")
		else:
			_bind_patrol_support(null, global_position.x, global_position.x, false)
			_warn_patrol_once("nenhum spawn seguro; mantendo a posição original")
	_placement_pending = false
	_update_visuals()

func plan_jump(source: SolidPlatform, preferred_direction: int) -> Dictionary:
	if not _owner_is_ready() or not _support_is_valid(source) or not _standing_on(source):
		return {}
	var source_rect := _support_rect(source)
	var start_position := _owner.global_position
	var body_left := start_position.x + _body_rect.position.x
	var body_right := start_position.x + _body_rect.end.x
	if body_left < source_rect.position.x - 1.0 or body_right > source_rect.end.x + 1.0:
		return {}

	var candidates: Array[Dictionary] = []
	for node in _owner.get_tree().get_nodes_in_group(&"solid_platform"):
		var support := node as SolidPlatform
		if support == null or support == source or support.name == &"Floor":
			continue

			continue
		var rect := _support_rect(support)

			continue

		var rise := source_rect.position.y - rect.position.y

		var maximum_x := rect.end.x - EDGE_MARGIN - _body_rect.end.x
		if maximum_x < minimum_x:
			continue

		var rank := (0.0 if preferred else 10000.0) + absf(rect.get_center().x - start_position.x)
		candidates.append({"support": support, "rect": rect, "min_x": minimum_x, "max_x": maximum_x, "rank": rank})

		return float(a["rank"]) < float(b["rank"])
	)

		var support := candidate["support"] as SolidPlatform
		var rect: Rect2 = candidate["rect"]
		var xs := _candidate_landing_xs(
			start_position.x,
			float(candidate["min_x"]),
			float(candidate["max_x"])

func _source_clear_landing_xs(
	source_rect: Rect2,
	minimum_x: float,
	maximum_x: float,
	preferred_x: float
) -> Array[float]:
	var candidates := _candidate_landing_xs(preferred_x, minimum_x, maximum_x)
	for offset in SOURCE_CLEARANCE_OFFSETS:
		var left_clear_x := source_rect.position.x - _body_rect.end.x - float(offset)
		var right_clear_x := source_rect.end.x - _body_rect.position.x + float(offset)
		if left_clear_x >= minimum_x and left_clear_x <= maximum_x:
			candidates.append(left_clear_x)
		if right_clear_x >= minimum_x and right_clear_x <= maximum_x:
			candidates.append(right_clear_x)
	var result: Array[float] = []
	for landing_x in candidates:
		var body_left := landing_x + _body_rect.position.x
		var body_right := landing_x + _body_rect.end.x
		if body_right > source_rect.position.x + 0.01 and body_left < source_rect.end.x - 0.01:
			continue
		var unique := true
		for existing in result:
			if absf(existing - landing_x) <= POSITION_EPSILON:
				unique = false
				break
		if unique:
			result.append(landing_x)
	result.sort_custom(func(a: float, b: float) -> bool:
		return absf(a - preferred_x) < absf(b - preferred_x)
	)
	return result

func _build_route(
	source: SolidPlatform,
	source_rect: Rect2,
	destination: SolidPlatform,
	destination_rect: Rect2,
	start_position: Vector2,
	landing_x: float
) -> Dictionary:
	if not _support_is_valid(source) or not _support_is_valid(destination):
		return {}
	var landing_min_x := destination_rect.position.x + EDGE_MARGIN - _body_rect.position.x
	var landing_max_x := destination_rect.end.x - EDGE_MARGIN - _body_rect.end.x
	if landing_x < landing_min_x - POSITION_EPSILON or landing_x > landing_max_x + POSITION_EPSILON:
		return {}
	var dx := landing_x - start_position.x
	if absf(dx) > MAX_HORIZONTAL:
		return {}
	var rise := source_rect.position.y - destination_rect.position.y
	if rise > MAX_RISE or rise < -MAX_DROP:
		return {}

	var apex_height := minf(maxf(MIN_APEX, rise + MIN_APEX_CLEARANCE), MAX_APEX)
	var launch_vy := -sqrt(2.0 * GRAVITY * apex_height)
	var landing_y := destination_rect.position.y - 0.5 - _body_rect.end.y
	var dy := landing_y - start_position.y
	var discriminant := launch_vy * launch_vy + 2.0 * GRAVITY * dy
	if discriminant < 0.0:
		return {}
	var final_vy := sqrt(discriminant)
	var flight_time := (-launch_vy + final_vy) / GRAVITY
	if flight_time <= 0.0 or flight_time > MAX_FLIGHT_TIME:
		return {}
	if apex_height + 0.01 < rise + MIN_APEX_CLEARANCE:
		return {}
	var launch_vx := dx / flight_time
	if absf(launch_vx) > MAX_HORIZONTAL_SPEED:
		return {}
	if not _trajectory_is_clear(
		start_position,
		launch_vx,
		launch_vy,
		flight_time,
		source_rect.position.y,
		source,
		destination
	):
		return {}
	return {
		"source": source,
		"destination": destination,
		"start_position": start_position,
		"landing_position": Vector2(landing_x, landing_y),
		"landing_x": landing_x,
		"launch_velocity": Vector2(launch_vx, launch_vy),
		"gravity": GRAVITY,
		"flight_time": flight_time
	}

func _trajectory_is_clear(
	start_position: Vector2,
	launch_vx: float,
	launch_vy: float,
	flight_time: float,
	source_top: float,
	source: SolidPlatform,
	_destination: SolidPlatform
) -> bool:
	var final_vy := launch_vy + GRAVITY * flight_time
	var maximum_speed := maxf(
		Vector2(launch_vx, launch_vy).length(),
		Vector2(launch_vx, final_vy).length()
	)
	var steps := maxi(1, int(ceil(maximum_speed * flight_time / MAX_SEGMENT_LENGTH)))
	var sample_times: Array[float] = [0.0]
	for step_index in range(1, steps):
		sample_times.append(flight_time * float(step_index) / float(steps))
	if is_instance_valid(source) and launch_vy < 0.0:
		var start_bottom := start_position.y + _body_rect.end.y
		var clearance := start_bottom - (source_top - EXIT_CLEARANCE)
		var discriminant := launch_vy * launch_vy - 2.0 * GRAVITY * clearance
		if clearance > 0.0 and discriminant >= 0.0:
			var clear_time := (-launch_vy - sqrt(discriminant)) / GRAVITY
			if clear_time > 0.000001 and clear_time < flight_time - 0.000001:
				sample_times.append(clear_time)
	sample_times.append(flight_time)
	sample_times.sort()

	var unique_times: Array[float] = []
	for t in sample_times:
		if unique_times.is_empty() or t - unique_times.back() > 0.000001:
			unique_times.append(t)
	var space := _owner.get_world_2d().direct_space_state
	var previous_center := start_position + _body_rect.get_center()
	var previous_source_exception := is_instance_valid(source) and launch_vy < 0.0 \
		and start_position.y + _body_rect.end.y > source_top - EXIT_CLEARANCE
	for step_index in range(unique_times.size()):
		var t: float = unique_times[step_index]
		var node_position := start_position + Vector2(launch_vx * t, launch_vy * t + 0.5 * GRAVITY * t * t)
		var center := node_position + _body_rect.get_center()
		var vy := launch_vy + GRAVITY * t
		var bottom := node_position.y + _body_rect.end.y
		var allow_source_exit := (
			bottom > source_top - EXIT_CLEARANCE
			and vy < 0.0
			and is_instance_valid(source)
		)
		if _overlaps_at(space, center, source, allow_source_exit):
			return false
		if step_index > 0:
			_query.transform = Transform2D(0.0, previous_center)
			_query.motion = center - previous_center
			_query.exclude = _exclusions(source, previous_source_exception)
			var cast := space.cast_motion(_query)
			if cast.size() > 0 and cast[0] < 0.9999:
				return false
		previous_center = center
		previous_source_exception = allow_source_exit
	return true

const EDGE_HORIZONTAL_SCALE: float = 360.0
const EDGE_VERTICAL_SCALE: float = 200.0
const PLAYER_SUPPORT_TOLERANCE: float = 5.0
const MAX_PLAYER_FALL_LOOKAHEAD: float = 320.0

func choose_player_support(
	player: PhysicsBody2D,
	supports: Array[SolidPlatform],
	jump_solver: Object,
	previous_support: SolidPlatform
) -> SolidPlatform:
	if not is_instance_valid(player):
		return null
	var collision_shape := player.get_node_or_null("CollisionShape2D") as CollisionShape2D
	var player_rect := Rect2(player.global_position, Vector2.ZERO)
	var feet_y := player.global_position.y
	if collision_shape != null and not collision_shape.disabled and collision_shape.shape != null:
		var local_rect := collision_shape.shape.get_rect()
		player_rect = Rect2(collision_shape.global_position + local_rect.position, local_rect.size)
		feet_y = player_rect.end.y
	var character := player as CharacterBody2D
	var grounded := character != null and character.is_on_floor()
	var best_ground: SolidPlatform
	var best_ground_delta: float = INF
	var best_below: SolidPlatform
	var best_below_delta: float = INF
	for support in supports:
		if not bool(jump_solver.call("is_support_compatible", support)):
			continue
		var rect: Rect2 = jump_solver.call("support_rect", support)
		if player_rect.end.x < rect.position.x or player_rect.position.x > rect.end.x:
			continue
		var vertical_delta := rect.position.y - feet_y
		if absf(vertical_delta) <= PLAYER_SUPPORT_TOLERANCE:
			if absf(vertical_delta) < best_ground_delta:
				best_ground = support
				best_ground_delta = absf(vertical_delta)
		elif vertical_delta >= -PLAYER_SUPPORT_TOLERANCE and vertical_delta <= MAX_PLAYER_FALL_LOOKAHEAD:
			if vertical_delta < best_below_delta:
				best_below = support
				best_below_delta = vertical_delta
	if grounded and best_ground != null:
		return best_ground
	if is_instance_valid(previous_support) and bool(jump_solver.call("is_support_compatible", previous_support)):
		return previous_support
	if best_ground != null:
		return best_ground
	return best_below

func find_path(
	source: SolidPlatform,
	destination: SolidPlatform,
	supports: Array[SolidPlatform],
	jump_solver: Object,
	blocked_edges: Dictionary
) -> Array[SolidPlatform]:
	if source == destination:
		return [source]
	if not is_instance_valid(source) or not is_instance_valid(destination):
		return []
	var costs: Dictionary = {source.get_instance_id(): 0.0}
	var previous: Dictionary = {}
	var supports_by_id: Dictionary = {}
	var visited: Dictionary = {}
	var open: Array[SolidPlatform] = [source]
	supports_by_id[source.get_instance_id()] = source
	supports_by_id[destination.get_instance_id()] = destination
	for support in supports:
		supports_by_id[support.get_instance_id()] = support
	while not open.is_empty():
		var current_index: int = -1
		var current: SolidPlatform
		var current_cost: float = INF
		for index in range(open.size()):
			var candidate: SolidPlatform = open[index]
			var candidate_cost: float = float(costs.get(candidate.get_instance_id(), INF))
			if candidate_cost < current_cost:
				current_index = index
				current = candidate
				current_cost = candidate_cost
		if current_index < 0 or current == null:
			break
		open.remove_at(current_index)
		var current_id: int = current.get_instance_id()
		if visited.has(current_id):
			continue
		visited[current_id] = true
		if current == destination:
			break
		for neighbor in supports:
			if neighbor == current or not bool(jump_solver.call("can_reach_support", current, neighbor)):
				continue
			var neighbor_id: int = neighbor.get_instance_id()
			if visited.has(neighbor_id):
				continue
			if float(blocked_edges.get(edge_key(current, neighbor), 0.0)) > 0.0:
				continue
			var source_rect: Rect2 = jump_solver.call("support_rect", current)
			var destination_rect: Rect2 = jump_solver.call("support_rect", neighbor)
			var horizontal_cost := absf(source_rect.get_center().x - destination_rect.get_center().x) / EDGE_HORIZONTAL_SCALE
			var vertical_cost := absf(source_rect.position.y - destination_rect.position.y) / EDGE_VERTICAL_SCALE
			var edge_cost := 1.0 + horizontal_cost * 0.2 + vertical_cost * 0.25
			var next_cost: float = current_cost + edge_cost
			if next_cost >= float(costs.get(neighbor_id, INF)):
				continue
			costs[neighbor_id] = next_cost
			previous[neighbor_id] = current_id
			supports_by_id[neighbor_id] = neighbor
			if not open.has(neighbor):
				open.append(neighbor)
	var cursor_id: int = destination.get_instance_id()
	if not previous.has(cursor_id):
		return []
	var path: Array[SolidPlatform] = [destination]
	while cursor_id != source.get_instance_id():
		if not previous.has(cursor_id):
			return []
		cursor_id = int(previous[cursor_id])
		var step: SolidPlatform = supports_by_id.get(cursor_id) as SolidPlatform
		if step == null:
			return []
		path.append(step)
	path.reverse()
	return path

func edge_key(source: SolidPlatform, destination: SolidPlatform) -> String:
	if not is_instance_valid(source) or not is_instance_valid(destination):
		return ""
	return "%d>%d" % [source.get_instance_id(), destination.get_instance_id()]

const MAX_PLATFORM_COUNT := 5

const TOP_LIMIT := 376

static func generate_layout(platform_count: int, requested_seed: int, spawn_x: float) -> Dictionary:
	var rng := RandomNumberGenerator.new()
	if requested_seed == 0:
		rng.randomize()
	else:
		rng.seed = requested_seed
	var seed_used: int = rng.seed
	if platform_count < MIN_PLATFORM_COUNT or platform_count > MAX_PLATFORM_COUNT:
		return _invalid_result(seed_used, "quantidade atual de plataformas fora do intervalo suportado (2..5)")

	var minimum_width: int = MIN_RANDOM_WIDTH if platform_count == 3 else MIN_FALLBACK_WIDTH
	var minimum_footprint := platform_count * minimum_width + (platform_count - 1) * GAP_MIN
	var spawn_left_min := ceili((spawn_x + 38.0) / 8.0) * 8
	var first_left_min := maxi(FIRST_LEFT_MIN, spawn_left_min)
	var first_left_max := mini(
		FIRST_LEFT_MAX,
		floori(float(RIGHT_LIMIT - minimum_footprint) / 8.0) * 8
	)
	var entries: Array[Dictionary] = []
	if first_left_min <= first_left_max:
		var first_left := rng.randi_range(first_left_min / 8, first_left_max / 8) * 8
		var slack_units := floori(float(RIGHT_LIMIT - first_left - minimum_footprint) / 8.0)
		var width_capacity_units := int((MAX_PLATFORM_WIDTH - minimum_width) / 8)
		var gap_capacity_units := int((GAP_MAX - GAP_MIN) / 8)
		var total_capacity_units := platform_count * width_capacity_units \
			+ (platform_count - 1) * gap_capacity_units
		# Reserva uma unidade de 8 px quando a folga permite.
		var budget_limit := mini(slack_units, total_capacity_units)
		if budget_limit > 0:
			budget_limit -= 1
		var extra_units := rng.randi_range(0, budget_limit)
		var horizontal_budget := _distribute_horizontal_budget(
			platform_count,
			minimum_width,
			extra_units,
			rng
		)
		var widths: Array[int] = horizontal_budget.get("widths", [])
		var gaps: Array[int] = horizontal_budget.get("gaps", [])
		var rises := _distribute_vertical_rise(platform_count, rng)
		var left := float(first_left)
		var previous_top := FIRST_TOP
		for index in platform_count:
			var top := FIRST_TOP if index == 0 else previous_top - rises[index - 1]
			previous_top = top
			var rect := Rect2(Vector2(left, top), Vector2(widths[index], PLATFORM_THICKNESS))
			entries.append(_entry_from_rect(rect))
			left = rect.end.x
			if index < platform_count - 1:
				left += gaps[index]

	if _entries_are_safe(entries, spawn_x):
		return _success_result(entries, seed_used, false)

	var fallback_entries := _safe_fallback(platform_count)
	if not _entries_are_safe(fallback_entries, spawn_x):
		return _invalid_result(seed_used, "fallback fixo não satisfaz os limites de sala")
	return _success_result(fallback_entries, seed_used, true)

static func _distribute_horizontal_budget(
	platform_count: int,
	minimum_width: int,
	extra_units: int,
	rng: RandomNumberGenerator
) -> Dictionary:
	var widths: Array[int] = []
	var gaps: Array[int] = []
	for index in platform_count:
		widths.append(minimum_width)
		if index < platform_count - 1:
			gaps.append(GAP_MIN)

	var slot_capacities: Array[int] = []
	var slot_is_gap: Array[bool] = []
	var slot_target_indices: Array[int] = []
	var width_capacity_units := int((MAX_PLATFORM_WIDTH - minimum_width) / 8)
	var gap_capacity_units := int((GAP_MAX - GAP_MIN) / 8)
	for index in platform_count:
		slot_capacities.append(width_capacity_units)
		slot_is_gap.append(false)
		slot_target_indices.append(index)
		if index < platform_count - 1:
			slot_capacities.append(gap_capacity_units)
			slot_is_gap.append(true)
			slot_target_indices.append(index)

	var slot_order: Array[int] = []
	for slot_index in slot_capacities.size():
		slot_order.append(slot_index)
	for index in range(slot_order.size() - 1, 0, -1):
		var swap_index := rng.randi_range(0, index)
		var previous_index: int = slot_order[index]
		slot_order[index] = slot_order[swap_index]
		slot_order[swap_index] = previous_index

	var suffix_capacity: Array[int] = []
	suffix_capacity.resize(slot_order.size() + 1)
	suffix_capacity[slot_order.size()] = 0
	for offset in slot_order.size():
		var position := slot_order.size() - 1 - offset
		suffix_capacity[position] = suffix_capacity[position + 1] + slot_capacities[slot_order[position]]

	var remaining_units := extra_units
	for position in slot_order.size():
		var slot_index: int = slot_order[position]
		var minimum_allocation := maxi(0, remaining_units - suffix_capacity[position + 1])
		var maximum_allocation := mini(slot_capacities[slot_index], remaining_units)
		var allocation := rng.randi_range(minimum_allocation, maximum_allocation)
		if slot_is_gap[slot_index]:
			gaps[slot_target_indices[slot_index]] += allocation * 8
		else:
			widths[slot_target_indices[slot_index]] += allocation * 8
		remaining_units -= allocation

	return {"widths": widths, "gaps": gaps}

static func _distribute_vertical_rise(
	platform_count: int,
	rng: RandomNumberGenerator
) -> Array[int]:
	var edge_count := platform_count - 1
	var remaining_units := rng.randi_range(0, MAX_RISE_PER_STEP / 8)
	var rises: Array[int] = []
	for index in edge_count:
		var remaining_edges := edge_count - index - 1
		var minimum_units := maxi(0, remaining_units - (MAX_RISE_PER_STEP / 8) * remaining_edges)
		var maximum_units := mini(MAX_RISE_PER_STEP / 8, remaining_units)
		var rise_units := rng.randi_range(minimum_units, maximum_units)
		rises.append(rise_units * 8)
		remaining_units -= rise_units
	return rises

static func _entries_are_safe(entries: Array[Dictionary], spawn_x: float) -> bool:
	if entries.size() < MIN_PLATFORM_COUNT:
		return false
	var previous: Rect2
	for index in entries.size():
		var rect: Rect2 = entries[index].get("rect", Rect2())
		if rect.size.y != PLATFORM_THICKNESS or rect.size.x < MIN_FALLBACK_WIDTH or rect.size.x > MAX_PLATFORM_WIDTH:
			return false
		if rect.position.x < 64.0 or rect.end.x > RIGHT_LIMIT or fposmod(rect.position.x, 8.0) > 0.01 or fposmod(rect.size.x, 8.0) > 0.01:
			return false
		if rect.position.y < TOP_LIMIT or rect.position.y > FIRST_TOP or rect.end.y > 464.0:
			return false
		if 560.0 - rect.end.y < 96.0:
			return false
		if index == 0:
			if absf(rect.position.y - FIRST_TOP) > 0.01:
				return false
			if rect.position.x < FIRST_LEFT_MIN or rect.position.x > FIRST_LEFT_MAX or rect.position.x - (spawn_x + 14.0) < 24.0:
				return false
		else:
			var gap := rect.position.x - previous.end.x
			var rise := previous.position.y - rect.position.y
			if gap < GAP_MIN or gap > GAP_MAX or rise < 0.0 or rise > MAX_RISE_PER_STEP:
				return false
			if rect.intersects(previous):
				return false
		previous = rect
	return true

func _enter_tree() -> void:
	if authored_layout_enabled:
		prepare_authored_layout()

func prepare_authored_layout() -> bool:
	if not authored_layout_enabled:
		return true
	if _authored_layout_prepared:
		return _authored_layout_valid

	_authored_layout_prepared = true
	_authored_layout_valid = false
	_authored_layout_error = ""
	_authored_target_position = Vector2.ZERO
	_authored_target_node = null
	_authored_target_support = null
	_authored_target_support_path = NodePath("")
	_authored_target_patrol_enabled = false
	_authored_target_patrol_min_x = 0.0
	_authored_target_patrol_max_x = 0.0
	_authored_world_obstacles.clear()
	_prepared_platform_rects.clear()

	var platform_set_result: Dictionary = _detect_authored_platform_set()
	if not bool(platform_set_result.get("valid", false)):
		return _fail_authored_layout(String(platform_set_result.get("error", "fixture de plataformas inválida")))
	var authored_platform_names: Array[StringName] = platform_set_result.get("names", [])
	var generation: Dictionary = RoomLayoutGenerator.generate_layout(
		authored_platform_names.size(),
		room_layout_seed,
		_get_player_spawn_x_for_layout()
	)
	_prepared_room_layout_seed = int(generation.get("seed", 0))
	if not bool(generation.get("valid", false)):
		return _fail_authored_layout(String(generation.get("error", "gerador de layout recusou a sala")))
	var generated_platforms: Array[Dictionary] = generation.get("platforms", [])
	if generated_platforms.size() != authored_platform_names.size():
		return _fail_authored_layout("gerador retornou quantidade diferente de plataformas")
	var target := get_node_or_null("TrainingTarget") as TrainingTarget
	if target == null or not target.authored_support_enabled:
		return _fail_authored_layout("TrainingTarget não está no modo de suporte autoral")
	var primary_index := int(generation.get("primary_index", -1))
	var fallback_index := int(generation.get("fallback_index", -1))
	if primary_index < 0 or primary_index >= authored_platform_names.size() or fallback_index < 0 or fallback_index >= authored_platform_names.size():
		return _fail_authored_layout("gerador não definiu suportes primário e fallback")
	target.fixed_support_path = NodePath("../" + String(authored_platform_names[primary_index]))
	target.fallback_support_path = NodePath("../" + String(authored_platform_names[fallback_index]))
	for platform_index in authored_platform_names.size():
		var platform_name: StringName = authored_platform_names[platform_index]
		var platform := get_node_or_null(NodePath(String(platform_name))) as SolidPlatform
		if platform == null:
			return _fail_authored_layout("plataforma ausente ou de tipo incorreto: %s" % String(platform_name))
		var generated: Dictionary = generated_platforms[platform_index]
		var generated_size: Vector2 = generated.get("size", Vector2.ZERO)
		var generated_position: Vector2 = generated.get("position", Vector2.ZERO)
		platform.randomize_position = false
		platform.randomize_width = false
		platform.size = generated_size
		platform.position = generated_position
		var platform_result: Dictionary = _validate_authored_platform(
			platform,
			generated_position,
			generated_size
		)
		if not bool(platform_result.get("valid", false)):
			return _fail_authored_layout(String(platform_result.get("error", "plataforma inválida")))
		_prepared_platform_rects[platform_name] = platform_result.get("rect", Rect2())

	if not _validate_generated_route(authored_platform_names):
		return false
	if not _validate_authored_room_boundaries():
		return false
	if not _collect_authored_world_obstacles():
		return false
	if not _validate_authored_spawns_and_patrol():
		return false
	if not _validate_authored_target(_prepared_platform_rects):
		return false

	_authored_layout_valid = true
	return true

func is_authored_layout_valid() -> bool:
	return not authored_layout_enabled or (_authored_layout_prepared and _authored_layout_valid)

func get_authored_layout_error() -> String:
	return _authored_layout_error

func get_prepared_authored_target_position(target: Node) -> Variant:
	if not authored_layout_enabled or not _authored_layout_prepared or not _authored_layout_valid or target != _authored_target_node:
		return null
	return _authored_target_position

func get_prepared_authored_target_binding(target: Node) -> Dictionary:
	if not authored_layout_enabled or not _authored_layout_prepared or not _authored_layout_valid or target != _authored_target_node:
		return {}
	return {
		"position": _authored_target_position,
		"support": _authored_target_support,
		"support_path": _authored_target_support_path,
		"patrol_enabled": _authored_target_patrol_enabled,
		"patrol_min_x": _authored_target_patrol_min_x,
		"patrol_max_x": _authored_target_patrol_max_x,
		"edge_margin": AUTHORED_EDGE_MARGIN,
	}

func get_prepared_room_layout_seed() -> int:
	return _prepared_room_layout_seed

func _fail_authored_layout(message: String) -> bool:
	_authored_layout_valid = false
	_authored_layout_error = message
	push_error("Main: layout autoral inválido — " + message)
	return false

func _detect_authored_platform_set() -> Dictionary:
	var present_names: Array[StringName] = []
	for child in get_children():
		var child_name: StringName = StringName(child.name)
		if child_name == &"Floor":
			continue
		var platform := child as SolidPlatform
		var known_platform: bool = KNOWN_AUTHORED_PLATFORM_NAMES.has(child_name)
		if not known_platform:
			if platform != null:
				return {"valid": false, "error": "plataforma autoral desconhecida: %s" % String(child_name)}
			continue
		if platform == null:
			return {"valid": false, "error": "nó de plataforma tem tipo incorreto: %s" % String(child_name)}
		present_names.append(child_name)

	if present_names.size() < RoomLayoutGenerator.MIN_PLATFORM_COUNT \
		or present_names.size() > RoomLayoutGenerator.MAX_PLATFORM_COUNT:
		return {"valid": false, "error": "quantidade de plataformas fora do intervalo suportado (2..5)"}
	return {"valid": true, "names": present_names}

func _get_player_spawn_x_for_layout() -> float:
	var spawn_marker := get_node_or_null(player_spawn_marker_path) as Marker2D
	return spawn_marker.position.x if spawn_marker != null else FALLBACK_PLAYER_SPAWN.x

func _validate_generated_route(platform_names: Array[StringName]) -> bool:
	var previous_rect := Rect2()
	for index in platform_names.size():
		var platform_name: StringName = platform_names[index]
		if not _prepared_platform_rects.has(platform_name):
			return _fail_authored_layout("retângulo físico ausente para %s" % String(platform_name))
		var rect: Rect2 = _prepared_platform_rects[platform_name]
		if rect.size.y != 24.0 or rect.size.x < 104.0 or rect.size.x > 216.0:
			return _fail_authored_layout("dimensão física inválida em %s" % String(platform_name))
		if rect.position.x < 64.0 or rect.end.x > 980.0 or fposmod(rect.position.x, 8.0) > 0.01 or fposmod(rect.size.x, 8.0) > 0.01:
			return _fail_authored_layout("%s ultrapassa a faixa de geração" % String(platform_name))
		if rect.position.y < 376.0 or rect.position.y > 440.0:
			return _fail_authored_layout("%s está fora dos limites verticais" % String(platform_name))
		if rect.end.y > 464.0 or 560.0 - rect.end.y < 96.0:
			return _fail_authored_layout("%s reduz a folga livre do piso abaixo de 96 px" % String(platform_name))
		if index == 0:
			if absf(rect.position.y - 440.0) > 0.01:
				return _fail_authored_layout("a primeira plataforma deve ter topo y=440")
			var spawn_x := _get_player_spawn_x_for_layout()
			if rect.position.x < 208.0 or rect.position.x > 256.0 or rect.position.x - (spawn_x + 14.0) < 24.0:
				return _fail_authored_layout("a primeira plataforma invade a margem do spawn")
		else:
			var gap := rect.position.x - previous_rect.end.x
			var rise := previous_rect.position.y - rect.position.y
			if gap < 56.0 or gap > 96.0 or rise < 0.0 or rise > 64.0:
				return _fail_authored_layout("vão ou desnível inválido entre plataformas")
			if _positive_overlap_area(rect, previous_rect) > AUTHORED_OVERLAP_AREA_LIMIT:
				return _fail_authored_layout("plataformas adjacentes se sobrepõem")
		previous_rect = rect
	return true

func _validate_authored_room_boundaries() -> bool:
	var floor_platform := get_node_or_null("Floor") as SolidPlatform
	if floor_platform == null:
		return _fail_authored_layout("Floor ausente ou de tipo incorreto")
	if floor_platform.position.distance_to(Vector2(576.0, 604.0)) > 0.01:
		return _fail_authored_layout("Floor fora da posição 576×604")
	if floor_platform.size.distance_to(Vector2(1152.0, 88.0)) > 0.01:
		return _fail_authored_layout("Floor deve medir 1152×88")
	if absf(floor_platform.rotation) > 0.001 or not floor_platform.scale.is_equal_approx(Vector2.ONE):
		return _fail_authored_layout("Floor tem transform incompatível")
	var floor_collision := floor_platform.get_node_or_null("CollisionShape2D") as CollisionShape2D
	if floor_collision == null or floor_collision.disabled:
		return _fail_authored_layout("Floor sem CollisionShape2D ativo")
	var floor_rectangle := floor_collision.shape as RectangleShape2D
	if floor_rectangle == null or floor_rectangle.size.distance_to(Vector2(1152.0, 88.0)) > 0.01:
		return _fail_authored_layout("shape físico de Floor deve medir 1152×88")
	if floor_collision.position.length() > 0.01 or absf(floor_collision.rotation) > 0.001 or not floor_collision.scale.is_equal_approx(Vector2.ONE):
		return _fail_authored_layout("shape físico de Floor está deslocado")

	var world_collision := get_node_or_null("WorldCollision") as StaticBody2D
	if world_collision == null:
		return _fail_authored_layout("WorldCollision ausente ou de tipo incorreto")
	if world_collision.position.length() > 0.01 or absf(world_collision.rotation) > 0.001 or not world_collision.scale.is_equal_approx(Vector2.ONE):
		return _fail_authored_layout("WorldCollision tem transform incompatível")

	var boundary_names: Array[StringName] = [
		&"LeftWall", &"RightWallUpper", &"RightWallLower", &"ExitGate", &"RightOuterGuard", &"Ceiling",
	]
	var boundary_positions: Array[Vector2] = [
		Vector2(32.0, 324.0), Vector2(1120.0, 216.0), Vector2(1120.0, 604.0),
		Vector2(1120.0, 496.0), Vector2(1156.0, 324.0), Vector2(576.0, -16.0),
	]
	var boundary_sizes: Array[Vector2] = [
		Vector2(64.0, 648.0), Vector2(64.0, 432.0), Vector2(64.0, 88.0),
		Vector2(64.0, 128.0), Vector2(8.0, 648.0), Vector2(1152.0, 32.0),
	]
	for boundary_index in boundary_names.size():
		var boundary := world_collision.get_node_or_null(NodePath(String(boundary_names[boundary_index]))) as CollisionShape2D
		if boundary == null:
			return _fail_authored_layout("shape de limite ausente: %s" % String(boundary_names[boundary_index]))
		if boundary_names[boundary_index] != &"ExitGate" and boundary.disabled:
			return _fail_authored_layout("shape de limite desativado: %s" % String(boundary_names[boundary_index]))
		if boundary.position.distance_to(boundary_positions[boundary_index]) > 0.01:
			return _fail_authored_layout("%s fora da posição autoral" % String(boundary_names[boundary_index]))
		var rectangle := boundary.shape as RectangleShape2D
		if rectangle == null or rectangle.size.distance_to(boundary_sizes[boundary_index]) > 0.01:
			return _fail_authored_layout("%s tem shape físico incompatível" % String(boundary_names[boundary_index]))
		if absf(boundary.rotation) > 0.001 or not boundary.scale.is_equal_approx(Vector2.ONE):
			return _fail_authored_layout("%s tem transform incompatível" % String(boundary_names[boundary_index]))

	var ceiling_band := world_collision.get_node_or_null("CeilingBand") as CollisionShape2D
	if ceiling_band == null or ceiling_band.disabled:
		return _fail_authored_layout("faixa física visível do teto ausente ou desativada")
	if ceiling_band.position.distance_to(Vector2(576.0, 32.0)) > 0.01:
		return _fail_authored_layout("CeilingBand fora da faixa y=0..64")
	var ceiling_rectangle := ceiling_band.shape as RectangleShape2D
	if ceiling_rectangle == null or ceiling_rectangle.size.distance_to(Vector2(1152.0, 64.0)) > 0.01:
		return _fail_authored_layout("CeilingBand deve medir 1152×64")
	if absf(ceiling_band.rotation) > 0.001 or not ceiling_band.scale.is_equal_approx(Vector2.ONE):
		return _fail_authored_layout("CeilingBand tem transform incompatível")
	return true

func _validate_authored_platform(
	platform: SolidPlatform,
	expected_position: Vector2,
	expected_size: Vector2
) -> Dictionary:
	if platform.position.distance_to(expected_position) > 0.01:
		return {"valid": false, "error": "%s fora da posição autoral" % platform.name}
	if platform.size.distance_to(expected_size) > 0.01:
		return {"valid": false, "error": "%s fora do tamanho autoral" % platform.name}
	if platform.randomize_position or platform.randomize_width:
		return {"valid": false, "error": "%s ainda permite aleatoriedade" % platform.name}
	if absf(platform.rotation) > 0.001 or not platform.scale.is_equal_approx(Vector2.ONE):
		return {"valid": false, "error": "%s tem transform autoral incompatível" % platform.name}

	var collision := platform.get_node_or_null("CollisionShape2D") as CollisionShape2D
	if collision == null or collision.disabled:
		return {"valid": false, "error": "%s sem CollisionShape2D ativo" % platform.name}
	var rectangle := collision.shape as RectangleShape2D
	if rectangle == null or rectangle.size.distance_to(expected_size) > 0.01:
		return {"valid": false, "error": "%s tem shape diferente do tamanho autoral" % platform.name}
	if collision.position.length() > 0.01 or absf(collision.rotation) > 0.001 or not collision.scale.is_equal_approx(Vector2.ONE):
		return {"valid": false, "error": "%s tem shape fora da origem local" % platform.name}
	return {"valid": true, "rect": Rect2(platform.position - expected_size * 0.5, expected_size)}

func _collect_authored_world_obstacles() -> bool:
	_authored_world_obstacles.clear()
	for child in get_children():
		var body := child as StaticBody2D
		if body == null or (body.collision_layer & WORLD_LAYER_BIT_MASK) == 0:
			continue
		var found_shape := false
		for shape_child in body.get_children():
			var collision := shape_child as CollisionShape2D
			if collision == null or collision.disabled:
				continue
			if collision.name == &"RightOuterGuard":
				continue
			if collision.shape == null:
				return _fail_authored_layout("shape ausente em %s" % body.name)
			found_shape = true
			_authored_world_obstacles.append({
				"owner": body,
				"rect": _collision_shape_aabb_in_main(collision),
			})
		if not found_shape:
			return _fail_authored_layout("nenhum shape ativo em World layer 1: %s" % body.name)
	return true

func _validate_authored_spawns_and_patrol() -> bool:
	var player_body := get_node_or_null("Player") as CollisionObject2D
	var player_spawn := get_node_or_null("PlayerSpawnMarker") as Marker2D
	if player_body == null or player_spawn == null:
		return _fail_authored_layout("spawn do jogador ausente")
	var player_bounds_result: Dictionary = _body_aabb(player_body, false)
	if not bool(player_bounds_result.get("valid", false)):
		return _fail_authored_layout("collider do jogador ausente")
	var player_bounds: Rect2 = player_bounds_result.get("rect", Rect2())
	var player_spawn_offset: Vector2 = player_spawn.position - player_body.position
	player_bounds = Rect2(player_bounds.position + player_spawn_offset, player_bounds.size)
	if not _validate_clear_of_world_layer(player_bounds, "spawn do jogador"):
		return false

	var worker_body := get_node_or_null("RivetWorker") as CollisionObject2D
	var patrol_area := get_node_or_null("WorkerPatrolArea") as Node2D
	if worker_body == null or patrol_area == null:
		return _fail_authored_layout("spawn ou patrulha do worker ausente")
	var worker_bounds_result: Dictionary = _body_aabb(worker_body, false)
	var worker_local_result: Dictionary = _body_aabb(worker_body, true)
	if not bool(worker_bounds_result.get("valid", false)) or not bool(worker_local_result.get("valid", false)):
		return _fail_authored_layout("collider do worker ausente")
	var worker_bounds: Rect2 = worker_bounds_result.get("rect", Rect2())
	var worker_local_bounds: Rect2 = worker_local_result.get("rect", Rect2())
	if not _validate_clear_of_world_layer(worker_bounds, "spawn do worker"):
		return false

	var left_limit := patrol_area.get_node_or_null("LeftLimit") as Marker2D
	var right_limit := patrol_area.get_node_or_null("RightLimit") as Marker2D
	if left_limit == null or right_limit == null:
		return _fail_authored_layout("limites de patrulha ausentes")
	var left_center: Vector2 = _local_transform_to_main(left_limit) * Vector2.ZERO
	var right_center: Vector2 = _local_transform_to_main(right_limit) * Vector2.ZERO
	var minimum_center_x := minf(left_center.x, right_center.x)
	var maximum_center_x := maxf(left_center.x, right_center.x)
	var sweep_bounds := Rect2(
		Vector2(minimum_center_x + worker_local_bounds.position.x, worker_body.position.y + worker_local_bounds.position.y),
		Vector2(maximum_center_x - minimum_center_x + worker_local_bounds.size.x, worker_local_bounds.size.y)
	)
	if sweep_bounds.position.x < 64.0 or sweep_bounds.end.x > 1088.0:
		return _fail_authored_layout("patrulha do worker cruza as faces internas das paredes")
	return _validate_clear_of_world_layer(sweep_bounds, "intervalo de patrulha do worker")

func _validate_authored_target(platform_rects: Dictionary) -> bool:
	var target := get_node_or_null("TrainingTarget") as TrainingTarget
	if target == null or not target.authored_support_enabled:
		return _fail_authored_layout("TrainingTarget não está no modo de suporte autoral")
	if target.collision_layer != 8:
		return _fail_authored_layout("TrainingTarget deve permanecer na collision layer 8")
	if target.collision_mask != 3:
		return _fail_authored_layout("TrainingTarget deve manter collision_mask 3 para World e Player")
	var player_body := get_node_or_null("Player") as CollisionObject2D
	if player_body == null or player_body.collision_mask != 9:
		return _fail_authored_layout("Player deve manter collision_mask 9 para World e TrainingTarget vivo")
	var target_local_result: Dictionary = _body_aabb(target, true)
	if not bool(target_local_result.get("valid", false)):
		return _fail_authored_layout("collider do TrainingTarget ausente")
	var target_local_bounds: Rect2 = target_local_result.get("rect", Rect2())
	if target.fixed_support_path.is_empty() or target.fallback_support_path.is_empty():
		return _fail_authored_layout("caminho de suporte autoral vazio")

	var support_paths: Array[NodePath] = [target.fixed_support_path, target.fallback_support_path]
	var first_safe_position: Dictionary = {}
	var selected_patrol: Dictionary = {}
	for support_path in support_paths:
		var support := _resolve_authored_support(target, support_path)
		if support == null or not platform_rects.has(support.name):
			continue
		var support_is_compatible := _is_target_support_compatible(support)
		var support_rect: Rect2 = platform_rects[support.name]
		if support_rect.size.x < target_local_bounds.size.x + AUTHORED_EDGE_MARGIN * 2.0:
			continue
		var target_origin := Vector2(
			support_rect.get_center().x - target_local_bounds.get_center().x,
			support_rect.position.y - target_local_bounds.end.y
		)
		var target_bounds: Rect2 = Rect2(target_local_bounds.position + target_origin, target_local_bounds.size)
		if target_bounds.position.x < support_rect.position.x + AUTHORED_EDGE_MARGIN - 0.01:
			continue
		if target_bounds.end.x > support_rect.end.x - AUTHORED_EDGE_MARGIN + 0.01:
			continue
		if absf(target_bounds.end.y - support_rect.position.y) > AUTHORED_CONTACT_TOLERANCE:
			continue
		if not _find_world_layer_overlap(target_bounds, support).is_empty():
			continue
		var minimum_center_x := support_rect.position.x + AUTHORED_EDGE_MARGIN - target_local_bounds.position.x
		var maximum_center_x := support_rect.end.x - AUTHORED_EDGE_MARGIN - target_local_bounds.end.x
		var route_is_useful := support_is_compatible and maximum_center_x - minimum_center_x >= 8.0

func _positive_overlap_area(first: Rect2, second: Rect2) -> float:
	var overlap_width := minf(first.end.x, second.end.x) - maxf(first.position.x, second.position.x)
	var overlap_height := minf(first.end.y, second.end.y) - maxf(first.position.y, second.position.y)
	if overlap_width <= 0.0 or overlap_height <= 0.0:
		return 0.0
	return overlap_width * overlap_height

func _body_aabb(body: CollisionObject2D, relative_to_body: bool) -> Dictionary:
	var found_shape := false
	var combined_bounds := Rect2()
	var body_transform := _local_transform_to_main(body)
	for shape_child in body.get_children():
		var collision := shape_child as CollisionShape2D
		if collision == null or collision.disabled or collision.shape == null:
			continue
		var shape_transform := _local_transform_to_main(collision)
		if relative_to_body:
			shape_transform = body_transform.affine_inverse() * shape_transform
		var shape_bounds := _transform_rect(collision.shape.get_rect(), shape_transform)
		if found_shape:
			combined_bounds = combined_bounds.merge(shape_bounds)
		else:
			combined_bounds = shape_bounds
			found_shape = true
	return {"valid": found_shape, "rect": combined_bounds}

func _collision_shape_aabb_in_main(collision: CollisionShape2D) -> Rect2:

func _local_transform_to_main(node: Node2D) -> Transform2D:
	var transforms: Array[Transform2D] = []
	var current: Node = node
	while current != self:
		var current_2d := current as Node2D
		if current_2d == null:
			return Transform2D.IDENTITY
		transforms.append(current_2d.transform)
		current = current.get_parent()
		if current == null:
			return Transform2D.IDENTITY
	var result := Transform2D.IDENTITY
	for transform_index in range(transforms.size() - 1, -1, -1):
		result = result * transforms[transform_index]
	return result

func _transform_rect(rect: Rect2, transform: Transform2D) -> Rect2:
	var point_a := transform * rect.position
	var point_b := transform * Vector2(rect.end.x, rect.position.y)
	var point_c := transform * rect.end
	var point_d := transform * Vector2(rect.position.x, rect.end.y)
	var minimum_x := minf(minf(point_a.x, point_b.x), minf(point_c.x, point_d.x))
	var minimum_y := minf(minf(point_a.y, point_b.y), minf(point_c.y, point_d.y))
	var maximum_x := maxf(maxf(point_a.x, point_b.x), maxf(point_c.x, point_d.x))
	var maximum_y := maxf(maxf(point_a.y, point_b.y), maxf(point_c.y, point_d.y))
	return Rect2(minimum_x, minimum_y, maximum_x - minimum_x, maximum_y - minimum_y)
