## Layout e copy
- Área de cada linha: 24 glifos (fonte 5×11, avanço 6 px); três linhas no máximo.
- Cada tópico começa com `- `; continuação do mesmo tópico começa com dois espaços.
- Quebrar o conteúdo em palavras com limite de 22 glifos, reservando duas colunas para o prefixo. Preservar palavras, pontuação e spans de cor através das quebras.
- Usar ciano `#52BAC7` no traço; cinza `#B6BEC3` nos rótulos, valores atuais, setas e condicionais; amarelo `#D2A75C` no benefício nominal ou valor novo.

func _wrap_topic(value: String) -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	if value.is_empty():
		return result
	var source_colors := _topic_colors(value)
	var content := ""
	var content_colors: Array[Color] = []
	var source_offset := 0
	for word: String in value.split(" ", false):
		if word.length() > TOPIC_CONTENT_WIDTH:
			return []
		var word_start := value.find(word, source_offset)
		if word_start < 0:
			return []
		var needs_space := not content.is_empty()
		if needs_space and content.length() + 1 + word.length() > TOPIC_CONTENT_WIDTH:
			result.append(_make_topic_line(content, content_colors, result.is_empty()))
			content = word
			content_colors = []
			for color_index in range(word_start, word_start + word.length()):
				content_colors.append(source_colors[color_index])
		else:
			if needs_space:
				content += " "
				content_colors.append(TOPIC_TEXT_COLOR)
			content += word
			for color_index in range(word_start, word_start + word.length()):
				content_colors.append(source_colors[color_index])
		source_offset = word_start + word.length()
	if not content.is_empty():
		result.append(_make_topic_line(content, content_colors, result.is_empty()))
	return result

func _make_topic_line(content: String, content_colors: Array[Color], first_line: bool) -> Dictionary:
	var prefix := "- " if first_line else "  "
	var colors: Array[Color] = [TOPIC_DASH_COLOR, TOPIC_TEXT_COLOR] if first_line else [TOPIC_TEXT_COLOR, TOPIC_TEXT_COLOR]
	colors.append_array(content_colors)
	return {"text": prefix + content, "colors": colors}

func _topic_colors(value: String) -> Array[Color]:
	var colors: Array[Color] = []
	for _index in range(value.length()):
		colors.append(TOPIC_TEXT_COLOR)
	if value.begins_with("Até ") or value.begins_with("Uma vez"):
		return colors
	var number_start := -1
	var number_end := -1
	var comparison_index := value.find(">")
	var search_start := 0
	if comparison_index >= 0:
		search_start = comparison_index + 1
		while search_start < value.length() and value.substr(search_start, 1) == " ":
			search_start += 1
		if search_start < value.length():
			if _is_sign(value.substr(search_start, 1)):
				number_start = search_start
				search_start += 1
			elif _is_digit(value.substr(search_start, 1)):
				number_start = search_start
	else:
		for index in range(value.length()):
			if _is_digit(value.substr(index, 1)):
				number_start = index
				if index > 0 and _is_sign(value.substr(index - 1, 1)):
					number_start = index - 1
				search_start = index
				break
	if number_start < 0:
		return colors
	number_end = search_start
	while number_end < value.length():
		var character := value.substr(number_end, 1)
		if _is_digit(character) or character == "," or character == ".":
			number_end += 1
		else:
			break
	if number_end == search_start:
		return colors
	if number_end < value.length() and value.substr(number_end, 1).to_lower() == "s":
		number_end += 1
	for index in range(number_start, number_end):
		colors[index] = TOPIC_VALUE_COLOR
	return colors

func _is_digit(value: String) -> bool:
	return value >= "0" and value <= "9"

func _is_sign(value: String) -> bool:
	return value == "+" or value == "-"

func _wrap_text(value: String, maximum_characters: int, maximum_lines: int) -> Array[String]:
	var result: Array[String] = []
	var current := ""
	for word: String in value.replace("\n", " ").split(" ", false):
		if word.length() > maximum_characters:
			return []
		var candidate := word if current.is_empty() else current + " " + word
		if candidate.length() <= maximum_characters:
			current = candidate
			continue
		result.append(current)
		current = word
		if result.size() >= maximum_lines:
			return []
	if not current.is_empty():
		result.append(current)
	if result.size() > maximum_lines:
		return []
	return result

static func load_font_bundle() -> Dictionary:
	if not ResourceLoader.exists(FONT_METADATA_PATH) or not ResourceLoader.exists(FONT_ATLAS_PATH):
		return {"valid": false, "reason": "fonte bitmap do HUD ausente"}
	var metadata_resource: Resource = ResourceLoader.load(FONT_METADATA_PATH)
	var atlas_resource: Resource = ResourceLoader.load(FONT_ATLAS_PATH)
	var metadata_json: JSON = metadata_resource as JSON
	var font_atlas: Texture2D = atlas_resource as Texture2D
	if metadata_json == null or font_atlas == null:
		return {"valid": false, "reason": "metadados ou atlas da fonte bitmap inválidos"}
	var raw_data: Variant = metadata_json.data
	if typeof(raw_data) != TYPE_DICTIONARY:
		return {"valid": false, "reason": "font.json não contém um objeto"}
	var font_data: Dictionary = raw_data as Dictionary
	var raw_glyphs: Variant = font_data.get("glyphs", null)
	if typeof(raw_glyphs) != TYPE_ARRAY:
		return {"valid": false, "reason": "font.json não contém a lista de glifos"}
	var glyphs: Dictionary = {}
	var glyph_list: Array = raw_glyphs as Array
	for glyph_variant: Variant in glyph_list:
		if typeof(glyph_variant) != TYPE_DICTIONARY:
			continue
		var glyph_data: Dictionary = glyph_variant as Dictionary
		var codepoint: int = int(glyph_data.get("codepoint", -1))
		var rect_variant: Variant = glyph_data.get("rect", null)
		if codepoint < 0 or typeof(rect_variant) != TYPE_DICTIONARY:
			continue
		var rect_data: Dictionary = rect_variant as Dictionary
		var glyph_rect: Rect2i = Rect2i(
			int(rect_data.get("x", -1)),
			int(rect_data.get("y", -1)),
			int(rect_data.get("w", 0)),
			int(rect_data.get("h", 0))
		)
		if glyph_rect.position.x < 0 or glyph_rect.position.y < 0 \
			or glyph_rect.size.x <= 0 or glyph_rect.size.y <= 0 \
			or glyph_rect.end.x > font_atlas.get_width() \
			or glyph_rect.end.y > font_atlas.get_height():
			continue
		glyphs[String.chr(codepoint)] = {
			"rect": glyph_rect,
			"advance": maxi(int(glyph_data.get("advance", 0)), 1),
		}
	var advance: int = int(font_data.get("advance", 0))
	var line_height: int = int(font_data.get("line_height", 0))
	if glyphs.is_empty() or advance <= 0 or line_height <= 0 or not glyphs.has(" "):
		return {"valid": false, "reason": "métricas incompletas da fonte bitmap"}
	return {"valid": true, "atlas": font_atlas, "glyphs": glyphs, "advance": advance, "line_height": line_height}

			var base_health: int = maxi(base_maximum_health, 1)
			var current_health: int = mini(base_health + maxi(current_rank, 0) * definition.maximum_health_per_rank, 100)
			var next_health: int = mini(base_health + maxi(next_rank, 0) * definition.maximum_health_per_rank, 100)
			var health_gain: int = maxi(next_health - current_health, 0)
			var recovered: int = acquisition_heal(upgrade_id, current_rank, next_rank)
			var health_topic: String
			if current_rank == 0:
				health_topic = "Vida máxima: +%d" % health_gain
			else:
				health_topic = "Vida máxima: %d > %d" % [current_health, next_health]
			return PackedStringArray([
				health_topic,
				"Cura na escolha: +%d vida" % recovered,
			])
	return definition.effect_topics.duplicate()
