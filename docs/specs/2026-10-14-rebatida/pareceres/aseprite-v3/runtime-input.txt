class_name PlayerVisual
extends Node2D

const IDLE_ANIMATION: StringName = &"idle"
const WALK_ANIMATION: StringName = &"walk"
const DASH_ANIMATION: StringName = &"dash"
const JUMP_RISE_ANIMATION: StringName = &"jump_rise"
const JUMP_APEX_ANIMATION: StringName = &"jump_apex"
const JUMP_FALL_ANIMATION: StringName = &"jump_fall"
const JUMP_LAND_ANIMATION: StringName = &"jump_land"
const HURT_ANIMATION: StringName = &"hurt"
const PARRY_BODY_ANIMATION: StringName = &"parry_body"
const JUMP_APEX_SPEED_LIMIT: float = 80.0
const WALK_REFERENCE_SPEED: float = 260.0
const WALK_SPEED_SCALE_MIN: float = 0.65
const WALK_SPEED_SCALE_MAX: float = 1.35
const PARRY_FRAME_DURATIONS: Array[float] = [0.030, 0.035, 0.025, 0.025, 0.040, 0.045]
const PARRY_TOTAL_DURATION: float = 0.200

@onready var dash_trail: Line2D = $DashTrail
@onready var character_sprite: AnimatedSprite2D = $AnimatedSprite2D

var _facing_direction: int = 1
var _is_walking: bool = false
var _is_dashing: bool = false
var _is_grounded: bool = true
var _vertical_speed: float = 0.0
var _has_ground_contact: bool = false
var _airborne_seen: bool = false
var _is_landing: bool = false
var _is_invulnerable: bool = false
var _is_hurt: bool = false
var _is_dead: bool = false
var _is_parrying: bool = false
var _parry_progress: float = 0.0
var _parry_facing: int = 1
var _dash_phase: float = 0.0
var _walk_speed_magnitude: float = 0.0

func _ready() -> void:
	character_sprite.animation_changed.connect(_apply_facing)
	character_sprite.animation_finished.connect(_on_character_animation_finished)
	_apply_facing()
	character_sprite.play(IDLE_ANIMATION)

func _process(delta: float) -> void:
	if _is_dead:
		return
	if _is_invulnerable:
		var damage_pulse := 0.5 + 0.5 * sin(Time.get_ticks_msec() * 0.035)
		if _is_hurt:
			modulate = Color(1.0, 0.88 + damage_pulse * 0.1, 0.84 + damage_pulse * 0.1, 1.0)
		else:
			modulate = Color(1.0, 0.56 + damage_pulse * 0.34, 0.5 + damage_pulse * 0.42, 0.45 + damage_pulse * 0.35)
	else:
		modulate = Color.WHITE
	if not _is_dashing:
		return

	_dash_phase = fmod(_dash_phase + delta * 16.0, TAU)
	_update_dash_trail_scale()

func set_facing_direction(direction: int) -> void:
	if direction == 0 or direction == _facing_direction:
		return
	_facing_direction = 1 if direction > 0 else -1
	_apply_facing()

func set_walking(value: bool, horizontal_speed: float = 0.0) -> void:
	_walk_speed_magnitude = absf(horizontal_speed)
	if value == _is_walking:
		_set_walk_speed_scale()
		return
	_is_walking = value
	_set_walk_speed_scale()
	_apply_animation_state()

func _set_walk_speed_scale() -> void:
	if _is_walking and _is_grounded and not _is_landing and not _is_dashing and not _is_hurt and not _is_dead and not _is_parrying:
		character_sprite.speed_scale = clampf(
			_walk_speed_magnitude / WALK_REFERENCE_SPEED,
			WALK_SPEED_SCALE_MIN,
			WALK_SPEED_SCALE_MAX
		)
	else:
		character_sprite.speed_scale = 1.0

func set_grounded(value: bool, vertical_speed: float = 0.0) -> void:
	_vertical_speed = vertical_speed
	if not value:
		if _has_ground_contact:
			_airborne_seen = true
		_is_grounded = false
		_is_landing = false
		_apply_animation_state()
		return

	var landed_after_airborne := not _is_grounded and _airborne_seen
	_is_grounded = true
	_has_ground_contact = true
	_airborne_seen = false
	if landed_after_airborne and not _is_dashing and not _is_dead and not _is_hurt:
		_is_landing = true
		character_sprite.speed_scale = 1.0
		character_sprite.play(JUMP_LAND_ANIMATION)
		return

	_apply_animation_state()

func reset_air_state() -> void:
	_is_grounded = true
	_vertical_speed = 0.0
	_has_ground_contact = false
	_airborne_seen = false
	_is_landing = false
	_apply_animation_state()

func set_dashing(value: bool) -> void:
	if value == _is_dashing:
		return
	if value and _is_dead:
		return
	_is_dashing = value
	dash_trail.visible = _is_dashing
	if _is_dashing:
		_is_landing = false
		_dash_phase = 0.0
		_restart_dash_animation()
	else:
		_dash_phase = 0.0
		_apply_animation_state()
	_update_dash_trail_scale()

func set_invulnerable(value: bool) -> void:
	_is_invulnerable = value

func set_parry_state(active: bool, progress: float, frozen_direction: Vector2) -> void:
	if active:
		if not _is_parrying:
			_parry_facing = (1 if frozen_direction.x > 0.0 else -1) if not is_zero_approx(frozen_direction.x) else _facing_direction
			_is_parrying = true
			_apply_facing()
		_parry_progress = clampf(progress, 0.0, 1.0)
		_apply_animation_state()
		return
	if not _is_parrying:
		return
	_is_parrying = false
	_parry_progress = 0.0
	_apply_facing()
	_apply_animation_state()

func play_hurt() -> void:
	if _is_dead:
		return
	_is_hurt = true
	_is_landing = false
	character_sprite.speed_scale = 1.0

	character_sprite.frame_progress = 0.0
	character_sprite.play(HURT_ANIMATION)

func set_dead(value: bool) -> void:
	if value:
		if _is_dead:
			return
		_is_dead = true
		_is_invulnerable = false
		_is_hurt = false
		_is_parrying = false

		_is_landing = false
		_airborne_seen = false

		set_dashing(false)
		character_sprite.speed_scale = 1.0
		character_sprite.stop()

		return

	_is_dead = false
	_is_invulnerable = false
	_is_hurt = false
	_is_parrying = false
	_parry_progress = 0.0
	_is_walking = false
	_is_grounded = true
	_vertical_speed = 0.0
	_has_ground_contact = false
	_airborne_seen = false

	character_sprite.frame = 0
	character_sprite.frame_progress = 0.0
	_apply_animation_state()
	modulate = Color.WHITE

func _apply_animation_state() -> void:
	if _is_dead:
		character_sprite.speed_scale = 1.0
		character_sprite.stop()
		return

	if _is_hurt:
		character_sprite.speed_scale = 1.0
		if character_sprite.animation != HURT_ANIMATION:
			character_sprite.play(HURT_ANIMATION)
		return

	if _is_dashing:
		character_sprite.speed_scale = 1.0
		if character_sprite.animation != DASH_ANIMATION:
			character_sprite.play(DASH_ANIMATION)
		return

	if not _is_grounded:
		character_sprite.speed_scale = 1.0
		var target_animation := JUMP_APEX_ANIMATION
		if _vertical_speed < -JUMP_APEX_SPEED_LIMIT:
			target_animation = JUMP_RISE_ANIMATION
		elif _vertical_speed > JUMP_APEX_SPEED_LIMIT:
			target_animation = JUMP_FALL_ANIMATION
		if character_sprite.animation != target_animation:
			character_sprite.play(target_animation)
		return

	if _is_landing:
		character_sprite.speed_scale = 1.0
		if character_sprite.animation != JUMP_LAND_ANIMATION or not character_sprite.is_playing():
			character_sprite.play(JUMP_LAND_ANIMATION)
		return

	if _is_parrying:
		character_sprite.speed_scale = 1.0
		if character_sprite.animation != PARRY_BODY_ANIMATION:
			character_sprite.stop()
			character_sprite.animation = PARRY_BODY_ANIMATION
		elif character_sprite.is_playing():
			character_sprite.stop()
		_sync_parry_frame()
		return

	var target_animation := WALK_ANIMATION if _is_walking else IDLE_ANIMATION
	_set_walk_speed_scale()
	if character_sprite.animation != target_animation or not character_sprite.is_playing():
		character_sprite.play(target_animation)

func _restart_dash_animation() -> void:
	character_sprite.speed_scale = 1.0
	if _is_hurt:
		return
	character_sprite.stop()
	character_sprite.animation = DASH_ANIMATION
	character_sprite.frame = 0
	character_sprite.frame_progress = 0.0
	character_sprite.play(DASH_ANIMATION)

func _on_character_animation_finished() -> void:
	if _is_hurt and character_sprite.animation == HURT_ANIMATION:
		_is_hurt = false
		_apply_animation_state()
		return
	if not _is_landing or not _is_grounded or character_sprite.animation != JUMP_LAND_ANIMATION:
		return
	_is_landing = false
	_apply_animation_state()

func _apply_facing() -> void:
	var native_direction := -1
	if character_sprite.animation == DASH_ANIMATION or character_sprite.animation == PARRY_BODY_ANIMATION:
		native_direction = 1
	var facing_direction := _parry_facing if _is_parrying and character_sprite.animation == PARRY_BODY_ANIMATION else _facing_direction
	character_sprite.flip_h = facing_direction != native_direction
	_update_dash_trail_scale()

func _sync_parry_frame() -> void:
	var elapsed := minf(_parry_progress * PARRY_TOTAL_DURATION, PARRY_TOTAL_DURATION - 0.000001)
	for frame_index in range(PARRY_FRAME_DURATIONS.size()):
		var frame_duration := PARRY_FRAME_DURATIONS[frame_index]
		if elapsed < frame_duration or frame_index == PARRY_FRAME_DURATIONS.size() - 1:
			character_sprite.set_frame_and_progress(
				frame_index,
				clampf(elapsed / frame_duration, 0.0, 1.0)
			)
			return
		elapsed -= frame_duration

func _update_dash_trail_scale() -> void:
	var pulse_scale := 1.0
	if _is_dashing:
		var dash_pulse := 0.5 + 0.5 * sin(_dash_phase)
		pulse_scale += dash_pulse * 0.2
	dash_trail.scale.x = float(_facing_direction) * pulse_scale

func _process(delta: float) -> void:
	var player := get_parent().get_parent() as Player
	if player == null or player.is_dead:
		visible = false
		_set_arm_mask(false)
		return
	visible = true
	scale.x = -1.0 if _body_sprite.flip_h else 1.0
	var world_direction := _attack_world_direction if _attack_kind != &"" else _aim_world_direction
	_attack_direction = _world_direction_to_local(world_direction)
	_advance_attack(delta)
	_ground_rig_active = _body_sprite.animation == &"idle" or _body_sprite.animation == &"walk"
	_set_arm_mask(_ground_rig_active)
	_update_pose()
	queue_redraw()

func _advance_attack(delta: float) -> void:
	if _attack_kind == &"":
		return
	if _attack_kind == &"parry":
		if _racket_controller == null or not _racket_controller.is_swing_active():
			clear_transient()
			return
		_attack_elapsed = _parry_duration * _racket_controller.get_swing_progress()
		return
	_attack_elapsed += delta
	var shot_elapsed := _attack_elapsed
	for index in SHOT_POSES:
		if shot_elapsed < POSE_DURATION[index]:
			_pose_index = index
			return
		shot_elapsed -= POSE_DURATION[index]
	clear_transient()

func _update_pose() -> void:
	var direction := _attack_direction.normalized()
	var shoulder := SHOULDER
	var head_angle_offset := 0.0
	if _attack_kind == &"parry":
		var parry_progress := clampf(_attack_elapsed / _parry_duration, 0.0, 1.0)
		var parry_segment := 0
		while parry_segment < PARRY_KEY_TIME.size() - 2 \
			and parry_progress >= PARRY_KEY_TIME[parry_segment + 1]:
			parry_segment += 1
		var segment_start := PARRY_KEY_TIME[parry_segment]
		var segment_end := PARRY_KEY_TIME[parry_segment + 1]
		var segment_progress := (parry_progress - segment_start) / (segment_end - segment_start)
		var eased_progress := _ease_parry_segment(parry_segment, segment_progress)
		var grip_offset := PARRY_GRIP_OFFSET[parry_segment].lerp(
			PARRY_GRIP_OFFSET[parry_segment + 1], eased_progress
		)
		if character_is_parry_body():
			head_angle_offset = _parry_body_head_angle(parry_progress)
		else:
			head_angle_offset = lerpf(
				PARRY_HEAD_ANGLE[parry_segment], PARRY_HEAD_ANGLE[parry_segment + 1], eased_progress
			)
		if _ground_rig_active:
			var side := direction.rotated(PI * 0.5)
			var target_grip := shoulder + direction * grip_offset.x + side * grip_offset.y
			_grip = _clamp_grip(target_grip, shoulder, direction)
			_elbow = _solve_elbow(shoulder, _grip, direction)
		else:
			_grip = _native_hand_anchor()
			_elbow = _grip
	elif _attack_kind == &"shot":
		head_angle_offset = SHOT_HEAD_ANGLE[_pose_index]
		if _ground_rig_active:
			var shot_racket_direction := direction.rotated(head_angle_offset)
			var contact := to_local(global_position + _shot_contact_offset)
			_grip = _clamp_grip(contact - shot_racket_direction * 18.0, shoulder, direction)
			_elbow = _solve_elbow(shoulder, _grip, direction)
		else:
			_grip = _native_hand_anchor()
			_elbow = _grip
	else:
		if _ground_rig_active:
			_grip = shoulder + direction * 24.0
			_elbow = _solve_elbow(shoulder, _grip, direction)
		else:
			_grip = _native_hand_anchor()
			_elbow = _grip
	var racket_direction := direction.rotated(head_angle_offset)
	_head_center = _grip + racket_direction * 18.0
	_head_direction = racket_direction
	_head.position = _head_center
	_head.rotation = _head_direction.angle()
	_head.visible = true

func _ease_parry_segment(segment: int, progress: float) -> float:
	var t := clampf(progress, 0.0, 1.0)
	match segment:
		0:
			return t * t * t
		1:
			return 1.0 - pow(1.0 - t, 3.0)
		2, 3:
			return t * t
		4, 6:
			return 1.0 - pow(1.0 - t, 3.0)
		_:
			return t * t * (3.0 - 2.0 * t)

func character_is_parry_body() -> bool:
	return _body_sprite.animation == &"parry_body"

func _parry_body_head_angle(progress: float) -> float:
	var elapsed := clampf(progress, 0.0, 1.0) * 0.200
	for pose_index in range(PARRY_BODY_DURATIONS.size()):
		var duration := PARRY_BODY_DURATIONS[pose_index]
		if elapsed < duration or pose_index == PARRY_BODY_DURATIONS.size() - 1:
			var transition := clampf(elapsed / duration, 0.0, 1.0)
			return lerp_angle(
				PARRY_BODY_HEAD_ANGLE[pose_index],
				PARRY_BODY_HEAD_ANGLE[pose_index + 1],
				transition
			)
		elapsed -= duration
	return 0.0

func _clamp_grip(target: Vector2, shoulder: Vector2, fallback_axis: Vector2) -> Vector2:
	var offset := target - shoulder
	var distance := offset.length()
	var axis := offset / distance if distance > 0.001 else fallback_axis.normalized()
	if axis.is_zero_approx():
		axis = Vector2.RIGHT
	if distance < MIN_ARM_REACH:
		return shoulder + axis * MIN_ARM_REACH
	if distance > MAX_ARM_REACH:
		return shoulder + axis * MAX_ARM_REACH
	return target

func _solve_elbow(shoulder: Vector2, grip: Vector2, preferred_direction: Vector2) -> Vector2:
	var offset := grip - shoulder
	var raw_distance := offset.length()
	var fallback_axis := preferred_direction.normalized()
	if fallback_axis.is_zero_approx():
		fallback_axis = Vector2.RIGHT
	var side_reference := fallback_axis.orthogonal()
	var axis := offset / raw_distance if raw_distance > 0.001 else fallback_axis
	if axis.is_zero_approx():
		axis = Vector2.RIGHT
	var distance := clampf(raw_distance, MIN_ARM_REACH, UPPER_ARM_LENGTH + FOREARM_LENGTH - 0.001)
	var along := (UPPER_ARM_LENGTH * UPPER_ARM_LENGTH - FOREARM_LENGTH * FOREARM_LENGTH + distance * distance) / (2.0 * distance)
	var perpendicular := sqrt(maxf(UPPER_ARM_LENGTH * UPPER_ARM_LENGTH - along * along, 0.0))
	var side := axis.orthogonal()
	if side.dot(side_reference) < -ARM_SIDE_EPSILON:
		side = -side
	return shoulder + axis * along + side * perpendicular
