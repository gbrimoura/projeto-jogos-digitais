func configure_rank(rank_before: int, rank_after: int, rank_max: int, reward_kind: StringName) -> bool:
	if reward_kind == &"recovery":
		_hide_prisms()
		return true
	if rank_before < 0 or rank_after <= rank_before or rank_max < rank_after or rank_max <= 0:
		return false
	_slot_count = _slots_for_maximum(rank_max)
	if _slot_count == 0:
		_hide_prisms()
		return true
	_owned_slots = clampi(rank_before, 0, _slot_count)
	_offered_slot = _owned_slots if _owned_slots < _slot_count else -1
	visible = true
	queue_redraw()
	return true

func _slots_for_maximum(rank_max: int) -> int:
	if rank_max == THREE_RANK_SLOTS:
		return THREE_RANK_SLOTS
	if rank_max == FIVE_RANK_SLOTS:
		return FIVE_RANK_SLOTS
	if rank_max == TEN_RANK_SLOTS:
		return TEN_RANK_SLOTS
	return 0

func _hide_prisms() -> void:
	_slot_count = 0
	_owned_slots = 0
	_offered_slot = -1
	visible = false
	queue_redraw()

func _draw() -> void:
	if _slot_count <= 0:
		return
	var total_width: int = _slot_count * PRISM_WIDTH + (_slot_count - 1) * PRISM_GAP
	var start_x: int = floori((size.x - float(total_width)) * 0.5)
	var start_y: int = floori((size.y - float(PRISM_HEIGHT)) * 0.5)
	for slot in range(_slot_count):
		var state: PrismState = PrismState.EMPTY
		if slot < _owned_slots:
			state = PrismState.OWNED
		elif slot == _offered_slot:
			state = PrismState.OFFERED
		_draw_prism(start_x + slot * (PRISM_WIDTH + PRISM_GAP), start_y, state)

func _draw_prism(x: int, y: int, state: PrismState) -> void:
	var outline := EMPTY_OUTLINE
	var fill := NAVY
	var highlight := EMPTY_HIGHLIGHT
	var shadow := EMPTY_SHADOW
	if state == PrismState.OWNED:
		outline = OWNED_OUTLINE
		fill = OWNED_FILL
		highlight = OWNED_HIGHLIGHT
		shadow = OWNED_SHADOW
	elif state == PrismState.OFFERED:
		outline = OFFERED_OUTLINE
		fill = OFFERED_FILL
		highlight = OFFERED_HIGHLIGHT
		shadow = OFFERED_SHADOW

	_pixel_rect(x + 2, y, 4, 1, outline)
	_pixel_rect(x + 1, y + 1, 6, 1, outline)
	_pixel_rect(x, y + 2, 1, 6, outline)
	_pixel_rect(x + 7, y + 2, 1, 6, outline)
	_pixel_rect(x + 1, y + 8, 6, 1, outline)
	_pixel_rect(x + 2, y + 9, 4, 1, outline)

	_pixel_rect(x + 2, y + 1, 4, 1, fill)
	_pixel_rect(x + 1, y + 2, 6, 6, fill)
	_pixel_rect(x + 2, y + 8, 4, 1, fill)
	_pixel_rect(x + 2, y + 1, 4, 1, highlight)
	_pixel_rect(x + 5, y + 2, 2, 3, highlight)
	_pixel_rect(x + 1, y + 2, 2, 3, shadow)

	_pixel_rect(x + 1, y + 5, 2, 3, fill)
	_pixel_rect(x + 5, y + 5, 2, 3, fill)

func _wrap_topic(value: String) -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	if value.is_empty():
		return result
	var source_colors := _topic_colors(value)
	var content := ""
	var content_colors: Array[Color] = []
	var source_offset := 0
	for word: String in value.split(" ", false):
		if word.length() > TOPIC_CONTENT_WIDTH:
			return []
		var word_start := value.find(word, source_offset)
		if word_start < 0:
			return []
		var needs_space := not content.is_empty()
		if needs_space and content.length() + 1 + word.length() > TOPIC_CONTENT_WIDTH:
			result.append(_make_topic_line(content, content_colors, result.is_empty()))
			content = word
			content_colors = []
			for color_index in range(word_start, word_start + word.length()):
				content_colors.append(source_colors[color_index])
		else:
			if needs_space:
				content += " "
				content_colors.append(TOPIC_TEXT_COLOR)
			content += word
			for color_index in range(word_start, word_start + word.length()):
				content_colors.append(source_colors[color_index])
		source_offset = word_start + word.length()
	if not content.is_empty():
		result.append(_make_topic_line(content, content_colors, result.is_empty()))
	return result

func _make_topic_line(content: String, content_colors: Array[Color], first_line: bool) -> Dictionary:
	var prefix := "- " if first_line else "  "
	var colors: Array[Color] = []
	colors.append(TOPIC_DASH_COLOR if first_line else TOPIC_TEXT_COLOR)
	colors.append(TOPIC_TEXT_COLOR)
	colors.append_array(content_colors)
	return {"text": prefix + content, "colors": colors}

func _topic_colors(value: String) -> Array[Color]:
	var colors: Array[Color] = []
	for _index in range(value.length()):
		colors.append(TOPIC_TEXT_COLOR)
	if value.begins_with("Até ") or value.begins_with("Uma vez"):
		return colors
	var number_start := -1
	var number_end := -1
	var comparison_index := value.find(">")
	var search_start := 0
	if comparison_index >= 0:
		search_start = comparison_index + 1
		while search_start < value.length() and value.substr(search_start, 1) == " ":
			search_start += 1
		if search_start < value.length():
			if _is_sign(value.substr(search_start, 1)):
				number_start = search_start
				search_start += 1
			elif _is_digit(value.substr(search_start, 1)):
				number_start = search_start
	else:
		for index in range(value.length()):
			if _is_digit(value.substr(index, 1)):
				number_start = index
				if index > 0 and _is_sign(value.substr(index - 1, 1)):
					number_start = index - 1
				search_start = index
				break
	if number_start < 0:
		return colors
	number_end = search_start
	while number_end < value.length():
		var character := value.substr(number_end, 1)
		if _is_digit(character) or character == "," or character == ".":
			number_end += 1
		else:
			break
	if number_end == search_start:
		return colors
	if number_end < value.length() and value.substr(number_end, 1).to_lower() == "s":
		number_end += 1
	for index in range(number_start, number_end):
		colors[index] = TOPIC_VALUE_COLOR
	return colors

func _is_digit(value: String) -> bool:
	return value >= "0" and value <= "9"

func _is_sign(value: String) -> bool:
	return value == "+" or value == "-"

func _wrap_text(value: String, maximum_characters: int, maximum_lines: int) -> Array[String]:
	var result: Array[String] = []
	var current := ""
	for word: String in value.replace("\n", " ").split(" ", false):
		if word.length() > maximum_characters:
			return []
		var candidate := word if current.is_empty() else current + " " + word
		if candidate.length() <= maximum_characters:
			current = candidate
			continue
		result.append(current)
		current = word
		if result.size() >= maximum_lines:
			return []
	if not current.is_empty():
		result.append(current)
	if result.size() > maximum_lines:
		return []
	return result

func _on_mouse_entered() -> void:
	_hovered = true
	_apply_frame_state()

func _on_mouse_exited() -> void:
	_hovered = false
	_apply_frame_state()

func _on_focus_entered() -> void:
	_apply_frame_state()

func _on_focus_exited() -> void:
	_apply_frame_state()

func _on_button_down() -> void:
	_pressed_visual = true
	_apply_frame_state()

func _on_button_up() -> void:
	_pressed_visual = false
	_apply_frame_state()

func _apply_frame_state() -> void:
	if _frame == null:
		return
	if _pressed_visual:
		_frame.texture = FRAME_PRESSED
	elif _hovered or has_focus():
		_frame.texture = FRAME_FOCUS
	else:
		_frame.texture = FRAME_NORMAL
