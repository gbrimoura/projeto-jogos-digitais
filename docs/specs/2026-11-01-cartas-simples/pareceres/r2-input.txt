  local function load_font()
    lock_palette()
    local raw = read(FONTJSON)
    assert(raw:find('"cell":{"w":5,"h":11}', 1, true), "font cell changed")
    assert(raw:find('"advance":6', 1, true), "font advance changed")
    local atlas = Image { fromFile = FONTPNG }
    assert(atlas.width == 84 and atlas.height == 105, "card font atlas dimensions changed")
    local glyphs = {}
    local pattern = '"character":"([^"]*)","codepoint":(%d+),"rect":{"x":(%d+),"y":(%d+),"w":(%d+),"h":(%d+)},"advance":(%d+)'
    for ch, _, x, y, w, h, advance in raw:gmatch(pattern) do
      glyphs[ch] = { x = tonumber(x), y = tonumber(y), w = tonumber(w), h = tonumber(h), advance = tonumber(advance) }
    end
    assert(glyphs.A and glyphs["-"] and glyphs[" "], "font glyphs missing")
    return atlas, glyphs
  end
  local function chars(value)
    local out, i = {}, 1
    while i <= #value do
      local b = value:byte(i)
      local n = (b < 128 and 1) or (b < 224 and 2) or (b < 240 and 3) or 4
      out[#out + 1] = value:sub(i, i + n - 1)
      i = i + n
    end
    return out
  end
  local function draw_text(im, x, y, value, color, atlas, glyphs)
    local cursor = x
    for _, ch in ipairs(chars(value)) do
      if ch == "+" then
        for xx = 1, 3 do pixel(im, cursor + xx, y + 5, color) end
        for yy = 3, 7 do pixel(im, cursor + 2, y + yy, color) end
        cursor = cursor + 6
      else
        local g = assert(glyphs[ch], "preview glyph missing: " .. ch)
        for yy = 0, g.h - 1 do
          for xx = 0, g.w - 1 do
            if pc.rgbaA(atlas:getPixel(g.x + xx, g.y + yy)) > 0 then
              pixel(im, cursor + xx, y + yy, color)
            end
          end
        end
        cursor = cursor + g.advance
      end
    end
  end
  local function center_text(im, center, y, value, color, atlas, glyphs)
    local width = 0
    for _, ch in ipairs(chars(value)) do width = width + (ch == "+" and 6 or assert(glyphs[ch], "preview glyph missing: " .. ch).advance) end
    draw_text(im, center - math.floor(width / 2), y, value, color, atlas, glyphs)
  end
  local function save_integer_scale(source, path, scale)
    local result = image(source.width * scale, source.height * scale)
    for y = 0, source.height - 1 do
      for x = 0, source.width - 1 do
        local value = source:getPixel(x, y)
        for dy = 0, scale - 1 do
          for dx = 0, scale - 1 do result:drawPixel(x * scale + dx, y * scale + dy, value) end
        end
      end
    end
    assert(result:saveAs(path), "preview export failed: " .. path)
  end

func configure(row: Dictionary, font_bundle: Dictionary) -> bool:
	if not bool(font_bundle.get("valid", false)):
		return false
	var definition: UpgradeDefinitionResource = row.get("definition") as UpgradeDefinitionResource
	if definition == null:
		return false
	var illustration: Texture2D = row.get("icon", definition.icon) as Texture2D
	if illustration == null:
		return false
	if not _configure_text(_title, String(row.get("title", definition.title)), font_bundle, 1):
		return false
	var summary_text := String(row.get("description", definition.description))
	if summary_text.is_empty():
		summary_text = definition.description
	var summary_lines := _wrap_text(summary_text, 24, 2)
	if summary_lines.size() != 1 and summary_lines.size() != 2:
		return false
	if not _configure_text(_summary, "\n".join(PackedStringArray(summary_lines)), font_bundle, 2):
		return false
	var topic_lines: Array[String] = []
	var topics_variant: Variant = row.get("topics", definition.effect_topics)
	if typeof(topics_variant) == TYPE_PACKED_STRING_ARRAY or typeof(topics_variant) == TYPE_ARRAY:
		for topic_variant: Variant in topics_variant:
			var wrapped_topic := _wrap_text(String(topic_variant), 24, 3)
			for line: String in wrapped_topic:
				topic_lines.append(line)
	if topic_lines.is_empty() or topic_lines.size() > _topic_lines.size():
		return false
	for index in range(_topic_lines.size()):
		var line := topic_lines[index] if index < topic_lines.size() else ""
		if not _configure_text(_topic_lines[index], line, font_bundle, 1):
			return false
	if not _configure_text(_badge, String(row.get("rank_label", "")), font_bundle, 1):
		return false
	_illustration.texture = illustration
	_illustration.show()
	return true

func _configure_text(control: HudBitmapText, value: String, font_bundle: Dictionary, lines: int) -> bool:
	if control == null or not control.configure_font(font_bundle):
		return false
	control.max_lines = lines
	control.set_text(value)
	return true

func _wrap_text(value: String, maximum_characters: int, maximum_lines: int) -> Array[String]:
	var result: Array[String] = []
	var current := ""
	for word: String in value.replace("\n", " ").split(" ", false):
		if word.length() > maximum_characters:
			return []
		var candidate := word if current.is_empty() else current + " " + word
		if candidate.length() <= maximum_characters:
			current = candidate
			continue
		result.append(current)
		current = word
		if result.size() >= maximum_lines:
			return []
	if not current.is_empty():
		result.append(current)
	if result.size() > maximum_lines:
		return []
	return result
