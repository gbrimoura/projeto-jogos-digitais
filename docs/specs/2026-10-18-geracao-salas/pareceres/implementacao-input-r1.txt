func prepare_authored_layout() -> bool:
	if not authored_layout_enabled:
		return true
	if _authored_layout_prepared:
		return _authored_layout_valid

	_authored_layout_prepared = true
	_authored_layout_valid = false
	_authored_layout_error = ""
	_authored_target_position = Vector2.ZERO
	_authored_target_node = null
	_authored_world_obstacles.clear()
	_prepared_platform_rects.clear()

	var platform_set_result: Dictionary = _detect_authored_platform_set()
	if not bool(platform_set_result.get("valid", false)):
		return _fail_authored_layout(String(platform_set_result.get("error", "fixture de plataformas inválida")))
	var authored_platform_names: Array[StringName] = platform_set_result.get("names", [])
	var generation: Dictionary = RoomLayoutGenerator.generate_layout(
		authored_platform_names.size(),
		room_layout_seed,
		_get_player_spawn_x_for_layout()
	)
	_prepared_room_layout_seed = int(generation.get("seed", 0))
	if not bool(generation.get("valid", false)):
		return _fail_authored_layout(String(generation.get("error", "gerador de layout recusou a sala")))
	var generated_platforms: Array[Dictionary] = generation.get("platforms", [])
	if generated_platforms.size() != authored_platform_names.size():
		return _fail_authored_layout("gerador retornou quantidade diferente de plataformas")
	var target := get_node_or_null("TrainingTarget") as TrainingTarget
	if target == null or not target.authored_support_enabled:
		return _fail_authored_layout("TrainingTarget não está no modo de suporte autoral")
	var primary_index := int(generation.get("primary_index", -1))
	var fallback_index := int(generation.get("fallback_index", -1))
	if primary_index < 0 or primary_index >= authored_platform_names.size() or fallback_index < 0 or fallback_index >= authored_platform_names.size():
		return _fail_authored_layout("gerador não definiu suportes primário e fallback")
	target.fixed_support_path = NodePath("../" + String(authored_platform_names[primary_index]))
	target.fallback_support_path = NodePath("../" + String(authored_platform_names[fallback_index]))
	for platform_index in authored_platform_names.size():
		var platform_name: StringName = authored_platform_names[platform_index]
		var platform := get_node_or_null(NodePath(String(platform_name))) as SolidPlatform
		if platform == null:
			return _fail_authored_layout("plataforma ausente ou de tipo incorreto: %s" % String(platform_name))
		var generated: Dictionary = generated_platforms[platform_index]
		var generated_size: Vector2 = generated.get("size", Vector2.ZERO)
		var generated_position: Vector2 = generated.get("position", Vector2.ZERO)
		platform.randomize_position = false
		platform.randomize_width = false
		platform.size = generated_size
		platform.position = generated_position
		var platform_result: Dictionary = _validate_authored_platform(
			platform,
			generated_position,
			generated_size
		)
		if not bool(platform_result.get("valid", false)):
			return _fail_authored_layout(String(platform_result.get("error", "plataforma inválida")))
		_prepared_platform_rects[platform_name] = platform_result.get("rect", Rect2())

	if not _validate_generated_route(authored_platform_names):
		return false
	if not _validate_authored_room_boundaries():

		return false
	if not _validate_authored_target(_prepared_platform_rects):

func _detect_authored_platform_set() -> Dictionary:
	var present_names: Array[StringName] = []
	for child in get_children():
		var child_name: StringName = StringName(child.name)
		if child_name == &"Floor":
			continue
		var platform := child as SolidPlatform
		var known_platform: bool = KNOWN_AUTHORED_PLATFORM_NAMES.has(child_name)
		if not known_platform:
			if platform != null:
				return {"valid": false, "error": "plataforma autoral desconhecida: %s" % String(child_name)}
			continue
		if platform == null:
			return {"valid": false, "error": "nó de plataforma tem tipo incorreto: %s" % String(child_name)}
		present_names.append(child_name)

	if present_names.size() < RoomLayoutGenerator.MIN_PLATFORM_COUNT \
		or present_names.size() > RoomLayoutGenerator.MAX_PLATFORM_COUNT:
		return {"valid": false, "error": "quantidade de plataformas fora do intervalo suportado (2..5)"}
	return {"valid": true, "names": present_names}

func _validate_generated_route(platform_names: Array[StringName]) -> bool:
	var previous_rect := Rect2()
	for index in platform_names.size():
		var platform_name: StringName = platform_names[index]
		if not _prepared_platform_rects.has(platform_name):
			return _fail_authored_layout("retângulo físico ausente para %s" % String(platform_name))
		var rect: Rect2 = _prepared_platform_rects[platform_name]
		if rect.size.y != 24.0 or rect.size.x < 104.0 or rect.size.x > 216.0:
			return _fail_authored_layout("dimensão física inválida em %s" % String(platform_name))
		if rect.position.x < 64.0 or rect.end.x > 980.0 or fposmod(rect.position.x, 8.0) > 0.01 or fposmod(rect.size.x, 8.0) > 0.01:
			return _fail_authored_layout("%s ultrapassa a faixa de geração" % String(platform_name))
		if rect.position.y < 376.0 or rect.position.y > 440.0:
			return _fail_authored_layout("%s está fora dos limites verticais" % String(platform_name))
		if rect.end.y > 464.0 or 560.0 - rect.end.y < 96.0:
			return _fail_authored_layout("%s reduz a folga livre do piso abaixo de 96 px" % String(platform_name))
		if index == 0:
			if absf(rect.position.y - 440.0) > 0.01:
				return _fail_authored_layout("a primeira plataforma deve ter topo y=440")
			var spawn_x := _get_player_spawn_x_for_layout()
			if rect.position.x < 208.0 or rect.position.x > 256.0 or rect.position.x - (spawn_x + 14.0) < 24.0:
				return _fail_authored_layout("a primeira plataforma invade a margem do spawn")
		else:
			var gap := rect.position.x - previous_rect.end.x
			var rise := previous_rect.position.y - rect.position.y
			if gap < 56.0 or gap > 96.0 or rise < 0.0 or rise > 64.0:
				return _fail_authored_layout("vão ou desnível inválido entre plataformas")
			if _positive_overlap_area(rect, previous_rect) > AUTHORED_OVERLAP_AREA_LIMIT:
				return _fail_authored_layout("plataformas adjacentes se sobrepõem")
		previous_rect = rect
	return true

func _validate_authored_room_boundaries() -> bool:
	var floor_platform := get_node_or_null("Floor") as SolidPlatform
	if floor_platform == null:
		return _fail_authored_layout("Floor ausente ou de tipo incorreto")
	if floor_platform.position.distance_to(Vector2(576.0, 604.0)) > 0.01:
		return _fail_authored_layout("Floor fora da posição 576×604")
	if floor_platform.size.distance_to(Vector2(1152.0, 88.0)) > 0.01:
		return _fail_authored_layout("Floor deve medir 1152×88")
	if absf(floor_platform.rotation) > 0.001 or not floor_platform.scale.is_equal_approx(Vector2.ONE):
		return _fail_authored_layout("Floor tem transform incompatível")
	var floor_collision := floor_platform.get_node_or_null("CollisionShape2D") as CollisionShape2D
	if floor_collision == null or floor_collision.disabled:
		return _fail_authored_layout("Floor sem CollisionShape2D ativo")
	var floor_rectangle := floor_collision.shape as RectangleShape2D
	if floor_rectangle == null or floor_rectangle.size.distance_to(Vector2(1152.0, 88.0)) > 0.01:
		return _fail_authored_layout("shape físico de Floor deve medir 1152×88")
	if floor_collision.position.length() > 0.01 or absf(floor_collision.rotation) > 0.001 or not floor_collision.scale.is_equal_approx(Vector2.ONE):
		return _fail_authored_layout("shape físico de Floor está deslocado")

	var world_collision := get_node_or_null("WorldCollision") as StaticBody2D

		return _fail_authored_layout("WorldCollision ausente ou de tipo incorreto")

		return _fail_authored_layout("WorldCollision tem transform incompatível")

func _validate_authored_platform(
	platform: SolidPlatform,
	expected_position: Vector2,
	expected_size: Vector2
) -> Dictionary:
	if platform.position.distance_to(expected_position) > 0.01:
		return {"valid": false, "error": "%s fora da posição autoral" % platform.name}
	if platform.size.distance_to(expected_size) > 0.01:
		return {"valid": false, "error": "%s fora do tamanho autoral" % platform.name}
	if platform.randomize_position or platform.randomize_width:
		return {"valid": false, "error": "%s ainda permite aleatoriedade" % platform.name}
	if absf(platform.rotation) > 0.001 or not platform.scale.is_equal_approx(Vector2.ONE):
		return {"valid": false, "error": "%s tem transform autoral incompatível" % platform.name}

	var collision := platform.get_node_or_null("CollisionShape2D") as CollisionShape2D
	if collision == null or collision.disabled:
		return {"valid": false, "error": "%s sem CollisionShape2D ativo" % platform.name}
	var rectangle := collision.shape as RectangleShape2D
	if rectangle == null or rectangle.size.distance_to(expected_size) > 0.01:
		return {"valid": false, "error": "%s tem shape diferente do tamanho autoral" % platform.name}
	if collision.position.length() > 0.01 or absf(collision.rotation) > 0.001 or not collision.scale.is_equal_approx(Vector2.ONE):
		return {"valid": false, "error": "%s tem shape fora da origem local" % platform.name}
	return {"valid": true, "rect": Rect2(platform.position - expected_size * 0.5, expected_size)}

func _collect_authored_world_obstacles() -> bool:
	_authored_world_obstacles.clear()
	for child in get_children():
		var body := child as StaticBody2D
		if body == null or (body.collision_layer & WORLD_LAYER_BIT_MASK) == 0:
			continue
		var found_shape := false
		for shape_child in body.get_children():
			var collision := shape_child as CollisionShape2D
			if collision == null or collision.disabled:
				continue
			if collision.shape == null:
				return _fail_authored_layout("shape ausente em %s" % body.name)
			found_shape = true
			_authored_world_obstacles.append({
				"owner": body,
				"rect": _collision_shape_aabb_in_main(collision),
			})
		if not found_shape:
			return _fail_authored_layout("nenhum shape ativo em World layer 1: %s" % body.name)
	return true

func _validate_authored_spawns_and_patrol() -> bool:
	var player_body := get_node_or_null("Player") as CollisionObject2D
	var player_spawn := get_node_or_null("PlayerSpawnMarker") as Marker2D
	if player_body == null or player_spawn == null:
		return _fail_authored_layout("spawn do jogador ausente")
	var player_bounds_result: Dictionary = _body_aabb(player_body, false)
	if not bool(player_bounds_result.get("valid", false)):
		return _fail_authored_layout("collider do jogador ausente")
	var player_bounds: Rect2 = player_bounds_result.get("rect", Rect2())
	var player_spawn_offset: Vector2 = player_spawn.position - player_body.position
	player_bounds = Rect2(player_bounds.position + player_spawn_offset, player_bounds.size)
	if not _validate_clear_of_world_layer(player_bounds, "spawn do jogador"):
		return false

	var worker_body := get_node_or_null("RivetWorker") as CollisionObject2D
	var patrol_area := get_node_or_null("WorkerPatrolArea") as Node2D
	if worker_body == null or patrol_area == null:
		return _fail_authored_layout("spawn ou patrulha do worker ausente")
	var worker_bounds_result: Dictionary = _body_aabb(worker_body, false)
	var worker_local_result: Dictionary = _body_aabb(worker_body, true)
	if not bool(worker_bounds_result.get("valid", false)) or not bool(worker_local_result.get("valid", false)):
		return _fail_authored_layout("collider do worker ausente")
	var worker_bounds: Rect2 = worker_bounds_result.get("rect", Rect2())
	var worker_local_bounds: Rect2 = worker_local_result.get("rect", Rect2())
	if not _validate_clear_of_world_layer(worker_bounds, "spawn do worker"):
		return false

	var left_limit := patrol_area.get_node_or_null("LeftLimit") as Marker2D
	var right_limit := patrol_area.get_node_or_null("RightLimit") as Marker2D
	if left_limit == null or right_limit == null:
		return _fail_authored_layout("limites de patrulha ausentes")
	var left_center: Vector2 = _local_transform_to_main(left_limit) * Vector2.ZERO
	var right_center: Vector2 = _local_transform_to_main(right_limit) * Vector2.ZERO
	var minimum_center_x := minf(left_center.x, right_center.x)
	var maximum_center_x := maxf(left_center.x, right_center.x)
	var sweep_bounds := Rect2(
		Vector2(minimum_center_x + worker_local_bounds.position.x, worker_body.position.y + worker_local_bounds.position.y),
		Vector2(maximum_center_x - minimum_center_x + worker_local_bounds.size.x, worker_local_bounds.size.y)
	)
	if sweep_bounds.position.x < 64.0 or sweep_bounds.end.x > 1088.0:
		return _fail_authored_layout("patrulha do worker cruza as faces internas das paredes")
	return _validate_clear_of_world_layer(sweep_bounds, "intervalo de patrulha do worker")

func _validate_authored_target(platform_rects: Dictionary) -> bool:
	var target := get_node_or_null("TrainingTarget") as TrainingTarget
	if target == null or not target.authored_support_enabled:
		return _fail_authored_layout("TrainingTarget não está no modo de suporte autoral")
	if target.collision_layer != 8:
		return _fail_authored_layout("TrainingTarget deve permanecer na collision layer 8")
	var player_body := get_node_or_null("Player") as CollisionObject2D
	if player_body == null or player_body.collision_mask != 9:
		return _fail_authored_layout("Player deve manter collision_mask 9 para World e TrainingTarget vivo")
	var target_local_result: Dictionary = _body_aabb(target, true)
	if not bool(target_local_result.get("valid", false)):
		return _fail_authored_layout("collider do TrainingTarget ausente")
	var target_local_bounds: Rect2 = target_local_result.get("rect", Rect2())
	if target.fixed_support_path.is_empty() or target.fallback_support_path.is_empty():
		return _fail_authored_layout("caminho de suporte autoral vazio")

	var support_paths: Array[NodePath] = [target.fixed_support_path, target.fallback_support_path]
	for support_path in support_paths:
		var support := _resolve_authored_support(target, support_path)
		if support == null or not platform_rects.has(support.name):
			continue
		var support_rect: Rect2 = platform_rects[support.name]
		if support_rect.size.x < target_local_bounds.size.x + AUTHORED_EDGE_MARGIN * 2.0:
			continue
		var target_origin := Vector2(
			support_rect.get_center().x - target_local_bounds.get_center().x,
			support_rect.position.y - target_local_bounds.end.y
		)
		var target_bounds: Rect2 = Rect2(target_local_bounds.position + target_origin, target_local_bounds.size)
		if target_bounds.position.x < support_rect.position.x + AUTHORED_EDGE_MARGIN - 0.01:
			continue
		if target_bounds.end.x > support_rect.end.x - AUTHORED_EDGE_MARGIN + 0.01:
			continue
		if absf(target_bounds.end.y - support_rect.position.y) > AUTHORED_CONTACT_TOLERANCE:
			continue
		if not _find_world_layer_overlap(target_bounds, support).is_empty():
			continue
		_authored_target_position = target_origin
		_authored_target_node = target
		return true

	return _fail_authored_layout("suportes primário e fallback do TrainingTarget são inválidos")

func _resolve_authored_support(target: TrainingTarget, path: NodePath) -> SolidPlatform:
	if target.get_parent() != self:
		return null
	var path_parts: PackedStringArray = String(path).split("/", false)
	if path_parts.size() != 2 or path_parts[0] != ".." or path_parts[1].is_empty():
		return null
	return get_node_or_null(NodePath(path_parts[1])) as SolidPlatform

func _validate_clear_of_world_layer(
	bounds: Rect2,
	label: String,
	allowed_support: CollisionObject2D = null
) -> bool:
	var blocker := _find_world_layer_overlap(bounds, allowed_support)
	if blocker.is_empty():
		return true
	return _fail_authored_layout("%s intersecta World layer 1 em %s" % [label, blocker])

func _find_world_layer_overlap(bounds: Rect2, allowed_support: CollisionObject2D = null) -> String:
	for obstacle_data: Dictionary in _authored_world_obstacles:
		var owner: CollisionObject2D = obstacle_data.get("owner") as CollisionObject2D
		if owner == null or owner == allowed_support:
			continue
		var obstacle_bounds: Rect2 = obstacle_data.get("rect", Rect2())
		if _positive_overlap_area(bounds, obstacle_bounds) > AUTHORED_OVERLAP_AREA_LIMIT:
			return String(owner.name)
	return ""

func _positive_overlap_area(first: Rect2, second: Rect2) -> float:
	var overlap_width := minf(first.end.x, second.end.x) - maxf(first.position.x, second.position.x)
	var overlap_height := minf(first.end.y, second.end.y) - maxf(first.position.y, second.position.y)
	if overlap_width <= 0.0 or overlap_height <= 0.0:
		return 0.0
	return overlap_width * overlap_height

func _body_aabb(body: CollisionObject2D, relative_to_body: bool) -> Dictionary:
	var found_shape := false
	var combined_bounds := Rect2()
	var body_transform := _local_transform_to_main(body)
	for shape_child in body.get_children():
		var collision := shape_child as CollisionShape2D
		if collision == null or collision.disabled or collision.shape == null:
			continue
		var shape_transform := _local_transform_to_main(collision)
		if relative_to_body:
			shape_transform = body_transform.affine_inverse() * shape_transform
		var shape_bounds := _transform_rect(collision.shape.get_rect(), shape_transform)
		if found_shape:
			combined_bounds = combined_bounds.merge(shape_bounds)
		else:
			combined_bounds = shape_bounds
			found_shape = true
	return {"valid": found_shape, "rect": combined_bounds}

func _collision_shape_aabb_in_main(collision: CollisionShape2D) -> Rect2:
	return _transform_rect(collision.shape.get_rect(), _local_transform_to_main(collision))

func _local_transform_to_main(node: Node2D) -> Transform2D:
	var transforms: Array[Transform2D] = []
	var current: Node = node
	while current != self:
		var current_2d := current as Node2D
		if current_2d == null:
			return Transform2D.IDENTITY
		transforms.append(current_2d.transform)
		current = current.get_parent()
		if current == null:
			return Transform2D.IDENTITY
	var result := Transform2D.IDENTITY
	for transform_index in range(transforms.size() - 1, -1, -1):
		result = result * transforms[transform_index]
	return result

func _transform_rect(rect: Rect2, transform: Transform2D) -> Rect2:
	var point_a := transform * rect.position
	var point_b := transform * Vector2(rect.end.x, rect.position.y)
	var point_c := transform * rect.end
	var point_d := transform * Vector2(rect.position.x, rect.end.y)
	var minimum_x := minf(minf(point_a.x, point_b.x), minf(point_c.x, point_d.x))
	var minimum_y := minf(minf(point_a.y, point_b.y), minf(point_c.y, point_d.y))
	var maximum_x := maxf(maxf(point_a.x, point_b.x), maxf(point_c.x, point_d.x))
	var maximum_y := maxf(maxf(point_a.y, point_b.y), maxf(point_c.y, point_d.y))
	return Rect2(minimum_x, minimum_y, maximum_x - minimum_x, maximum_y - minimum_y)

func _ready() -> void:
	if authored_layout_enabled and not is_authored_layout_valid():
		_disable_invalid_authored_room()
		_show_authored_layout_error()
		return
	set_process_input(true)
	if not combat_room.room_cleared.is_connected(_on_room_cleared):
		combat_room.room_cleared.connect(_on_room_cleared)
	if player != null and not player.died.is_connected(_on_player_died):
		player.died.connect(_on_player_died)
	if exit_door != null:
		var transition_callback := Callable(self, "_on_exit_door_transition_requested")
		if not exit_door.is_connected("transition_requested", transition_callback):
			exit_door.connect("transition_requested", transition_callback)

	var title_label := get_node_or_null(room_title_label_path) as Label
	if title_label != null:
		title_label.text = room_title

	if player != null:
		player.reset_to_spawn(_get_player_spawn_position())

	# Os filhos já executaram _ready. O controlador mantém o estado da run
	# entre encontros; cenas standalone continuam consumindo o snapshot one-shot.
	var incoming_snapshot: Dictionary = {}
	if not _run_controller_managed:
		incoming_snapshot = RoomTransition.consume_pending_snapshot(_load_current_packed_scene())
	if _run_controller_managed and _managed_run_build != null:
		run_build = _managed_run_build
	else:
		run_build = RunBuild.new()
	if player != null:
		if not incoming_snapshot.is_empty() and player.health != null:
			var restored_build := RunBuild.new()
			if restored_build.replace_state(
				incoming_snapshot.get("upgrade_ids"),
				incoming_snapshot.get("counts_by_id")
			):
				var health_restored := player.health.restore_health_for_room_transition(
					int(incoming_snapshot["hp_current"]),
					int(incoming_snapshot["hp_max"])
				)
				if health_restored:
					run_build = restored_build
		player.apply_run_build(run_build)

	_upgrade_offer = UpgradeOfferScene.instantiate() as UpgradeOffer
	if _upgrade_offer != null:
		add_child(_upgrade_offer)
		_upgrade_offer.upgrade_selected.connect(_on_upgrade_selected)
		_upgrade_offer.hide()

	_update_build_status()
	if exit_door != null:
		exit_door.call("set_room_cleared", false)
	if combat_room.room_is_cleared:
		_on_room_cleared()

func _exit_tree() -> void:
	if _upgrade_offer != null and is_instance_valid(_upgrade_offer):
		_upgrade_offer.close_offer()
	_guard_active = false
	_guard_waiting_for_release = false
	_guard_quiet_remaining = 0.0
	# Do not unpause during teardown; the new room inherits the actual tree state.
	_owned_offer_pause = false

func _input(event: InputEvent) -> void:
	if authored_layout_enabled and not is_authored_layout_valid():
		for action in DISCRETE_INPUT_ACTIONS:
			if event.is_action_pressed(action):
				get_viewport().set_input_as_handled()
				return
	if _upgrade_flow_state == UpgradeFlowState.OFFER_OPEN and event.is_action_pressed("interact"):
		get_viewport().set_input_as_handled()
		return
	if not _guard_active:
		return
	for action in DISCRETE_INPUT_ACTIONS:
		if event.is_action_pressed(action):
			_guard_waiting_for_release = true
			_guard_quiet_remaining = 0.0
			get_viewport().set_input_as_handled()
			return

func _process(delta: float) -> void:
	if not _guard_active:
		return
	if _any_discrete_input_pressed():
		_guard_waiting_for_release = true
		_guard_quiet_remaining = 0.0
		return
	if _guard_waiting_for_release:
		_guard_waiting_for_release = false
		_guard_quiet_remaining = maxf(post_upgrade_input_guard_seconds, 0.0)
		if _guard_quiet_remaining <= 0.0:
			_end_input_guard()
		return

	_guard_quiet_remaining = maxf(_guard_quiet_remaining - delta, 0.0)
	if _guard_quiet_remaining <= 0.0:
		_end_input_guard()

func _physics_process(_delta: float) -> void:
	if authored_layout_enabled and not is_authored_layout_valid():
		return
	if player == null or not is_instance_valid(player) or player.is_dead:
		return
	if player.global_position.y > 760.0:
		player.reset_to_spawn(_get_player_spawn_position())

func is_gameplay_action_blocked(action: StringName) -> bool:
	if authored_layout_enabled and not is_authored_layout_valid():
		return true
	if not DISCRETE_INPUT_ACTIONS.has(action):
		return false
	if _upgrade_flow_state in [
		UpgradeFlowState.VICTORY_PENDING,
		UpgradeFlowState.OFFER_OPEN,
		UpgradeFlowState.SELECTED,
	]:
		return true
	return _guard_active

func _on_room_cleared() -> void:
	if authored_layout_enabled and not is_authored_layout_valid():
		return
	if _upgrade_flow_state != UpgradeFlowState.INACTIVE:
		return
	_upgrade_flow_state = UpgradeFlowState.VICTORY_PENDING
	if player != null and is_instance_valid(player):
		player.ball_launcher.request_stop_active_sweep()
	_queue_post_flush_resolution()

func _on_player_died() -> void:
	_death_pending = true
	_upgrade_flow_state = UpgradeFlowState.CANCELLED
	if player != null and is_instance_valid(player):
		player.ball_launcher.request_stop_active_sweep()
	_queue_post_flush_resolution()

func _queue_post_flush_resolution() -> void:
	if _post_flush_resolution_queued:
		return
	_post_flush_resolution_queued = true
	call_deferred("_resolve_after_physics_flush")

func _resolve_after_physics_flush() -> void:
	if authored_layout_enabled and not is_authored_layout_valid():
		_post_flush_resolution_queued = false
		return
	_post_flush_resolution_queued = false
	if player != null and is_instance_valid(player):
		player.ball_launcher.cleanup_active_ball_after_flush()
	combat_room.cleanup_hostile_projectiles_after_flush()
	combat_room.refresh_status_after_physics()

	if _death_pending or player == null or not is_instance_valid(player) or player.is_dead:
		_death_pending = true
		_upgrade_flow_state = UpgradeFlowState.CANCELLED
		_transition_in_progress = false
		run_build.clear()
		RoomTransition.clear_pending_snapshot()
		_offered_upgrade_ids.clear()
		if _upgrade_offer != null and is_instance_valid(_upgrade_offer):
			_upgrade_offer.close_offer()
		if exit_door != null:
			exit_door.call("set_room_cleared", false)
		_close_owned_offer_pause()
		_end_input_guard()
		_update_build_status()
		if player != null and is_instance_valid(player):
			player.finish_death_after_physics()
		return

	if _upgrade_flow_state != UpgradeFlowState.VICTORY_PENDING or not combat_room.room_is_cleared:
		return
	if reward_enabled and run_build.total_count() == 0:
		_open_upgrade_offer()
	else:
		_release_exit_or_complete()

func _open_upgrade_offer() -> void:
	if _upgrade_offer == null or not is_instance_valid(_upgrade_offer):
		_upgrade_flow_state = UpgradeFlowState.CANCELLED
		push_error("Main: não foi possível criar a oferta de upgrades.")
		if _run_controller_managed:
			_release_exit_or_complete()
		return
	_offered_upgrade_ids.clear()
	for definition in _upgrade_definitions:
		_offered_upgrade_ids.append(definition.upgrade_id)
	if not _upgrade_offer.present_options(_upgrade_definitions):
		_offered_upgrade_ids.clear()
		_upgrade_flow_state = UpgradeFlowState.CANCELLED
		push_error("Main: definições inválidas para a oferta.")
		if _run_controller_managed:
			_release_exit_or_complete()
		return

	_owned_offer_pause = not get_tree().paused
	_upgrade_flow_state = UpgradeFlowState.OFFER_OPEN
	if _owned_offer_pause:
		get_tree().paused = true

func _on_upgrade_selected(upgrade_id: StringName) -> void:
	if _upgrade_flow_state != UpgradeFlowState.OFFER_OPEN:
		return
	if player == null or not is_instance_valid(player) or player.is_dead:
		return
	if not _offered_upgrade_ids.has(upgrade_id):
		return
	if not RunBuild.is_known_upgrade_id(upgrade_id):
		return
	if run_build.count_for(upgrade_id) != 0 or run_build.total_count() >= RunBuild.MAX_UPGRADES:
		return

	# State changes before the build and player configurations are mutated.
	_upgrade_flow_state = UpgradeFlowState.SELECTED
	if not run_build.add_upgrade(upgrade_id):
		_upgrade_flow_state = UpgradeFlowState.OFFER_OPEN
		return
	player.apply_run_build(run_build)
	_update_build_status()
	_offered_upgrade_ids.clear()
	if _upgrade_offer != null and is_instance_valid(_upgrade_offer):
		_upgrade_offer.close_offer()
	_begin_input_guard()
	_close_owned_offer_pause()
	_release_exit_or_complete()

func _release_exit_or_complete() -> void:
	if exit_door == null:
		_upgrade_flow_state = UpgradeFlowState.COMPLETED
		return
	_upgrade_flow_state = UpgradeFlowState.EXIT_RELEASED
	_door_unlock_pending = true
	_release_exit_door_if_ready()

func _release_exit_door_if_ready() -> void:
	if not _door_unlock_pending or _guard_active or _upgrade_flow_state != UpgradeFlowState.EXIT_RELEASED:
		return
	if not combat_room.room_is_cleared or player == null or not is_instance_valid(player) or player.is_dead:
		return
	_door_unlock_pending = false
	exit_door.call("set_room_cleared", true)

func _notify_run_progression_ready() -> void:
	if not _run_controller_managed or _run_progression_notified:
		return
	_run_progression_notified = true
	run_progression_ready.emit()

func _begin_input_guard() -> void:
	_guard_active = true
	_guard_waiting_for_release = _any_discrete_input_pressed()
	_guard_quiet_remaining = 0.0 if _guard_waiting_for_release else maxf(post_upgrade_input_guard_seconds, 0.0)
	if not _guard_waiting_for_release and _guard_quiet_remaining <= 0.0:
		_end_input_guard()

func _end_input_guard() -> void:
	_guard_active = false
	_guard_waiting_for_release = false
	_guard_quiet_remaining = 0.0
	_release_exit_door_if_ready()

func _any_discrete_input_pressed() -> bool:
	for action in DISCRETE_INPUT_ACTIONS:
		if Input.is_action_pressed(action):
			return true
	return false

func _close_owned_offer_pause() -> void:
	if _owned_offer_pause and get_tree().paused:
		get_tree().paused = false
	_owned_offer_pause = false

func _on_exit_door_transition_requested(requested_player: Player) -> void:
	if _transition_in_progress or _upgrade_flow_state != UpgradeFlowState.EXIT_RELEASED:
		return
	if _guard_active or not combat_room.room_is_cleared:
		exit_door.call("set_transition_busy", false)
		return
	if requested_player == null or not is_instance_valid(requested_player) or requested_player != player or requested_player.is_dead:
		exit_door.call("report_transition_failure", "JOGADOR INDISPONÍVEL")
		return
	if not bool(exit_door.call("is_player_in_range", requested_player)):
		exit_door.call("set_transition_busy", false)
		return

	_transition_in_progress = true
	if _run_controller_managed:
		exit_door.call("set_transition_busy", true)
		call_deferred("_complete_managed_progression_deferred", requested_player)
		return

	var destination_path := next_room_scene_path.strip_edges()
	if destination_path.is_empty():
		_upgrade_flow_state = UpgradeFlowState.COMPLETED
		exit_door.call("set_terminal_complete")
		return

	call_deferred("_change_room_deferred", destination_path, requested_player)

func _complete_managed_progression_deferred(requested_player: Player) -> void:
	if not _transition_in_progress or _upgrade_flow_state != UpgradeFlowState.EXIT_RELEASED:
		return
	var player_is_valid := (
		requested_player != null
		and is_instance_valid(requested_player)
		and requested_player == player
		and not requested_player.is_dead
	)
	var request_is_valid := (
		player_is_valid
		and not _guard_active
		and combat_room.room_is_cleared
		and exit_door != null
		and bool(exit_door.call("is_player_in_range", requested_player))
	)
	if not request_is_valid:
		_transition_in_progress = false
		if exit_door != null:
			exit_door.call("set_transition_busy", false)
			if not player_is_valid or not combat_room.room_is_cleared:
				exit_door.call("set_room_cleared", false)
		return
	_notify_run_progression_ready()

func _change_room_deferred(destination_path: String, requested_player: Player) -> void:
	if not _transition_in_progress or _upgrade_flow_state != UpgradeFlowState.EXIT_RELEASED:
		return
	if not combat_room.room_is_cleared or _guard_active:
		_cancel_transition_request(destination_path, "")
		return
	if requested_player == null or not is_instance_valid(requested_player) or requested_player != player or requested_player.is_dead:
		_cancel_transition_request(destination_path, "")
		return
	if exit_door == null or not bool(exit_door.call("is_player_in_range", requested_player)):
		_cancel_transition_request(destination_path, "")
		return

	var destination_scene := ResourceLoader.load(destination_path, "PackedScene") as PackedScene
	if destination_scene == null:
		_cancel_transition_request(destination_path, "FALHA AO CARREGAR SALA")
		return
	var canonical_destination_path := destination_scene.resource_path
	if canonical_destination_path.is_empty() or not canonical_destination_path.begins_with("res://"):
		_cancel_transition_request(destination_path, "FALHA AO CARREGAR SALA")
		return

	if not RoomTransition.prepare_run_snapshot(
		destination_scene,
		requested_player.current_health,
		requested_player.maximum_health,
		run_build
	):
		_cancel_transition_request(canonical_destination_path, "FALHA AO PREPARAR TRANSIÇÃO")
		return
	var change_result := get_tree().change_scene_to_packed(destination_scene)
	if change_result != OK:
		_cancel_transition_request(canonical_destination_path, "FALHA AO CARREGAR SALA")

func _cancel_transition_request(_destination_path: String, message: String) -> void:
	RoomTransition.clear_pending_snapshot()
	_transition_in_progress = false
	if exit_door == null:
		return
	if not message.is_empty():
		exit_door.call("report_transition_failure", message)
	elif _upgrade_flow_state == UpgradeFlowState.EXIT_RELEASED and combat_room.room_is_cleared:
		exit_door.call("set_transition_busy", false)
	else:
		exit_door.call("set_room_cleared", false)

func _update_build_status() -> void:
	var label := get_node_or_null(build_status_label_path) as Label
	if label == null:
		return
	if run_build == null or run_build.total_count() == 0:
		label.text = "BUILD: VAZIA"
		return

	var upgrade_id := run_build.get_upgrade_ids()[0]
	var display_name := ""
	for definition in _upgrade_definitions:
		if definition.upgrade_id == upgrade_id:
			display_name = definition.title
			break
	label.text = "BUILD: %s" % display_name.to_upper()

func _get_player_spawn_position() -> Vector2:
	var spawn_marker := get_node_or_null(player_spawn_marker_path) as Marker2D
	if spawn_marker != null:
		return spawn_marker.global_position
	return FALLBACK_PLAYER_SPAWN

func _current_scene_path() -> String:
	if not scene_file_path.is_empty():
		return scene_file_path
	var current_scene := get_tree().current_scene
	if current_scene != null:
		return current_scene.scene_file_path
	return ""

func _load_current_packed_scene() -> PackedScene:
	var current_path := _current_scene_path()
	if current_path.is_empty():
		return null
	return ResourceLoader.load(current_path, "PackedScene") as PackedScene

	_transition_busy = false
	_failure_message = message.strip_edges()
	_update_presentation()

func _unhandled_input(event: InputEvent) -> void:
	if door_role != 0 or not interaction_enabled or _terminal_complete:
		return
	if not event is InputEventKey:
		return
	var key_event := event as InputEventKey
	if key_event.echo or not key_event.is_action_pressed("interact"):
		return
	if _room_root != null \
		and _room_root.has_method("is_gameplay_action_blocked") \
		and bool(_room_root.call("is_gameplay_action_blocked", &"interact")):
		get_viewport().set_input_as_handled()
		return
	if _transition_busy or not _room_is_cleared:
		return
	if not Input.is_action_just_pressed("interact"):
		return

	var player := _get_live_player_in_range()
	if player == null:
		return

	_transition_busy = true
	_update_presentation()
	transition_requested.emit(player)
	get_viewport().set_input_as_handled()

func _on_body_entered(body: Node2D) -> void:
	if body is Player and body.is_in_group("player"):
		_players_in_range[body] = true

func _on_body_exited(body: Node2D) -> void:
	_players_in_range.erase(body)

func is_player_in_range(player: Player) -> bool:
	if door_role != 0 or not interaction_enabled or _terminal_complete:
		return false
	if player == null or not is_instance_valid(player) or not _players_in_range.has(player):
		return false
	return get_overlapping_bodies().has(player)

func _get_live_player_in_range() -> Player:
	for candidate in _players_in_range.keys():
		if not is_instance_valid(candidate):
			continue
		var player := candidate as Player
		if player != null and player.is_in_group("player") and not player.is_dead and is_player_in_range(player):
			return player
	return null

func set_terminal_complete() -> void:
	_terminal_complete = true
	_room_is_cleared = false
	_transition_busy = false
	_failure_message = ""
	_update_presentation()

func _update_presentation() -> void:
	if not is_instance_valid(_door_panel):
		return
	if door_role != 0:
		_door_panel.color = Color("#302a22")
		_portal_glow.color = Color("#d68448", 0.18)
		_door_rim.default_color = Color("#d68448")
		_door_seal.visible = true
		_door_label.text = "ENTRADA\nACESSO BLOQUEADO"
		return
	if _terminal_complete:
		_door_panel.color = Color("#18232b")
		_portal_glow.color = Color("#63c9b1", 0.32)
		_door_rim.default_color = Color("#79d9c0")
		_door_seal.visible = false
		_door_label.text = "ETAPA CONCLUÍDA"
		return

	var is_open := _room_is_cleared and not _transition_busy
	_door_panel.color = Color("#18232b") if is_open else Color("#24252a")
	_portal_glow.color = Color("#63c9b1", 0.32) if is_open else Color("#d68448", 0.2)
	_door_rim.default_color = Color("#79d9c0") if is_open else Color("#d68448")
	_door_seal.visible = not _room_is_cleared or _transition_busy

	if _transition_busy:
		_door_label.text = "ABRINDO SAÍDA..."
	elif not _room_is_cleared:
		_door_label.text = "SAÍDA BLOQUEADA\nELIMINE OS INIMIGOS"
	elif not _failure_message.is_empty():
		_door_label.text = "%s\nE TENTAR DE NOVO" % _failure_message
	else:
		_door_label.text = "SAÍDA ABERTA\n[E] ENTRAR"

FILE scripts/world/room_layout_generator.gd
extends RefCounted

const MIN_PLATFORM_COUNT := 2
const MAX_PLATFORM_COUNT := 5
const MIN_RANDOM_WIDTH := 160
const MIN_FALLBACK_WIDTH := 104
const MAX_PLATFORM_WIDTH := 216
const PLATFORM_THICKNESS := 24
const GAP_MIN := 56
const GAP_MAX := 96
const FIRST_LEFT_MIN := 208
const FIRST_LEFT_MAX := 256
const RIGHT_LIMIT := 980
const FIRST_TOP := 440
const TOP_LIMIT := 376
const MAX_RISE_PER_STEP := 64
const MAX_ATTEMPTS := 32

static func generate_layout(platform_count: int, requested_seed: int, spawn_x: float) -> Dictionary:
	var rng := RandomNumberGenerator.new()
	if requested_seed == 0:
		rng.randomize()
	else:
		rng.seed = requested_seed
	var seed_used: int = rng.seed
	if platform_count < MIN_PLATFORM_COUNT or platform_count > MAX_PLATFORM_COUNT:
		return _invalid_result(seed_used, "quantidade atual de plataformas fora do intervalo suportado (2..5)")

	var minimum_width: int = MIN_RANDOM_WIDTH if platform_count == 3 else MIN_FALLBACK_WIDTH
	var available_width := RIGHT_LIMIT - FIRST_LEFT_MIN - GAP_MIN * (platform_count - 1)
	var maximum_width := mini(MAX_PLATFORM_WIDTH, floori(float(available_width) / float(platform_count) / 8.0) * 8)
	if maximum_width < minimum_width:
		return _invalid_result(seed_used, "não há largura suficiente para uma rota segura com esta quantidade")

	for attempt in MAX_ATTEMPTS:
		var first_left := rng.randi_range(FIRST_LEFT_MIN / 8, FIRST_LEFT_MAX / 8) * 8
		var width_room := RIGHT_LIMIT - first_left - GAP_MIN * (platform_count - 1)
		var candidate_max_width := mini(maximum_width, floori(float(width_room) / float(platform_count) / 8.0) * 8)
		if candidate_max_width < minimum_width:
			continue

		var entries: Array[Dictionary] = []
		var left := float(first_left)
		var previous_top := FIRST_TOP
		for index in platform_count:
			var width := rng.randi_range(minimum_width / 8, candidate_max_width / 8) * 8
			var top := FIRST_TOP
			if index > 0:
				var rise := rng.randi_range(2, MAX_RISE_PER_STEP / 8) * 8
				top = maxi(TOP_LIMIT, previous_top - rise)
			previous_top = top
			var rect := Rect2(Vector2(left, top), Vector2(width, PLATFORM_THICKNESS))
			entries.append(_entry_from_rect(rect))
			left = rect.end.x
			if index < platform_count - 1:
				left += rng.randi_range(GAP_MIN / 8, GAP_MAX / 8) * 8

		if _entries_are_safe(entries, spawn_x):
			return _success_result(entries, seed_used, false)

	var fallback_entries := _safe_fallback(platform_count)
	if not _entries_are_safe(fallback_entries, spawn_x):
		return _invalid_result(seed_used, "fallback fixo não satisfaz os limites de sala")
	return _success_result(fallback_entries, seed_used, true)

static func _safe_fallback(platform_count: int) -> Array[Dictionary]:
	var entries: Array[Dictionary] = []
	if platform_count == 3:
		var centers := [324.0, 596.0, 860.0]
		var widths := [200.0, 184.0, 184.0]
		var tops := [440.0, 400.0, 376.0]
		for index in platform_count:
			var width: float = widths[index]
			var top: float = tops[index]
			var rect := Rect2(Vector2(centers[index] - width * 0.5, top), Vector2(width, PLATFORM_THICKNESS))
			entries.append(_entry_from_rect(rect))
		return entries

	var gap := GAP_MIN
	var available_width := RIGHT_LIMIT - FIRST_LEFT_MIN - gap * (platform_count - 1)
	var width := mini(MAX_PLATFORM_WIDTH, floori(float(available_width) / float(platform_count) / 8.0) * 8)
	if width < MIN_FALLBACK_WIDTH:
		return entries
	for index in platform_count:
		var left := float(FIRST_LEFT_MIN + index * (width + gap))
		var top := maxi(TOP_LIMIT, FIRST_TOP - mini(index, 2) * 32)
		var rect := Rect2(Vector2(left, top), Vector2(width, PLATFORM_THICKNESS))
		entries.append(_entry_from_rect(rect))
	return entries

static func _entries_are_safe(entries: Array[Dictionary], spawn_x: float) -> bool:
	if entries.size() < MIN_PLATFORM_COUNT:
		return false
	var previous: Rect2
	for index in entries.size():
		var rect: Rect2 = entries[index].get("rect", Rect2())
		if rect.size.y != PLATFORM_THICKNESS or rect.size.x < MIN_FALLBACK_WIDTH or rect.size.x > MAX_PLATFORM_WIDTH:
			return false
		if rect.position.x < 64.0 or rect.end.x > RIGHT_LIMIT or fposmod(rect.position.x, 8.0) > 0.01 or fposmod(rect.size.x, 8.0) > 0.01:
			return false
		if rect.position.y < TOP_LIMIT or rect.position.y > FIRST_TOP or rect.end.y > 464.0:
			return false
		if 560.0 - rect.end.y < 96.0:
			return false
		if index == 0:
			if absf(rect.position.y - FIRST_TOP) > 0.01:
				return false
			if rect.position.x < FIRST_LEFT_MIN or rect.position.x > FIRST_LEFT_MAX or rect.position.x - (spawn_x + 14.0) < 24.0:
				return false
		else:
			var gap := rect.position.x - previous.end.x
			var rise := previous.position.y - rect.position.y
			if gap < GAP_MIN or gap > GAP_MAX or rise < 0.0 or rise > MAX_RISE_PER_STEP:
				return false
			if rect.intersects(previous):
				return false
		previous = rect
	return true

static func _entry_from_rect(rect: Rect2) -> Dictionary:
	return {
		"position": rect.get_center(),
		"size": rect.size,
		"rect": rect,
	}

static func _success_result(entries: Array[Dictionary], seed_used: int, used_fallback: bool) -> Dictionary:
	return {
		"valid": true,
		"platforms": entries,
		"primary_index": entries.size() - 1,
		"fallback_index": 0,
		"seed": seed_used,
		"used_fallback": used_fallback,
	}

static func _invalid_result(seed_used: int, message: String) -> Dictionary:
	return {
		"valid": false,
		"platforms": [],
		"primary_index": -1,
		"fallback_index": -1,
		"seed": seed_used,
		"used_fallback": false,
		"error": message,
	}

END FILE

FILE scripts/world/run_controller.gd
extends Node2D

const RunBuild = preload("res://scripts/progression/run_build.gd")
const RoomTransition = preload("res://scripts/world/room_transition.gd")
const RewardScreenScript = preload("res://scripts/ui/reward_screen.gd")

@export_range(1, 20) var floors_per_run: int = 8
@export_range(1, 10) var common_rooms_per_floor: int = 3
@export_range(0.2, 10.0, 0.1) var transition_delay: float = 1.5
@export var common_room_scene: PackedScene
@export var boss_room_scene: PackedScene

@onready var room_container: Node2D = $RoomContainer
@onready var progress_label: Label = $HUD/ProgressLabel
@onready var status_label: Label = $HUD/StatusLabel

var current_floor: int = 1
var current_room: int = 1
var current_room_instance: Node
var transition_pending: bool = false
var _run_ended: bool = false
var _damage_multiplier: float = 1.0
var _speed_multiplier: float = 1.0
var _radius_multiplier: float = 1.0
var _run_player_health: int = -1
var _run_player_max_health: int = -1
var _run_build: RunBuild = RunBuild.new()
var _active_reward_screen: Node
var _owned_reward_pause: bool = false
var _restart_in_progress: bool = false


func _ready() -> void:
	# The run controller must handle overlays and progression while gameplay is paused.
	process_mode = Node.PROCESS_MODE_ALWAYS
	RoomTransition.clear_pending_snapshot()
	if common_room_scene == null:
		_fail_run("ERRO: cena de sala comum não definida.")
		return
	if boss_room_scene == null:
		_fail_run("ERRO: cena de chefe não definida.")
		return
	_load_current_encounter()


func _is_boss_room() -> bool:
	return current_room > common_rooms_per_floor


func _is_final_boss() -> bool:
	return current_floor == floors_per_run and _is_boss_room()


func _load_current_encounter() -> void:
	if _run_ended:
		return

	var packed_scene: PackedScene = boss_room_scene if _is_boss_room() else common_room_scene
	var next_room := packed_scene.instantiate()
	# RunRoot processes while paused; room descendants must still stop with gameplay.
	next_room.process_mode = Node.PROCESS_MODE_PAUSABLE
	if not next_room.has_method("configure_run_managed"):
		next_room.free()
		_fail_run("ERRO: a cena de sala não oferece integração gerenciada.")
		return
	if not next_room.has_signal("run_progression_ready"):
		next_room.free()
		_fail_run("ERRO: a sala não oferece o sinal run_progression_ready.")
		return
	var should_offer_build_upgrade := _run_build.total_count() == 0
	if not bool(next_room.call("configure_run_managed", _run_build, should_offer_build_upgrade)):
		var layout_error := ""
		if next_room.has_method("get_authored_layout_error"):
			layout_error = String(next_room.call("get_authored_layout_error"))
		next_room.free()
		var failure_message := "ERRO: não foi possível configurar a sala para esta run."
		if not layout_error.is_empty():
			failure_message += " Layout: " + layout_error
		_fail_run(failure_message)
		return

	var progression_callback := Callable(self, "_on_room_progression_ready").bind(next_room)
	next_room.connect("run_progression_ready", progression_callback, CONNECT_ONE_SHOT)
	room_container.add_child(next_room)
	current_room_instance = next_room

	var player := _find_player(next_room)
	if player == null or player.health == null:
		_discard_current_room("ERRO: a sala não contém um jogador com PlayerHealth.")
		return

	if _run_player_health < 0:
		_run_player_health = player.current_health
		_run_player_max_health = player.maximum_health
	else:
		if _run_player_health <= 0 or _run_player_max_health <= 0:
			_discard_current_room("ERRO: HP inválido; a run não pode continuar.")
			return
		if not player.health.restore_health_for_room_transition(
			_run_player_health,
			_run_player_max_health
		):
			_discard_current_room("ERRO: não foi possível restaurar o HP da run.")
			return

	if not player.health_changed.is_connected(_on_player_health_changed):
		player.health_changed.connect(_on_player_health_changed)
	if not player.died.is_connected(_on_player_died):
		player.died.connect(_on_player_died)

func take_damage(amount: int, _source: Node = null) -> bool:
	if _defeated_emitted or health <= 0 or amount <= 0:
		return false

	health = maxi(health - amount, 0)
	_flash_remaining = maxf(flash_duration, 0.0)
	_update_visuals()

		_cancel_attack()

func _physics_process(delta: float) -> void:
	_update_hit_flash(delta)
	if _defeated_emitted or _placement_pending:
		return
	if not _is_inside_current_scene():
		_cancel_attack()
		return
	if _attack_state != AttackState.IDLE and not _is_alive_player(_locked_player):
		_cancel_attack()
		return

	var physics_step := maxf(delta, 0.0)
	match _attack_state:
		AttackState.IDLE:
			var detected_player := _find_player_in_attack_area(global_position + attack_center_offset)
			if detected_player != null:
				_begin_windup(detected_player)
		AttackState.WINDUP:
			_update_windup(physics_step)
		AttackState.ACTIVE:
			_update_active(physics_step)
		AttackState.RECOVERY:
			_update_recovery(physics_step)

	_area_query = PhysicsShapeQueryParameters2D.new()
	_area_query.shape = _area_shape
	_area_query.collision_mask = PLAYER_MASK
	_area_query.collide_with_bodies = true
	_area_query.collide_with_areas = false
	_area_query.margin = 0.0
	_area_query.exclude = [get_rid()]

	_line_of_sight_query = PhysicsRayQueryParameters2D.new()
	_line_of_sight_query.collision_mask = WORLD_MASK
	_line_of_sight_query.collide_with_bodies = true
	_line_of_sight_query.collide_with_areas = false
	_line_of_sight_query.exclude = [get_rid()]

func _effective_attack_radius() -> float:
	return clampf(attack_radius, 16.0, 400.0)

func _find_player_in_attack_area(area_center: Vector2) -> PhysicsBody2D:
	if _area_query == null or _area_shape == null:
		return null
	_area_shape.radius = _effective_attack_radius()
	_area_query.transform = Transform2D(0.0, area_center)
	var overlaps: Array[Dictionary] = get_world_2d().direct_space_state.intersect_shape(
		_area_query,
		MAX_QUERY_RESULTS
	)
	for overlap in overlaps:
		var collider: Variant = overlap.get("collider")
		if not collider is PhysicsBody2D:
			continue
		var player_body := collider as PhysicsBody2D
		if not _is_alive_player(player_body):
			continue
		if _has_clear_line_of_sight(player_body, area_center):
			return player_body
	return null

func _is_alive_player(candidate: PhysicsBody2D) -> bool:
	if not is_instance_valid(candidate):
		return false
	if not candidate.is_in_group("player") or not candidate.has_method("take_damage"):
		return false
	var dead_value: Variant = candidate.get("is_dead")
	if typeof(dead_value) != TYPE_BOOL or bool(dead_value):
		return false
	return _node_is_inside_current_scene(candidate)

func _has_clear_line_of_sight(player_body: PhysicsBody2D, area_center: Vector2) -> bool:
	_line_of_sight_query.from = area_center
	_line_of_sight_query.to = _get_player_collider_center(player_body)
	var obstruction: Dictionary = get_world_2d().direct_space_state.intersect_ray(
		_line_of_sight_query
	)
	return obstruction.is_empty()

func _get_player_collider_center(player_body: PhysicsBody2D) -> Vector2:
	var player_shape := player_body.get_node_or_null("CollisionShape2D") as CollisionShape2D
	if player_shape != null and not player_shape.disabled:
		return player_shape.global_position
	return player_body.global_position

func _begin_windup(player_body: PhysicsBody2D) -> void:
	_locked_player = player_body
	_locked_attack_center = global_position + attack_center_offset
	_damaged_player_ids.clear()
	_attack_state = AttackState.WINDUP
	_state_remaining = maxf(windup_duration, 0.05)
	_recovery_fade_remaining = 0.0
	_update_attack_area_visual()

func _update_windup(delta: float) -> void:
	var before_step := _state_remaining
	_state_remaining = maxf(_state_remaining - delta, 0.0)
	if _state_remaining <= 0.0:
		var overshoot := maxf(delta - before_step, 0.0)
		_enter_active()
		_apply_attack_area()
		if overshoot >= maxf(active_window, 0.05):
			_enter_recovery()
		else:
			_state_remaining = maxf(active_window, 0.05) - overshoot
		return
	_update_attack_area_visual()

func _enter_active() -> void:
	_attack_state = AttackState.ACTIVE
	_state_remaining = maxf(active_window, 0.05)
	_update_attack_area_visual()
	_update_visuals()

func _update_active(delta: float) -> void:
	_apply_attack_area()
	_state_remaining = maxf(_state_remaining - delta, 0.0)
	if _state_remaining <= 0.0:
		_enter_recovery()

func _apply_attack_area() -> void:
	if _defeated_emitted or _attack_state != AttackState.ACTIVE:
		return
	var overlaps: Array[Dictionary] = _find_overlaps_at_locked_center()
	for overlap in overlaps:
		var collider: Variant = overlap.get("collider")
		if not collider is PhysicsBody2D:
			continue
		var player_body := collider as PhysicsBody2D
		if not _is_alive_player(player_body):
			continue
		var player_id := player_body.get_instance_id()
		if _damaged_player_ids.has(player_id):
			continue
		if not _has_clear_line_of_sight(player_body, _locked_attack_center):
			continue
		_damaged_player_ids[player_id] = true
		var damage_result: Variant = player_body.call(
			"take_damage",
			maxi(attack_damage, 1),
			self
		)
		if typeof(damage_result) == TYPE_BOOL and bool(damage_result):
			_set_attack_visual(&"impact", 1.0)

func _find_overlaps_at_locked_center() -> Array[Dictionary]:
	_area_shape.radius = _effective_attack_radius()
	_area_query.transform = Transform2D(0.0, _locked_attack_center)
	return get_world_2d().direct_space_state.intersect_shape(
		_area_query,
		MAX_QUERY_RESULTS
	)

func _enter_recovery() -> void:
	_attack_state = AttackState.RECOVERY
	_state_remaining = maxf(recovery_duration, 0.05)
	_recovery_fade_remaining = maxf(recovery_fade_duration, 0.0)
	_update_attack_area_visual()
	_update_visuals()

func _update_recovery(delta: float) -> void:
	_state_remaining = maxf(_state_remaining - delta, 0.0)
	_recovery_fade_remaining = maxf(_recovery_fade_remaining - delta, 0.0)
	_update_attack_area_visual()
	if _state_remaining <= 0.0:
		_attack_state = AttackState.IDLE
		_state_remaining = 0.0
		_recovery_fade_remaining = 0.0
		_locked_player = null
		_damaged_player_ids.clear()
		_set_attack_visual(&"hidden", 0.0)
		_update_visuals()

func _update_attack_area_visual() -> void:
	match _attack_state:
		AttackState.WINDUP:
			var windup_progress := 1.0 - clampf(
				_state_remaining / maxf(windup_duration, 0.05),
				0.0,
				1.0
			)
			_set_attack_visual(&"windup", 0.58 + windup_progress * 0.34)
		AttackState.ACTIVE:
			_set_attack_visual(&"active", 1.0)
		AttackState.RECOVERY:
			var fade_alpha := 0.0
			if recovery_fade_duration > 0.0:
				fade_alpha = _recovery_fade_remaining / recovery_fade_duration
			_set_attack_visual(&"recovery", clampf(fade_alpha, 0.0, 1.0))
		_:
			_set_attack_visual(&"hidden", 0.0)

func _set_attack_visual(phase: StringName, alpha: float) -> void:
	if not is_instance_valid(_attack_visual) or not _attack_visual.has_method("set_area_state"):
		return
	_attack_visual.call(
		"set_area_state",
		_effective_attack_radius(),
		attack_center_offset if _attack_state == AttackState.IDLE else _locked_attack_center - global_position,
		phase,
		alpha
	)

func _update_hit_flash(delta: float) -> void:
	if _flash_remaining <= 0.0:
		return
	_flash_remaining = maxf(_flash_remaining - maxf(delta, 0.0), 0.0)
	_update_visuals()

func _update_visuals() -> void:
	if not is_instance_valid(_body_sprite) or not is_instance_valid(_health_fill):
		return
	var desired_animation: StringName = &"impact" if _flash_remaining > 0.0 or _attack_state == AttackState.ACTIVE else &"idle"
	if _body_sprite.animation != desired_animation or not _body_sprite.is_playing():
		_body_sprite.play(desired_animation)
	var hit_flash := _body_sprite.material as ShaderMaterial
	if hit_flash != null:
		hit_flash.set_shader_parameter("flash_amount", 1.0 if _flash_remaining > 0.0 else 0.0)
	var health_ratio := clampf(float(health) / float(maxi(max_health, 1)), 0.0, 1.0)
	_health_fill.scale.x = health_ratio

func _cancel_attack() -> void:
	_attack_state = AttackState.IDLE
	_state_remaining = 0.0
	_recovery_fade_remaining = 0.0
	_locked_player = null
	_damaged_player_ids.clear()
	_set_attack_visual(&"hidden", 0.0)
	_update_visuals()

	return _node_is_inside_current_scene(self)

func _node_is_inside_current_scene(node: Node) -> bool:
	if not is_instance_valid(node):
		return false
	var scene_root := get_tree().current_scene
	return is_instance_valid(scene_root) and (scene_root == node or scene_root.is_ancestor_of(node))

func _get_collision_rect() -> Rect2:

		var collision := child as CollisionShape2D
		if collision != null and collision.shape != null:
			var rect := collision.shape.get_rect()
			rect.position += collision.position
			return rect
	# Sem CollisionShape2D: assume que a origem está no centro da base.
	return Rect2(-16.0, -32.0, 32.0, 32.0)

func _place_on_platform() -> void:
	_placement_pending = false
	var body := _get_collision_rect()
	var needed_width := body.size.x + edge_margin * 2.0

	# Prefere plataformas livres; o Floor só entra se não sobrar nenhuma,
	# e plataformas já ocupadas por outro inimigo ficam por último.
	var free: Array[SolidPlatform] = []
	var floors: Array[SolidPlatform] = []
	var taken: Array[SolidPlatform] = []
	for node in get_tree().get_nodes_in_group(&"solid_platform"):
		var candidate := node as SolidPlatform
		if candidate == null or candidate.size.x < needed_width:
			continue
		if candidate.name == &"Floor":
			floors.append(candidate)
		elif candidate.has_meta(CLAIMED_META):
			taken.append(candidate)
		else:
			free.append(candidate)

	var pool: Array[SolidPlatform] = free
	if pool.is_empty():
		pool = floors
	if pool.is_empty():
		pool = taken
	if pool.is_empty():
		push_warning("%s: nenhuma plataforma larga o bastante; mantendo a posição original." % name)
		return

	var rng := RandomNumberGenerator.new()
	if SolidPlatform.room_seed == 0:
		rng.randomize()
	else:
		rng.seed = hash([SolidPlatform.room_seed, get_index(), "training_target"])

	var platform := pool[rng.randi_range(0, pool.size() - 1)]
	platform.set_meta(CLAIMED_META, true)

	var platform_left := platform.global_position.x - platform.size.x * 0.5
	var platform_right := platform.global_position.x + platform.size.x * 0.5
	var platform_top := platform.global_position.y - platform.size.y * 0.5

	# Mantém a colisão do inimigo inteira sobre a plataforma, com a base tocando o topo.
	var min_x := platform_left + edge_margin - body.position.x
	var max_x := platform_right - edge_margin - body.end.x
	global_position = Vector2(rng.randf_range(min_x, max_x), platform_top - body.end.y)

@export var randomize_position: bool = true

@export var wall_margin: float = 16.0

@export var width_range: Vector2 = Vector2(96.0, 288.0)

@export var max_jump_height: float = 120.0

@export var max_jump_gap: float = 180.0

@export var max_drop_height: float = 200.0

@export var vertical_clearance: float = 70.0

func _ready() -> void:
	_sync_geometry()
	if Engine.is_editor_hint():
		return
	add_to_group(&"solid_platform")
	if _is_fixed_platform(self):
		set_meta(PLACED_META, true)
		return
	_place_randomly()
	set_meta(PLACED_META, true)

static func _is_fixed_platform(platform: SolidPlatform) -> bool:
	return platform.name == FIXED_NAME or not platform.randomize_position

func _get_rect() -> Rect2:
	return Rect2(position - size * 0.5, size)

func _place_randomly() -> void:
	var parent := get_parent()
	if parent == null:
		return

	# Plataformas já posicionadas (e as fixas, como o Floor) servem de âncora:
	# cada nova plataforma nasce ao alcance de uma delas, então a sala fica alcançável.
	var anchors: Array[Rect2] = []
	for sibling in parent.get_children():
		var other := sibling as SolidPlatform
		if other == null or other == self:
			continue
		if _is_fixed_platform(other) or other.has_meta(PLACED_META):
			anchors.append(other._get_rect())

	if anchors.is_empty():
		push_warning("%s: nenhuma plataforma de referência (Floor) encontrada." % name)
		return

	var rng := RandomNumberGenerator.new()
	if room_seed == 0:
		rng.randomize()
	else:
		rng.seed = hash([room_seed, get_index()])

	for _attempt in max_attempts:
		var new_size := _roll_size(rng) if randomize_width else size
		var anchor: Rect2 = anchors[rng.randi_range(0, anchors.size() - 1)]

		# Altura do topo em relação ao topo da âncora (negativo = mais alto).
		var rise := rng.randf_range(-max_jump_height, max_drop_height)
		var top := anchor.position.y + rise

		var gap_limit := _allowed_gap(rise)
		var center_x := rng.randf_range(
			anchor.position.x - gap_limit - new_size.x * 0.5,
			anchor.end.x + gap_limit + new_size.x * 0.5
		)

		var candidate := Rect2(Vector2(center_x - new_size.x * 0.5, top), new_size)
		candidate.position = Vector2(
			snappedf(candidate.position.x, grid_snap),
			snappedf(candidate.position.y, grid_snap)
		)

		# O snap pode mexer alguns pixels, então revalida o alcance depois dele.
		if not _is_reachable(candidate, anchor):
			continue
		if not _is_valid(candidate, anchors):
			continue

		size = new_size
		position = candidate.position + new_size * 0.5
		return

	push_warning("%s: não achou posição válida; mantendo a posição original." % name)

## Sorteia só a largura; a espessura continua a do editor.
func _roll_size(rng: RandomNumberGenerator) -> Vector2:
	var max_width := minf(width_range.y, room_rect.size.x * max_width_ratio)
	var min_width := minf(width_range.x, max_width)

	# Média de dois sorteios: favorece larguras médias em vez dos extremos.
	var t := (rng.randf() + rng.randf()) * 0.5
	var width := lerpf(min_width, max_width, t)

	return Vector2(maxf(snappedf(width, grid_snap), min_width), size.y)

## Vão horizontal permitido: quanto mais alto o pulo, menor o vão.
func _allowed_gap(rise: float) -> float:
	var rise_ratio := clampf(-rise / maxf(max_jump_height, 1.0), 0.0, 1.0)
	return max_jump_gap * (1.0 - 0.6 * rise_ratio)

func _is_reachable(candidate: Rect2, anchor: Rect2) -> bool:
	var rise := candidate.position.y - anchor.position.y
	if rise < -max_jump_height or rise > max_drop_height:
		return false
	var gap := maxf(0.0, maxf(candidate.position.x - anchor.end.x, anchor.position.x - candidate.end.x))
	return gap <= _allowed_gap(rise)

func _is_valid(candidate: Rect2, anchors: Array[Rect2]) -> bool:
	var bounds := room_rect.grow(-wall_margin)

	# Precisa sobrar espaço acima da plataforma para o jogador pular sem bater no teto.
	var with_headroom := Rect2(
		candidate.position - Vector2(0.0, vertical_clearance),
		candidate.size + Vector2(0.0, vertical_clearance)
	)
	if not bounds.encloses(with_headroom):
		return false

	var padded := candidate.grow_individual(
		horizontal_clearance, vertical_clearance,
		horizontal_clearance, vertical_clearance
	)
	for obstacle in anchors:
		if padded.intersects(obstacle):
			return false
	return true

func _sync_geometry() -> void:
	var collision_shape := get_node_or_null("CollisionShape2D") as CollisionShape2D
	if collision_shape != null:
		var rectangle := collision_shape.shape as RectangleShape2D
		if rectangle != null:
			if not rectangle.resource_local_to_scene:
				rectangle = rectangle.duplicate() as RectangleShape2D
				rectangle.resource_local_to_scene = true
				collision_shape.shape = rectangle
			rectangle.size = size

	var half_size := size * 0.5
	var visual_fill := get_node_or_null("Visuals/Fill") as Polygon2D
	if visual_fill != null:
		visual_fill.color = fill_color
		visual_fill.polygon = PackedVector2Array([
			Vector2(-half_size.x, -half_size.y),
			Vector2(half_size.x, -half_size.y),
			Vector2(half_size.x, half_size.y),
			Vector2(-half_size.x, half_size.y),
		])

	var top_edge := get_node_or_null("Visuals/TopEdge") as Line2D
	if top_edge != null:
		top_edge.points = PackedVector2Array([
			Vector2(-half_size.x, -half_size.y),
			Vector2(half_size.x, -half_size.y),
		])

	var floor_detail := get_node_or_null("FloorDetail") as Line2D
	if floor_detail != null:
		floor_detail.position = Vector2(0.0, -half_size.y + 32.0)
		floor_detail.points = PackedVector2Array([
			Vector2(-half_size.x, 0.0),
			Vector2(half_size.x, 0.0),
		])

	_sync_surface_skin()

func _sync_surface_skin() -> void:
	var surface_visual := get_node_or_null("Visuals/FactorySurfaceVisual") as Node2D
	var visual_fill := get_node_or_null("Visuals/Fill") as Polygon2D
	var top_edge := get_node_or_null("Visuals/TopEdge") as Line2D
	var floor_detail := get_node_or_null("FloorDetail") as Line2D

	if surface_visual != null:
		surface_visual.set("surface_size", size)
		surface_visual.set("surface_style", factory_surface_style)
		surface_visual.visible = use_factory_skin
	if visual_fill != null:
		visual_fill.visible = not use_factory_skin
	if top_edge != null:
		top_edge.visible = not use_factory_skin
	if floor_detail != null:
		floor_detail.visible = not use_factory_skin

FILE scripts/world/factory_wall_visual.gd
@tool
extends Node2D

const ATLAS_TILE_SIZE := 32
const DRAW_TILE_SIZE := 64
const SEGMENT_COUNT := 10
const FINAL_SOURCE_HEIGHT := 4
const FINAL_DRAW_HEIGHT := 8

@export var atlas_texture: Texture2D:
	set(value):
		atlas_texture = value
		queue_redraw()

@export var atlas_source_id: int = 0:
	set(value):
		atlas_source_id = value
		queue_redraw()

@export var atlas_coordinates: Vector2i = Vector2i(0, 1):
	set(value):
		atlas_coordinates = value
		queue_redraw()

@export var flip_h: bool = false:
	set(value):
		flip_h = value
		queue_redraw()

func _enter_tree() -> void:
	texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
	queue_redraw()

func _draw() -> void:
	if atlas_texture == null or atlas_source_id != 0 or atlas_coordinates.x < 0 or atlas_coordinates.y < 0:
		return

	var source_x := atlas_coordinates.x * ATLAS_TILE_SIZE
	var source_y := atlas_coordinates.y * ATLAS_TILE_SIZE
	var full_source := Rect2(source_x, source_y, ATLAS_TILE_SIZE, ATLAS_TILE_SIZE)
	if flip_h:
		# Refletir no centro do segmento aponta o acabamento do atlas para dentro.
		draw_set_transform(Vector2(DRAW_TILE_SIZE, 0.0), 0.0, Vector2(-1.0, 1.0))

	for segment_index in SEGMENT_COUNT:
		var destination_y := segment_index * DRAW_TILE_SIZE
		draw_texture_rect_region(
			atlas_texture,
			Rect2(0.0, destination_y, DRAW_TILE_SIZE, DRAW_TILE_SIZE),
			full_source
		)

	# A última peça reutiliza os quatro pixels superiores da célula, escalados por dois.
	draw_texture_rect_region(
		atlas_texture,
		Rect2(0.0, SEGMENT_COUNT * DRAW_TILE_SIZE, DRAW_TILE_SIZE, FINAL_DRAW_HEIGHT),
		Rect2(source_x, source_y, ATLAS_TILE_SIZE, FINAL_SOURCE_HEIGHT)
	)

	if flip_h:
		draw_set_transform(Vector2.ZERO, 0.0, Vector2.ONE)
END FILE

## Seed, aplicação e validação

`room_layout_seed = 0` cria um RNG local randomizado uma vez por instância. Um valor diferente de zero define a seed diretamente e reproduz posições, larguras e alturas. O helper mantém um único RNG para todo o layout. O preparo acontece em `_enter_tree` standalone ou por `configure_run_managed` antes de adicionar a sala ao container; posições, `size` e shapes locais são aplicados antes de `_ready`. Quando a sala managed entra na árvore, o preparo idempotente não sorteia novamente. O seed usado fica registrado no Main.

O gerador limita tentativas a 32 e então retorna fallback seguro declarado. As três posições de fallback são as posições do preview acima. Se ainda não houver layout válido, managed recebe falha de configuração antes de adicionar a sala; standalone desativa gameplay e mostra overlay de erro, preservando os gates existentes.

A validação lê os shapes físicos e AABBs após a aplicação, inclusive a forma local sincronizada pelo setter `SolidPlatform.size` → `_sync_geometry`; shapes retangulares compartilhados são duplicados para a cena local quando necessário. Os AABBs são construídos em coordenadas locais à raiz `Main`, por composição de transforms, sem queries físicas globais. As exports `randomize_position` e `randomize_width` já existentes são desligadas nas instâncias; as cenas serializam ambas como `false`. Confere limites, margens de spawn, corredor do chão, gaps, subida acumulada máxima, ausência de sobreposição, spawns do Player e RivetWorker, patrulha, teto e suporte do alvo. O primário do `TrainingTarget` é o último suporte na ordem X (o mais alto quando há subida; empates permanecem no último elemento); fallback é o primeiro à esquerda. Ambos são conferidos quanto a contato e margem lateral de 8 px. O posicionamento continua determinístico e não usa o RNG deferido do alvo.

As fixtures `Main` e `SecondRoom` usam ambas `scripts/world/main.gd`. A resolução da porta de saída é pelo caminho exato `ExitDoor`; não usa grupo ou procura por tipo que possa selecionar a `EntranceDoor`.

## Portas

Cada fixture contém `EntranceDoor` e `ExitDoor`. A entrada reutiliza `exit_door.tscn`, fica em (112,522), possui sensor desativado, não interage e não conecta sinal de transição; sua apresentação informa entrada bloqueada. A saída reutiliza a mesma cena em (1040,522), conserva o sensor em x=992..1088 e não abre a parede.

`ExitDoor` libera apenas após sala limpa e recompensa concluída. A guarda pós-recompensa mantém a saída fechada até terminar; seleção de recompensa E não aciona porta. Main valida E com estado liberado, clear, jogador correto/vivo, guarda encerrada e jogador no sensor, e repete as condições no deferred. Falha deferred limpa a ocupação para nova interação, sem emitir progressão; se o jogador morreu ou a sala deixou de estar limpa, a porta volta a trancar.

Em modo managed, Main mantém a porta visível e emite `run_progression_ready` no máximo uma vez depois das validações; RunController avança a run preservando HP/build. Falha de layout managed retorna erro antes de adicionar a instância: RunController a libera e chama `_fail_run`, que atualiza o HUD com erro e opção de reinício existente. Não remover a porta nem trocar a cena diretamente. Em Main standalone, manter `RoomTransition` e snapshot explícito de HP/build ao ir para SecondRoom. Em SecondRoom standalone, a saída finaliza a etapa e apresenta `ETAPA CONCLUÍDA`, sem destino vazio, estado `ABRINDO SAÍDA...` permanente ou loop; em managed, segue a progressão normal do controlador.
