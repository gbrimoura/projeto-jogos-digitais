class_name UpgradeOffer
extends CanvasLayer

signal upgrade_selected(upgrade_id: StringName)

const UpgradeCatalogScript = preload("res://scripts/progression/upgrade_catalog.gd")
const UpgradeCardScene: PackedScene = preload("res://scenes/ui/upgrade_card.tscn")
const STAGE_SIZE: Vector2 = Vector2(576.0, 324.0)

@onready var _stage: Control = $Stage
@onready var _heading: HudBitmapText = $Stage/Heading
@onready var _hint: HudBitmapText = $Stage/Hint
@onready var _cards_row: HBoxContainer = $Stage/Cards

var _offered_ids: Array[StringName] = []
var _cards: Array[UpgradeCard] = []
var _font_bundle: Dictionary = {}
var _selection_emitted: bool = false

func _ready() -> void:
	process_mode = Node.PROCESS_MODE_WHEN_PAUSED
	set_process_unhandled_key_input(false)
	_font_bundle = UpgradeCard.load_card_font_bundle()
	if bool(_font_bundle.get("valid", false)):
		_heading.configure_font(_font_bundle)
		_hint.configure_font(_font_bundle)
	_heading.set_text("SALA LIMPA - ESCOLHA UMA MELHORIA")
	_hint.set_text("SELECIONE UMA CARTA")
	var viewport: Viewport = get_viewport()
	if viewport != null and not viewport.size_changed.is_connected(_layout_stage):
		viewport.size_changed.connect(_layout_stage)
	_layout_stage()
	hide()

func present_options(
	options: Array,
	ranks_by_id: Dictionary = {},
	base_maximum_health: int = 5
) -> bool:
	if options.is_empty() or options.size() > 3 or not bool(_font_bundle.get("valid", false)):
		return false
	var next_cards: Array[Dictionary] = []
	var next_ids: Array[StringName] = []
	for option: Variant in options:
		var definition: UpgradeDefinitionResource
		var current_rank: int = 0
		if option is UpgradeDefinitionResource:
			definition = option as UpgradeDefinitionResource
			current_rank = int(ranks_by_id.get(definition.upgrade_id, ranks_by_id.get(String(definition.upgrade_id), 0)))
		elif typeof(option) == TYPE_DICTIONARY:
			var source_row: Dictionary = option
			definition = source_row.get("definition") as UpgradeDefinitionResource
			if definition != null:
				current_rank = int(source_row.get(
					"rank_before",
					ranks_by_id.get(definition.upgrade_id, ranks_by_id.get(String(definition.upgrade_id), 0))
				))
		else:
			return false
		if definition == null or next_ids.has(definition.upgrade_id):
			return false
		var row: Dictionary = UpgradeCatalogScript.card_preview(
			definition.upgrade_id,
			current_rank,
			base_maximum_health
		)
		if row.is_empty():
			return false
		row["definition"] = definition
		row["icon"] = definition.icon
		next_cards.append(row)
		next_ids.append(definition.upgrade_id)

	_clear_cards()
	for index in range(next_cards.size()):
		var card: UpgradeCard = UpgradeCardScene.instantiate() as UpgradeCard
		if card == null:
			_clear_cards()
			return false
		_cards_row.add_child(card)
		if not card.configure(next_cards[index], _font_bundle):
			_cards_row.remove_child(card)
			card.queue_free()
			_clear_cards()
			return false
		card.pressed.connect(_on_card_pressed.bind(index))
		_cards.append(card)

	_offered_ids = next_ids
	_selection_emitted = false
	set_process_unhandled_key_input(true)
	show()
	_layout_stage()
	_cards[0].grab_focus()
	return true

func close_offer() -> void:
	set_process_unhandled_key_input(false)
	for card in _cards:
		if is_instance_valid(card):
			card.disabled = true
	hide()

func _unhandled_key_input(event: InputEvent) -> void:
	if event.is_action_pressed("interact") or event.is_action_pressed(&"ui_cancel"):
		get_viewport().set_input_as_handled()

func _on_card_pressed(index: int) -> void:
	if _selection_emitted or index < 0 or index >= _offered_ids.size():
		return
	_selection_emitted = true
	for card in _cards:
		card.disabled = true
	upgrade_selected.emit(_offered_ids[index])

func _clear_cards() -> void:
	_offered_ids.clear()
	_cards.clear()
	for child: Node in _cards_row.get_children():
		_cards_row.remove_child(child)
		child.queue_free()

func _layout_stage() -> void:
	if _stage == null or get_viewport() == null:
		return
	var viewport_size: Vector2 = get_viewport().get_visible_rect().size
	var integer_scale: int = maxi(floori(minf(
		viewport_size.x / STAGE_SIZE.x,
		viewport_size.y / STAGE_SIZE.y
	)), 1)
	var scaled_size: Vector2 = STAGE_SIZE * float(integer_scale)
	_stage.scale = Vector2.ONE * float(integer_scale)
	_stage.position = ((viewport_size - scaled_size) * 0.5).floor()

static func load_card_font_bundle() -> Dictionary:
	if bool(_cached_card_font_bundle.get("valid", false)):
		return _cached_card_font_bundle.duplicate(true)
	var bundle: Dictionary = HudBitmapTextScript.load_font_bundle()
	if not bool(bundle.get("valid", false)):
		return bundle
	var atlas: Texture2D = ResourceLoader.load(CARD_FONT_ATLAS_PATH) as Texture2D
	var additions_data: Variant = JSON.parse_string(FileAccess.get_file_as_string(CARD_FONT_ADDITIONS_PATH))
	if atlas == null or typeof(additions_data) != TYPE_DICTIONARY:
		return {"valid": false, "reason": "atlas local ou adições da fonte de cartas inválidos"}
	var additions: Dictionary = additions_data
	var glyphs: Dictionary = bundle.get("glyphs", {})
	var raw_glyphs: Variant = additions.get("glyphs", [])
	if typeof(raw_glyphs) != TYPE_ARRAY:
		return {"valid": false, "reason": "lista de glifos locais inválida"}
	for glyph_variant: Variant in raw_glyphs:
		if typeof(glyph_variant) != TYPE_DICTIONARY:
			continue
		var glyph_data: Dictionary = glyph_variant
		var character: String = String(glyph_data.get("character", ""))
		var rect_variant: Variant = glyph_data.get("rect", null)
		if character.is_empty() or typeof(rect_variant) != TYPE_DICTIONARY:
			continue
		var rect_data: Dictionary = rect_variant
		var glyph_rect := Rect2i(
			int(rect_data.get("x", -1)),
			int(rect_data.get("y", -1)),
			int(rect_data.get("w", 0)),
			int(rect_data.get("h", 0))
		)
		if glyph_rect.position.x < 0 or glyph_rect.position.y < 0 \
			or glyph_rect.size.x <= 0 or glyph_rect.size.y <= 0 \
			or glyph_rect.end.x > atlas.get_width() or glyph_rect.end.y > atlas.get_height():
			continue
		glyphs[character] = {
			"rect": glyph_rect,
			"advance": maxi(int(glyph_data.get("advance", 6)), 1),
		}
	bundle["atlas"] = atlas
	bundle["glyphs"] = glyphs
	_cached_card_font_bundle = bundle.duplicate(true)
	return _cached_card_font_bundle.duplicate(true)

func configure(row: Dictionary, font_bundle: Dictionary) -> bool:
	if not bool(font_bundle.get("valid", false)):
		return false
	var definition: UpgradeDefinitionResource = row.get("definition") as UpgradeDefinitionResource
	if definition == null:
		return false
	var illustration: Texture2D = row.get("icon", definition.icon) as Texture2D
	if illustration == null:
		return false
	if not _configure_text(_title, String(row.get("title", definition.title)), font_bundle, 1):
		return false
	var summary_lines := _wrap_text(String(row.get("effect_summary", "")), 24, 2)
	if summary_lines.size() != 1 and summary_lines.size() != 2:
		return false
	if not _configure_text(_summary, "\n".join(PackedStringArray(summary_lines)), font_bundle, 2):
		return false
	var topic_lines: Array[String] = []
	var topics_variant: Variant = row.get("topics", definition.effect_topics)
	if typeof(topics_variant) == TYPE_PACKED_STRING_ARRAY or typeof(topics_variant) == TYPE_ARRAY:
		for topic_variant: Variant in topics_variant:
			var wrapped_topic := _wrap_text("- " + String(topic_variant), 24, 3)
			for line: String in wrapped_topic:
				topic_lines.append(line)
	if topic_lines.size() < 2 or topic_lines.size() > _topic_lines.size():
		return false
	for index in range(_topic_lines.size()):
		var line := topic_lines[index] if index < topic_lines.size() else ""
		if not _configure_text(_topic_lines[index], line, font_bundle, 1):
			return false
	if not _configure_text(_badge, String(row.get("rank_label", "")), font_bundle, 1):
		return false
	_illustration.texture = illustration
	_illustration.show()
	return true

func _configure_text(control: HudBitmapText, value: String, font_bundle: Dictionary, lines: int) -> bool:
	if control == null or not control.configure_font(font_bundle):
		return false
	control.max_lines = lines
	control.set_text(value)
	return true

func _wrap_text(value: String, maximum_characters: int, maximum_lines: int) -> Array[String]:
	var result: Array[String] = []
	var current := ""
	for word: String in value.replace("\n", " ").split(" ", false):
		if word.length() > maximum_characters:
			return []
		var candidate := word if current.is_empty() else current + " " + word
		if candidate.length() <= maximum_characters:
			current = candidate
			continue
		result.append(current)
		current = word
		if result.size() >= maximum_lines:
			return []
	if not current.is_empty():
		result.append(current)
	if result.size() > maximum_lines:
		return []
	return result

    assert(#got==#EXPECT,"palette size changed")
    local cs={}
    for i,h in ipairs(EXPECT) do assert(got[i]==h,"palette/order drift at "..i); cs[i]=pc.rgba(tonumber(h:sub(1,2),16),tonumber(h:sub(3,4),16),tonumber(h:sub(5,6),16),255) end
    local K={ink=cs[1],navy=cs[2],blue=cs[3],blue2=cs[4],iron=cs[5],steelDark=cs[6],steel=cs[7],steelLight=cs[8],silver=cs[9],rustDark=cs[10],rust=cs[11],copper=cs[12],amber=cs[13],cream=cs[14],ivory=cs[15],red=cs[16],redDark=cs[17],cyan=cs[18],white=cs[19]}
    return cs,K
  end
  local function setPal(spr,cs)
    local p=Palette(#cs)
    for i,c in ipairs(cs) do p:setColor(i-1,Color(pc.rgbaR(c),pc.rgbaG(c),pc.rgbaB(c),255)) end
    spr:setPalette(p)
  end

  local function px(im,x,y,c) if x>=0 and y>=0 and x<im.width and y<im.height then im:drawPixel(x,y,c) end end
  local function rect(im,x,y,w,h,c)
    if w<=0 or h<=0 then return end
    for yy=math.max(0,y),math.min(im.height-1,y+h-1) do for xx=math.max(0,x),math.min(im.width-1,x+w-1) do im:drawPixel(xx,yy,c) end end
  end
  local function line(im,x0,y0,x1,y1,c,width)
    width=width or 1
    local dx,sx=math.abs(x1-x0),x0<x1 and 1 or -1; local dy,sy=-math.abs(y1-y0),y0<y1 and 1 or -1; local err=dx+dy
    while true do
      local a=math.floor((width-1)/2); rect(im,x0-a,y0-a,width,width,c)
      if x0==x1 and y0==y1 then break end
      local e=2*err; if e>=dy then err=err+dy; x0=x0+sx end; if e<=dx then err=err+dx; y0=y0+sy end
    end
  end
  local function disk(im,cx,cy,r,c)
    for yy=-r,r do local hw=math.floor(math.sqrt(math.max(0,r*r-yy*yy))); rect(im,cx-hw,cy+yy,hw*2+1,1,c) end
  end
  local function ring(im,cx,cy,ro,ri,c)
    for y=-ro,ro do for x=-ro,ro do local d=x*x+y*y; if d<=ro*ro and d>=ri*ri then px(im,cx+x,cy+y,c) end end end
  end
  local function poly(im,pts,fill,outline)
    local y0,y1=pts[1].y,pts[1].y
    for _,p in ipairs(pts) do y0=math.min(y0,p.y); y1=math.max(y1,p.y) end
    for y=y0,y1 do
      local xs={}
      for i=1,#pts do local a,b=pts[i],pts[i%#pts+1]; if (a.y<=y and b.y>y) or (b.y<=y and a.y>y) then xs[#xs+1]=a.x+(y-a.y)*(b.x-a.x)/(b.y-a.y) end end
      table.sort(xs)
      for i=1,#xs-1,2 do for x=math.ceil(xs[i]),math.floor(xs[i+1]) do px(im,x,y,fill) end end
    end
    if outline then for i=1,#pts do local a,b=pts[i],pts[i%#pts+1]; line(im,a.x,a.y,b.x,b.y,outline,1) end end
  end
  local function paste(dst,src,x,y) dst:drawImage(src,x,y) end

local function main()
  local P=app.params or {}
  local ROOT=assert(P.root,"missing root parameter")
  local KIT=ROOT.."/assets/ui/upgrade_cards_v1"

  local OUT=KIT.."/export"

  local function read(path)
    local f=assert(io.open(path,"rb"),"cannot read "..path)
    local s=f:read("*a"); f:close(); return s
  end
  local function write(path,s)
    local f=assert(io.open(path,"wb"),"cannot write "..path)
    f:write(s); f:close()
  end
  local function img(w,h) return Image(w,h) end
  local function px(im,x,y,c)
    assert(x>=0 and y>=0 and x<im.width and y<im.height,"pixel out of range")
    im:drawPixel(x,y,c)
  end
  local function rect(im,x,y,w,h,c)
    for yy=y,y+h-1 do for xx=x,x+w-1 do px(im,xx,yy,c) end end
  end

  local baseJson=read(BASE.."/font.json")
  assert(baseJson:find('"cell":{"w":5,"h":11}',1,true),"source font cell changed")

## Single source for run upgrade definitions, caps, derived effects and card copy.
## This file intentionally does not preload or reference RunBuild.

const MAX_RUN_ACQUISITIONS: int = 220
const BASE_PROJECTILE_CONFIG: ProjectileConfig = preload("res://resources/projectile_config.tres")
const BASE_MOVEMENT_CONFIG: MovementConfig = preload("res://resources/player_movement_config.tres")

static var _definitions: Array[UpgradeDefinitionResource] = [
	preload("res://resources/upgrades/impacto_reforcado.tres"),
	preload("res://resources/upgrades/motor_acelerado.tres"),
	preload("res://resources/upgrades/carga_reativa.tres"),

	preload("res://resources/upgrades/dinamo_defesa.tres"),
	preload("res://resources/upgrades/chassi_reforcado.tres"),
]


static func definitions() -> Array[UpgradeDefinitionResource]:
	var result: Array[UpgradeDefinitionResource] = []
	for definition in _definitions:
		result.append(definition)
	return result


static func known_upgrade_ids() -> Array[StringName]:
	var result: Array[StringName] = []
	for definition in _definitions:
		if definition != null and not String(definition.upgrade_id).is_empty():
			result.append(definition.upgrade_id)
	return result


static func is_known_upgrade_id(upgrade_id: Variant) -> bool:
	return definition_for(upgrade_id) != null


static func definition_for(upgrade_id: Variant) -> UpgradeDefinitionResource:
	if typeof(upgrade_id) != TYPE_STRING and typeof(upgrade_id) != TYPE_STRING_NAME:
		return null
	var requested_id := StringName(upgrade_id)
	for definition in _definitions:
		if definition != null and definition.upgrade_id == requested_id:
			return definition
	return null


static func max_rank_for(upgrade_id: Variant, base_maximum_health: int = 5) -> int:
	var definition := definition_for(upgrade_id)
	if definition == null:
		return 0
	var cap: int = clampi(definition.rank_cap, 0, MAX_RUN_ACQUISITIONS)
	if definition.maximum_health_per_rank > 0:
		return clampi(100 - maxi(base_maximum_health, 1), 0, cap)
	return cap


static func can_acquire(upgrade_id: Variant, current_rank: int, base_maximum_health: int = 5) -> bool:
	return is_known_upgrade_id(upgrade_id) and current_rank >= 0 \
		and current_rank < max_rank_for(upgrade_id, base_maximum_health)


static func total_rank_count(ranks_by_id: Dictionary) -> int:
	var total: int = 0
	for upgrade_id in known_upgrade_ids():
		total += maxi(int(ranks_by_id.get(upgrade_id, ranks_by_id.get(String(upgrade_id), 0))), 0)
	return total


static func build_offer(
	ranks_by_id: Dictionary,
	room_seed: int,
	room_ordinal: int,
	base_maximum_health: int = 5
) -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	if total_rank_count(ranks_by_id) >= MAX_RUN_ACQUISITIONS:
		return result

	var new_candidates: Array[UpgradeDefinitionResource] = []
	var evolution_candidates: Array[UpgradeDefinitionResource] = []
	for definition in _definitions:
		if definition == null:
			continue
		var rank := _rank_for(ranks_by_id, definition.upgrade_id)
		if not can_acquire(definition.upgrade_id, rank, base_maximum_health):
			continue
		if rank == 0:
			new_candidates.append(definition)
		else:
			evolution_candidates.append(definition)

	var rng := RandomNumberGenerator.new()
	rng.seed = _offer_seed(room_seed, room_ordinal)
	var selected_definitions: Array[UpgradeDefinitionResource] = []
	if not new_candidates.is_empty() and not evolution_candidates.is_empty():
		selected_definitions.append(_take_random(new_candidates, rng))
		selected_definitions.append(_take_random(evolution_candidates, rng))

	var remaining: Array[UpgradeDefinitionResource] = []
	for definition in new_candidates:
		if not selected_definitions.has(definition):
			remaining.append(definition)
	for definition in evolution_candidates:
		if not selected_definitions.has(definition):
			remaining.append(definition)
	while selected_definitions.size() < 3 and not remaining.is_empty():
		selected_definitions.append(_take_random(remaining, rng))

	for definition in selected_definitions:
		var current_rank := _rank_for(ranks_by_id, definition.upgrade_id)
		var card := card_preview(definition.upgrade_id, current_rank, base_maximum_health)
		card["definition"] = definition
		result.append(card)
	return result


static func card_preview(
	upgrade_id: Variant,
	current_rank: int,
	base_maximum_health: int = 5
) -> Dictionary:
	var definition := definition_for(upgrade_id)
	if definition == null or not can_acquire(upgrade_id, current_rank, base_maximum_health):
		return {}
	var next_rank: int = current_rank + 1
	var rank_label := "NOVO" if current_rank == 0 else "NV.%d > %d" % [current_rank, next_rank]
	return {
		"upgrade_id": definition.upgrade_id,
		"definition": definition,
		"title": definition.title,
		"description": definition.description,
		"rank_before": current_rank,
		"rank_after": next_rank,
		"rank_label": rank_label,
		"effect_summary": effect_summary(upgrade_id, current_rank, next_rank, base_maximum_health),
		"topics": definition.effect_topics.duplicate(),
		"icon": definition.icon,
	}


static func effect_summary(
	upgrade_id: Variant,
	current_rank: int,
	next_rank: int,
	base_maximum_health: int = 5
) -> String:
	var definition := definition_for(upgrade_id)
	if definition == null:
		return ""
	match definition.upgrade_id:
		&"impacto_reforcado":
			var base_damage: int = maxi(BASE_PROJECTILE_CONFIG.damage, 1)
			return "%d > %d dano" % [base_damage + current_rank * definition.projectile_damage_per_rank, base_damage + next_rank * definition.projectile_damage_per_rank]
		&"motor_acelerado":
			var current_cooldown := dash_cooldown_for_rank(current_rank)
			var next_cooldown := dash_cooldown_for_rank(next_rank)
			return "%s > %s s" % [_format_decimal(current_cooldown), _format_decimal(next_cooldown)]
		&"carga_reativa":
			return "+%d > +%d dano" % [current_rank * definition.charged_damage_bonus_per_rank, next_rank * definition.charged_damage_bonus_per_rank]
		&"ricochete_pressao":
			return "+%d > +%d/parede" % [current_rank * definition.environmental_bounce_damage_per_rank, next_rank * definition.environmental_bounce_damage_per_rank]
		&"mola_retorno":
			var current_bounces: int = BASE_PROJECTILE_CONFIG.max_bounces + current_rank * definition.max_bounces_per_rank
			var next_bounces: int = BASE_PROJECTILE_CONFIG.max_bounces + next_rank * definition.max_bounces_per_rank
			var current_lifetime: float = BASE_PROJECTILE_CONFIG.lifetime + current_rank * definition.lifetime_per_rank
			var next_lifetime: float = BASE_PROJECTILE_CONFIG.lifetime + next_rank * definition.lifetime_per_rank
			return "%d > %d ric.; %s > %s s" % [current_bounces, next_bounces, _format_decimal(current_lifetime), _format_decimal(next_lifetime)]
		&"dinamo_defesa":
			return "%s > %s s por parry" % [
				_format_decimal(current_rank * definition.dash_cooldown_refund_per_rank),
				_format_decimal(next_rank * definition.dash_cooldown_refund_per_rank),
			]
		&"chassi_reforcado":
			var base_health: int = maxi(base_maximum_health, 1)
			var current_health: int = mini(base_health + current_rank * definition.maximum_health_per_rank, 100)
			var next_health: int = mini(base_health + next_rank * definition.maximum_health_per_rank, 100)
			return "%d > %d HP max; cura +%d" % [current_health, next_health, definition.health_recovered_per_rank]
	return definition.effect_summary


static func projectile_damage(
	base_damage: int,
	impact_rank: int,
	charge_rank: int,
	charged: bool,
	pressure_rank: int,
	environmental_bounce_count: int
) -> int:
	var damage: int = maxi(base_damage, 0)
	var impact_definition := definition_for(&"impacto_reforcado")
	var charge_definition := definition_for(&"carga_reativa")
	var pressure_definition := definition_for(&"ricochete_pressao")
	if impact_definition != null:
		damage += maxi(impact_rank, 0) * impact_definition.projectile_damage_per_rank
	if charged and charge_definition != null and charge_definition.reactive_charge_enabled:
		damage += maxi(charge_rank, 0) * charge_definition.charged_damage_bonus_per_rank
	if pressure_definition != null:
		var counted_bounces: int = mini(maxi(environmental_bounce_count, 0), pressure_definition.environmental_bounce_damage_count_cap)
		damage += maxi(pressure_rank, 0) * counted_bounces * pressure_definition.environmental_bounce_damage_per_rank
	return maxi(damage, 0)


static func dash_cooldown_for_rank(rank: int) -> float:

	var definition := definition_for(&"motor_acelerado")
	if definition == null or rank <= 0 or definition.dash_cooldown_by_rank.is_empty():
		return base_cooldown
	var index: int = mini(rank, definition.dash_cooldown_by_rank.size()) - 1
	return maxf(definition.dash_cooldown_by_rank[index], 0.0)

	if definition == null:

	var definition := definition_for(upgrade_id)
	if definition == null or definition.maximum_health_per_rank <= 0:
		return 0
	return maxi(next_rank - previous_rank, 0) * definition.health_recovered_per_rank

static func _take_random(candidates: Array[UpgradeDefinitionResource], rng: RandomNumberGenerator) -> UpgradeDefinitionResource:
	var index: int = rng.randi_range(0, candidates.size() - 1)
	return candidates.pop_at(index)


static func _offer_seed(room_seed: int, room_ordinal: int) -> int:
	var seed_value: int = (room_seed * 1103515245 + room_ordinal * 12345 + 0x5F3759DF) & 0x7FFFFFFF
	return seed_value if seed_value != 0 else 1


static func _format_decimal(value: float) -> String:
	var formatted := "%.2f" % value
	while formatted.ends_with("0"):
		formatted = formatted.left(formatted.length() - 1)
	if formatted.ends_with("."):
		formatted = formatted.left(formatted.length() - 1)
	return formatted.replace(".", ",")
